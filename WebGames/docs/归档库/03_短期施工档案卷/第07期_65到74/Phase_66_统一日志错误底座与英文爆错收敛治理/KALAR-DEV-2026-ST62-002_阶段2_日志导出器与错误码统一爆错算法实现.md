## 一、 核心算法与业务实现 (Core Algorithms & Implementation)

### 1.1 统一日志导出器 (`LogFileExporter`)

```gdscript
# 模块路径: res://backend/infrastructure/log_file_exporter.gd
class_name LogFileExporter extends RefCounted:

    static var _instance: LogFileExporter = null
    static var _config: LogExportConfigDTO = null
    static var _config_version: int = -1
    static var _sequence: int = 0            # 全局单调序号（与 LogRecordDTO.sequence 同源）
    static var _exported_count: int = 0      # 已落盘记录数（Inv-LG-3 对账）

    ## 单例 + 配置新鲜度守卫（热重载版本推进重建）
    static func get_instance() -> LogFileExporter:
        if _instance == null:
            _instance = new()
        _ensure_config_fresh()
        return _instance

    static func _ensure_config_fresh() -> void:
        if _config != null and _config_version == GameConfig.config_reload_version():
            return
        _config = LogExportConfigDTO.from_config()
        _config_version = GameConfig.config_reload_version()

    ## 旁路消费入口：广播层每发一条日志即调用一次（级别/通道过滤 → 原子追加）
    static func consume(record: LogRecordDTO) -> bool:
        if record == null:
            return false
        _ensure_config_fresh()
        if not _config.enabled:
            return false
        if not _level_allowed(record.level) or not _channel_allowed(record.channel):
            return false
        _sequence += 1
        record.sequence = _sequence
        return _append_atomic(record)

    static func _level_allowed(level: String) -> bool:
        # 级别权重序：debug < info < warn < error；record 级别 >= min_level 才落盘
        return _level_weight(level) >= _level_weight(_config.min_level)

    static func _level_weight(level: String) -> int:
        match level.to_upper():
            "DEBUG": return 0
            "INFO": return 1
            "WARN": return 2
            "ERROR": return 3
            _: return 1

    static func _channel_allowed(channel: String) -> bool:
        return _config.include_channels.is_empty() or channel in _config.include_channels

    ## 原子追加：行序列化 → 追加写临时文件 → flush → close → rename 原子替换
    static func _append_atomic(record: LogRecordDTO) -> bool:
        var line := JSON.stringify(record.to_dto())
        var dir := _config.directory
        if dir.is_empty():
            return false
        if not DirAccess.open(dir) != null:
            DirAccess.make_dir_recursive_absolute(dir)
        var base := "%s/%s" % [dir, _config.file_prefix]
        var target := _rotate_target_if_needed(base)
        var tmp := target + ".tmp"
        var f := FileAccess.open(tmp, FileAccess.WRITE)
        if f == null:
            return false
        f.store_line(line)
        f.flush()
        f.close()
        var err := DirAccess.rename_absolute(tmp, target)
        if err != OK:
            return false
        _exported_count += 1
        return true

    ## 轮转：当前主文件超 max_bytes 时按时间戳滚动为归档名，主文件重建（max_files 清理最旧）
    static func _rotate_target_if_needed(base: String) -> String:
        var main_path := base + ".log"
        if _config.max_bytes > 0 and FileAccess.file_exists(main_path) \
                and FileAccess.get_file_as_string(main_path).length() >= _config.max_bytes:
            var stamp := str(Time.get_unix_time_from_system())
            DirAccess.rename_absolute(main_path, "%s.%s.log" % [base, stamp])
            _prune_rotated(base)
        return main_path

    static func _prune_rotated(base: String) -> void:
        # 保留最近 max_files 份轮转文件，超出删除最旧（文件名含时间戳，字典序即时间序）
        if _config.max_files <= 0:
            return
        var rotated: Array = []
        var dir := DirAccess.open(_config.directory)
        if dir == null:
            return
        for f in dir.get_files():
            if f.begins_with(_config.file_prefix + ".") and f.ends_with(".log") and f != _config.file_prefix + ".log":
                rotated.append(f)
        rotated.sort()
        while rotated.size() > _config.max_files:
            DirAccess.remove_absolute("%s/%s" % [_config.directory, rotated[0]])
            rotated.remove_at(0)
```

**算法要点**：
- **旁路消费**：导出器订阅 `log_message_posted`（或由 `EventBus.emit_log_record` 直接调用），既有前端渲染路径零改动；
- **原子性**：tmp→flush→close→rename 复用 `SaveManager._atomic_write` 同款模式，任一步失败不破坏既有主文件（Inv-LG-3）；
- **轮转截断**：超 `max_bytes` 滚动归档 + `max_files` 清理最旧（Inv-LG-4 配置驱动）；
- **对账**：`_sequence` 与 `_exported_count` 提供导出对账断言（阶段4 测试比对广播序与文件行序）。

### 1.2 统一爆错收敛算法 (`ErrorReporter`)

```gdscript
# 模块路径: res://backend/infrastructure/error_reporter.gd
class_name ErrorReporter extends RefCounted:

    ## 统一错误上报入口：构造带 error_code 的 LogRecordDTO 并走统一通道广播
    ## error_code 必须为 errors_catalog.json 已登记常量（未登记回退 ERR_UNKNOWN 并计数）
    static func emit_error(
        channel: String,
        error_code: String,
        message: String = "",
        context: Dictionary = {}
    ) -> void:
        var entry := ErrorCodeRegistry.lookup(error_code)
        if entry == null:
            ErrorCodeRegistry.report_unregistered(error_code)   # 遥测计数（审查 L4 断点接通）
            error_code = "ERR_UNKNOWN"
        var record := LogRecordDTO.new()
        record.level = entry.severity if entry != null else "error"
        record.channel = channel
        record.error_code = error_code
        record.message = _resolve_message(error_code, message)
        record.context = context
        EventBus.emit_log_record(record)

    ## message 解析：优先 i18n（narratives.errors 表按 error_code 取键），缺省回退调用方传入文本或 code 本体
    static func _resolve_message(error_code: String, fallback: String) -> String:
        var key := "narratives.errors." + error_code
        var text := GameConfig.get_string(key, "", "")
        if not text.is_empty():
            return text
        return fallback if not fallback.is_empty() else error_code

    ## 存量 push_error/push_warning 收敛包装：保持失败关闭语义，仅换通道
    ## 用法：ErrorReporter.emit_error(channel, code, msg) 替代 push_error("Domain: msg")
    static func migrate_push_error(channel: String, msg: String) -> void:
        emit_error(channel, "ERR_UNKNOWN", msg, {})
```

**算法要点**：
- **错误码登记断言**：`lookup` 未命中 → 回退 `ERR_UNKNOWN` + `report_unregistered` 遥测计数（接通审查 L4 断点）；
- **i18n message**：narratives.errors 表驱动（Inv-LG-2），未登记回退调用方文本；
- **收敛包装**：`migrate_push_error` 让存量 `push_error` 调用点以最小 diff 换通道（阶段3 逐文件迁移清单执行）。

### 1.3 `EventBus` 统一通道扩展

```gdscript
# backend/infrastructure/event_bus.gd 增量扩展（不破坏既有签名）
# 新增：信号 log_record_emitted(record: LogRecordDTO)
# 新增：static var _record_seq: int = 0   # 统一序列（导出器/广播共用）

func emit_log_record(record: LogRecordDTO) -> void:
    if record == null:
        return
    _record_seq += 1
    record.sequence = _record_seq
    log_record_emitted.emit(record)
    # 兼容既有消费：同步转发 log_message_posted（前端战报渲染零改动）
    log_message_posted.emit(get_log_level(record.level), record.message)
    # 旁路导出（配置关闭时零开销）
    LogFileExporter.consume(record)

func emit_log(level_key: String, msg: String) -> void:
    # 旧签名保持：构造最小 record 委托 emit_log_record（channel 缺省 ""）
    var record := LogRecordDTO.new()
    record.level = level_key
    record.message = msg
    emit_log_record(record)
```

**兼容性保证**：`emit_log` 签名与 `log_message_posted` 信号完全保留，既有 83 套测试零改动；新增 `emit_log_record`/`log_record_emitted` 为纯增量。
