## 一、 核心算法与业务实现 (Core Algorithms & Implementation)

### 1.1 统一采集管线 (`LogCollector`)

```gdscript
# 模块路径: res://backend/infrastructure/log_collector.gd
class_name LogCollector extends RefCounted:

    static var _active_trace_id: String = ""   # 当前业务链路的 trace_id（入口签发）
    static var _seq: int = 0                    # 记录序号（与 EventBus 导出器同源语义）

    ## 业务入口调用：签发新 trace_id（战斗回合开始/交易流水开启/停机管线启动）
    static func begin_trace() -> String:
        _active_trace_id = _new_trace_id()
        return _active_trace_id

    static func _new_trace_id() -> String:
        # 确定性可复现：复用 DeterministicRNG 生成 8 字节十六进制（无入口回退 channel+时间戳派生）
        return "TRACE_%s" % _hex_random()

    ## 统一采集入口：构造结构化记录（构造期脱敏 + 字段补全）并走 EventBus 广播
    static func log(
        level: String,
        channel: String,
        message: String,
        context: Dictionary = {},
        error_code: String = "",
        duration_ms: int = 0,
        event_category: String = ""
    ) -> void:
        var record := StructuredLogRecord.new()
        _seq += 1
        record.sequence = _seq
        record.timestamp_utc = int(Time.get_unix_time_from_system())
        record.level = level
        record.channel = channel
        record.message = message
        record.error_code = error_code
        record.duration_ms = duration_ms
        record.event_category = event_category
        record.trace_id = _active_trace_id if not _active_trace_id.is_empty() else _derived_trace_id(channel)
        # 构造期脱敏（Inv-LS-2）：context 命中 redaction 规则的键在入流前掩码
        var redacted := RedactionRule.apply(context)
        record.context = redacted["dict"]
        record.redacted_keys = redacted["redacted_keys"]
        record.source_file = _caller_file()
        record.source_line = _caller_line()
        EventBus.emit_log_record(record)

    static func _caller_file() -> String:
        # 经 get_stack() 取调用方脚本路径（仅 basename，不含机器绝对路径）
        pass

    static func _caller_line() -> int:
        pass
```

**算法要点**：
- **入口签发**（Inv-LS-3）：`begin_trace()` 在业务入口（战斗回合/交易流水/停机管线）调用，后续 `log()` 自动继承 `_active_trace_id`；
- **构造期脱敏**（Inv-LS-2）：`RedactionRule.apply(context)` 在 `record` 入流前执行，`redacted_keys` 留痕——落盘/缓冲/检索只见掩码值；
- **字段补全**：source_file（仅 basename）/source_line 经 `get_stack()` 提取，无盘符绝对路径（遵守 LAYOUT/LINK 门禁精神）；
- **兼容**：P66 `EventBus.emit_log` 保持原签名，内部委托 `LogCollector.log`（channel 缺省），既有调用方零改动。

### 1.2 环形缓冲日志流 (`LogRingBuffer`)

```gdscript
# 模块路径: res://backend/infrastructure/log_ring_buffer.gd
class_name LogRingBuffer extends RefCounted:

    static var _buffer: Array[StructuredLogRecord] = []
    static var _capacity: int = 1000        # 默认容量（log.json structured/ring_buffer_capacity 驱动）
    static var _head: int = 0               # 覆盖写游标

    ## 容量配置新鲜度守卫（热重载版本推进重建）
    static func _ensure_capacity_fresh() -> void:
        var cap := GameConfig.get_int("infrastructure.log", "structured/ring_buffer_capacity", 1000)
        if cap != _capacity:
            _capacity = maxi(1, cap)
            _rebuild()

    ## 追加一条记录（满则覆盖最旧，有界收敛）
    static func append(record: StructuredLogRecord) -> void:
        _ensure_capacity_fresh()
        if record == null:
            return
        if _buffer.size() < _capacity:
            _buffer.append(record)
        else:
            _buffer[_head] = record
            _head = (_head + 1) % _capacity

    ## 按条件扫描（只读；供检索/运维终端使用）
    static func query(from_utc: int, levels: Array, channels: Array) -> Array:
        var out: Array = []
        for rec in _buffer:
            if rec == null:
                continue
            if from_utc > 0 and rec.timestamp_utc < from_utc:
                continue
            if not levels.is_empty() and not (rec.level in levels):
                continue
            if not channels.is_empty() and not (rec.channel in channels):
                continue
            out.append(rec)
        return out

    static func _rebuild() -> void:
        # 容量变更时保留最近 _capacity 条（尾段截取）
        if _buffer.size() > _capacity:
            _buffer = _buffer.slice(_buffer.size() - _capacity)
            _head = 0
```

**算法要点**：
- **有界收敛**：容量配置驱动（默认 1000），满则覆盖最旧——内存零无界增长（对齐 Phase 56 L8 有界惯例）；
- **只读查询**：`query` 供检索合并源之一（实时窗口），不写容器（Inv-LS-5）；
- **热重载重建**：容量变更保留最近 N 条，不丢冷数据（变更语义显式）。

### 1.3 遥测聚合算法 (`TelemetryAggregate.consume`)

```gdscript
# backend/infrastructure/telemetry_aggregate.gd 增量实现
static func consume(record: StructuredLogRecord) -> void:
    if record == null:
        return
    _maybe_roll_window(int(Time.get_unix_time_from_system()))
    var key := "%s|%s" % [record.channel, record.event_category]
    var slot: SlotMetric = _slots.get(key, null)
    if slot == null:
        slot = SlotMetric.new()
        slot.domain = record.channel
        slot.event_category = record.event_category
        slot.last_timestamp_utc = record.timestamp_utc
        _slots[key] = slot
    slot.count += 1
    slot.total_duration_ms += record.duration_ms
    slot.last_timestamp_utc = record.timestamp_utc
    _track_duration(slot, record.duration_ms)   # 环形分位缓冲（p50/p95）

static func _track_duration(slot: SlotMetric, duration_ms: int) -> void:
    # 内部保留最近 N 条耗时（log.json telemetry/sample_size，默认 256），排序取分位
    slot._duration_samples.append(duration_ms)
    if slot._duration_samples.size() > _sample_size:
        slot._duration_samples.remove_at(0)
    var sorted_samples := slot._duration_samples.duplicate()
    sorted_samples.sort()
    slot.p50_ms = sorted_samples[int(sorted_samples.size() * 0.50)]
    slot.p95_ms = sorted_samples[int(sorted_samples.size() * 0.95)]

static func _maybe_roll_window(now_utc: int) -> void:
    var window := GameConfig.get_int("infrastructure.log", "telemetry/window_seconds", 60)
    if _window_start_utc == 0:
        _window_start_utc = now_utc
        return
    if now_utc - _window_start_utc >= window:
        _slots.clear()
        _window_start_utc = now_utc
```

**算法要点**：
- **单向消费**（Inv-LS-4）：`consume` 仅读 `record` 聚合计数/耗时，绝不修改 `record` 级别/内容；
- **分位耗时**：滑动样本窗（默认 256 条）排序取 p50/p95——热路径（战斗回合）承压分布可观测；
- **窗口滑动**：配置驱动（默认 60s），满窗重置槽位，内存有界。

### 1.4 日志检索服务 (`LogQueryService`)

```gdscript
# 模块路径: res://backend/infrastructure/log_query_service.gd
class_name LogQueryService extends RefCounted:

    const MAX_RESULTS: int = 5000       # 命中上限（配置驱动前先以常量声明，阶段3 落 log.json）

    ## 六维过滤检索：环形缓冲（实时）+ 导出文件（历史）双源合并，只读
    static func query(criteria: LogQueryDTO.Criteria) -> LogQueryDTO.Result:
        var res := LogQueryDTO.Result.new()
        var page := maxi(1, criteria.page)
        var page_size := clampi(criteria.page_size, 1, 500)
        var all: Array = []
        # 源1：环形缓冲实时记录
        all.append_array(LogRingBuffer.query(criteria.from_utc, criteria.levels, criteria.channels))
        # 源2：导出文件历史记录（NDJSON 逐行解析 + 过滤；不读则跳过）
        if _export_file_exists():
            all.append_array(_query_export_files(criteria))
        # 统一过滤：trace_id/error_code/keyword + 时间窗
        var filtered: Array = []
        for rec in all:
            if not _match(rec, criteria):
                continue
            filtered.append(rec)
        res.total = filtered.size()
        res.truncated = filtered.size() > MAX_RESULTS
        var start := (page - 1) * page_size
        var end := mini(start + page_size, filtered.size())
        for i in range(start, mini(end, MAX_RESULTS)):
            res.records.append(filtered[i].to_dto())
        return res

    static func _match(rec: StructuredLogRecord, c: LogQueryDTO.Criteria) -> bool:
        if c.from_utc > 0 and rec.timestamp_utc < c.from_utc:
            return false
        if c.to_utc > 0 and rec.timestamp_utc > c.to_utc:
            return false
        if not c.trace_id.is_empty() and rec.trace_id != c.trace_id:
            return false
        if not c.error_code.is_empty() and rec.error_code != c.error_code:
            return false
        if not c.keyword.is_empty() and not rec.message.contains(c.keyword):
            return false
        return true

    static func _query_export_files(c: LogQueryDTO.Criteria) -> Array:

        pass
```

**算法要点**：
- **双源合并**：实时环形缓冲（近 N 条）+ 导出文件（历史持久）——故障定位既有热数据又有冷数据；
- **只读**（Inv-LS-5）：全程只读内存与文件，零写操作；
- **分页截断**：命中超 `MAX_RESULTS` 置 `truncated`，分页遍历不越界（`mini(end, MAX_RESULTS)`）；
- **文件解析**：NDJSON 逐行 JSON.parse，解析失败行跳过（宽容读取，不阻断检索）。
