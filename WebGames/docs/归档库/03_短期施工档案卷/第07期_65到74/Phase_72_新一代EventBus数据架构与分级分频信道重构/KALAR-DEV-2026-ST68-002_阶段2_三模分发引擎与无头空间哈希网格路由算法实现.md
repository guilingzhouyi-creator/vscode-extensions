  - `backend/domains/event_driven_audio/spatial_audio_solver.gd`（既有 `evaluate_spatial_audio`，本卷新增方法而非改写）
* **核心不变量约束断言**：
  - `Inv-EB2-1 (因果时序零漂移)`：`dispatch_now()` 派发的事件在当前 CPU 调用栈内同步内联完成，返回后全局状态 100% 最新。
  - `Inv-EB2-2 (空间精确裁剪零唤醒)`：监听者携带半径注册，派发时按 `欧氏距离 <= effect_radius + listener_radius` 精确过滤；范围外监听者回调与解包次数严格为 0。
  - `Inv-EB2-3 (网格哈希无退化)`：单元格尺寸配置驱动；极限密集（1,000+ 实体堆叠）下桶查找不劣于 $O(K)$（$K$ 邻域实体数）。
  - `Inv-EB2-4 (全周期对象池借还守恒)`：每个 `EventPool.borrow()` 借出包，在派发方完成分发后必须 `EventPool.recycle()`；借还差恒为 0、池外新建计数遥测可测。
  - `Inv-EB2-5 (空间位移动态哈希迁移)`：监听者跨格迁移原子化；旧格零唤醒、新格精确命中。
  - `Inv-EB2-6 (递归深度熔断断路保护)`：`dispatch_now()` 重入计数器达配置上限（默认 16）时经 `ErrorReporter` 断路拦截，杜绝级联栈溢出。
  - `Inv-EB2-7 (遍历分发动态修改免疫)`：派发采用数组快照隔离，回调内 `unbind/subscribe` 不引发跳项或越界。
  - `Inv-EB2-8 (表现层适配器单向驱动与零感知)`：EventBusCore 仅经 `IVisualPresentationAdapter` 抽象 API 驱动表现层，后端零感知下游为无模型/2D/3D。

## 一、 核心算法与引擎实现 (Core Algorithms & Engine Implementation)

### 1. 无头三维空间哈希网格 (`HeadlessSpatialHashGrid3D`)
* 模块路径: `res://backend/infrastructure/event_bus/headless_spatial_hash_grid_3d.gd`
* 职责: 纯数学空间分区 + **监听半径存储 + 欧氏距离精确裁剪 + token_id 去重**（保证 Inv-EB2-2 成立）。

```gdscript
class_name HeadlessSpatialHashGrid3D
extends RefCounted

var _cell_size: float = 16.0
var _grid: Dictionary = {}            # cell_key(int) -> Array[Dictionary]（含 callback + radius）
var _listeners: Dictionary = {}       # token_id(int) -> {center: Vector3, radius: float}

func _init(cell_size: float = 16.0) -> void:
	_cell_size = maxf(1.0, cell_size)

func _hash_cell(pos: Vector3) -> int:
	var cx := int(floor(pos.x / _cell_size))
	var cy := int(floor(pos.y / _cell_size))
	var cz := int(floor(pos.z / _cell_size))
	return (cx * 73856093) ^ (cy * 19349663) ^ (cz * 83492791)

func register_listener(token_id: int, center: Vector3, radius: float, callback: Callable) -> void:
	_listeners[token_id] = {"center": center, "radius": maxf(0.0, radius)}
	var key := _hash_cell(center)
	if not _grid.has(key):
		_grid[key] = []
	_grid[key].append({"token_id": token_id, "callback": callback})

func register_listener_2d(token_id: int, center_2d: Vector2, radius: float, callback: Callable, is_isometric_xz: bool = false) -> void:
	var pos_3d := Vector3(center_2d.x, 0.0, center_2d.y) if is_isometric_xz else Vector3(center_2d.x, center_2d.y, 0.0)
	register_listener(token_id, pos_3d, radius, callback)

func update_listener_position(token_id: int, new_center: Vector3, radius: float, callback: Callable) -> void:
	if not _listeners.has(token_id):
		register_listener(token_id, new_center, radius, callback)
		return
	var old_center: Vector3 = _listeners[token_id]["center"]
	var old_key := _hash_cell(old_center)
	var new_key := _hash_cell(new_center)
	_listeners[token_id] = {"center": new_center, "radius": maxf(0.0, radius)}
	if old_key == new_key:
		return
	if _grid.has(old_key):
		var arr: Array = _grid[old_key]
		for i in range(arr.size() - 1, -1, -1):
			if arr[i]["token_id"] == token_id:
				arr.remove_at(i)
		if arr.is_empty():
			_grid.erase(old_key)
	if not _grid.has(new_key):
		_grid[new_key] = []
	_grid[new_key].append({"token_id": token_id, "callback": callback})

func unregister_listener(token_id: int) -> void:
	if not _listeners.has(token_id):
		return
	var center: Vector3 = _listeners[token_id]["center"]
	_listeners.erase(token_id)
	var key := _hash_cell(center)
	if _grid.has(key):
		var arr: Array = _grid[key]
		for i in range(arr.size() - 1, -1, -1):
			if arr[i]["token_id"] == token_id:
				arr.remove_at(i)
		if arr.is_empty():
			_grid.erase(key)

## 派发：单元格粗裁剪 → 欧氏距离精确过滤（Inv-EB2-2）→ token_id 去重
func dispatch_spatial(pos: Vector3, radius: float, packet: EventPacket) -> int:
	var r := maxf(0.0, radius)
	var min_cx := int(floor((pos.x - r) / _cell_size))
	var max_cx := int(floor((pos.x + r) / _cell_size))
	var min_cy := int(floor((pos.y - r) / _cell_size))
	var max_cy := int(floor((pos.y + r) / _cell_size))
	var min_cz := int(floor((pos.z - r) / _cell_size))
	var max_cz := int(floor((pos.z + r) / _cell_size))

	var dispatched := 0
	var called_tokens := {} # token_id 去重（比 get_object_id 稳定）

	for cx in range(min_cx, max_cx + 1):
		for cy in range(min_cy, max_cy + 1):
			for cz in range(min_cz, max_cz + 1):
				var h := (cx * 73856093) ^ (cy * 19349663) ^ (cz * 83492791)
				if not _grid.has(h):
					continue
				for entry in _grid[h]:
					var tid: int = entry["token_id"]
					if called_tokens.has(tid):
						continue
					called_tokens[tid] = true
					var center: Vector3 = _listeners[tid]["center"]
					var lr: float = _listeners[tid]["radius"]
					if pos.distance_to(center) <= r + lr: # 精确欧氏过滤
						(entry["callback"] as Callable).call(packet)
						dispatched += 1
	return dispatched

## 便捷入口：按事件包自身空间头派发
func dispatch_packet(packet: EventPacket) -> int:
	if packet == null or not packet.is_spatial:
		return 0
	return dispatch_spatial(packet.world_position, packet.effect_radius, packet)
```

### 2. 新一代分发中枢引擎 (`EventBusCore`，统一门面)
* 模块路径: `res://backend/infrastructure/event_bus/event_bus_core.gd`
* 职责: 整型信道/位掩码路由 + 帧合批 + 空间分发 + 对象池协同；普通类可实例化（测试隔离），`get_instance()` 为应用级单例（GameBootstrap 装配）。

```gdscript
class_name EventBusCore
extends RefCounted

static var _instance: EventBusCore = null

var _channel_routes: Dictionary = {}   # channel_id(int) -> Array[Callable]
var _category_routes: Dictionary = {}  # mask(int) -> Array[Callable]
var _spatial_grid: HeadlessSpatialHashGrid3D = null
var _ring_buffer: PingPongRingBuffer = null
var _pool: EventPool = null
var _seq_counter: int = 0
var _dispatch_depth: int = 0
var _max_depth: int = 16

func _init() -> void:
	_spatial_grid = HeadlessSpatialHashGrid3D.new(16.0)
	_ring_buffer = PingPongRingBuffer.new(4096)
	_pool = EventPool.new(512)
	_max_depth = 16

## 应用级单例（GameBootstrap 装配；测试请用 EventBusCore.new() 独立实例）
static func get_instance() -> EventBusCore:
	if _instance == null:
		_instance = EventBusCore.new()
	return _instance

## 从 event_bus_config.json 加载参数（单元尺寸/缓冲容量/池容量/熔断深度）
func initialize_from_config() -> void:
	var cfg := GameConfig.get_dict("infrastructure.event_bus_config", "spatial_routing", {})
	var cell := GameConfig.get_float("infrastructure.event_bus_config", "spatial_routing/cell_size_meters", 16.0)
	_spatial_grid = HeadlessSpatialHashGrid3D.new(maxf(1.0, cell))
	var ring_cap := GameConfig.get_int("infrastructure.event_bus_config", "frame_batching/ring_buffer_capacity", 4096)
	_ring_buffer = PingPongRingBuffer.new(maxi(1, ring_cap))
	var pool_cap := GameConfig.get_int("infrastructure.event_bus_config", "pool/capacity", 512)
	_pool = EventPool.new(maxi(1, pool_cap))
	_max_depth = maxi(1, GameConfig.get_int("infrastructure.event_bus_config", "circuit_breaker/max_dispatch_depth", 16))

## 测试隔离：清空路由并复位单例
func clear_for_tests() -> void:
	_channel_routes.clear()
	_category_routes.clear()
	_spatial_grid = HeadlessSpatialHashGrid3D.new(16.0)
	_ring_buffer = PingPongRingBuffer.new(4096)
	_pool = EventPool.new(512)
	_seq_counter = 0
	_dispatch_depth = 0

## 对象池借还（统一 API）
func borrow_packet(channel_id: int, category_mask: int, payload: Variant = null) -> EventPacket:
	var packet := _pool.borrow()
	packet.channel_id = channel_id
	packet.category_mask = category_mask
	packet.payload_data = payload
	return packet

func recycle_packet(packet: EventPacket) -> void:
	_pool.recycle(packet)

# ---- 订阅门面（统一返回 EventBusSubscriptionToken） ----
func subscribe(channel_id: int, category_mask: int, callback: Callable) -> EventBusSubscriptionToken:
	var token := EventBusSubscriptionToken.new(self, channel_id, category_mask, callback, false)
	if not _channel_routes.has(channel_id):
		_channel_routes[channel_id] = []
	_channel_routes[channel_id].append(callback)
	if not _category_routes.has(category_mask):
		_category_routes[category_mask] = []
	_category_routes[category_mask].append(callback)
	return token

func on_spatial(pos: Vector3, radius: float, callback: Callable) -> EventBusSubscriptionToken:
	var token_id := _seq_counter + 1
	_seq_counter += 1
	_spatial_grid.register_listener(token_id, pos, radius, callback)
	return EventBusSubscriptionToken.new(self, 0, 0, callback, true, token_id)

func on_spatial_2d(pos_2d: Vector2, radius: float, callback: Callable, is_isometric_xz: bool = false) -> EventBusSubscriptionToken:
	var token_id := _seq_counter + 1
	_seq_counter += 1
	_spatial_grid.register_listener_2d(token_id, pos_2d, radius, callback, is_isometric_xz)
	return EventBusSubscriptionToken.new(self, 0, 0, callback, true, token_id)

func _unregister_token(token: EventBusSubscriptionToken) -> void:
	if token.is_spatial:
		_spatial_grid.unregister_listener(token.token_id)
	else:
		if _channel_routes.has(token.channel_id):
			_channel_routes[token.channel_id].erase(token.callback)
		if _category_routes.has(token.category_mask):
			_category_routes[token.category_mask].erase(token.callback)

# ---- 原语 ①：dispatch_now（即时同步因果 + 熔断经 ErrorReporter） ----
func dispatch_now(packet: EventPacket) -> void:
	if packet == null:
		return
	if _dispatch_depth >= _max_depth:
		ErrorReporter.emit("event_bus2_reentrancy_guard", {"channel_id": packet.channel_id, "max_depth": _max_depth})
		return
	_dispatch_depth += 1
	_seq_counter += 1
	packet.event_id = _seq_counter

	if _channel_routes.has(packet.channel_id):
		var subs: Array = (_channel_routes[packet.channel_id] as Array).duplicate()
		for cb in subs:
			if (cb as Callable).is_valid():
				cb.call(packet)
	for mask_key in _category_routes.keys():
		if (int(mask_key) & packet.category_mask) != 0:
			var subs: Array = (_category_routes[mask_key] as Array).duplicate()
			for cb in subs:
				if (cb as Callable).is_valid():
					cb.call(packet)
	_dispatch_depth -= 1

# ---- 原语 ②：enqueue_frame + flush_frame_events（帧级合批） ----
func enqueue_frame(packet: EventPacket) -> bool:
	if packet == null:
		return false
	if (packet.category_mask & EventCategoryMask.CORE_STATE) != 0:
		ErrorReporter.emit("event_bus2_core_state_in_frame_queue", {"channel_id": packet.channel_id})
		return false
	_seq_counter += 1
	packet.event_id = _seq_counter
	var ok := _ring_buffer.enqueue(packet)
	if not ok:
		ErrorReporter.emit("event_bus2_frame_queue_overflow", {"channel_id": packet.channel_id})
	return ok

func flush_frame_events() -> int:
	return _ring_buffer.flush_frame(func(packet: EventPacket) -> void:
		if _channel_routes.has(packet.channel_id):
			var subs: Array = (_channel_routes[packet.channel_id] as Array).duplicate()
			for cb in subs:
				if (cb as Callable).is_valid():
					cb.call(packet)
		recycle_packet(packet)
	)

# ---- 原语 ③：dispatch_spatial（2D/3D AoI 定向裁剪） ----
func dispatch_spatial(packet: EventPacket, origin: Vector3, radius: float) -> int:
	if packet == null:
		return 0
	_seq_counter += 1
	packet.event_id = _seq_counter
	packet.set_spatial_header(origin, radius)
	return _spatial_grid.dispatch_packet(packet)

func dispatch_spatial_2d(packet: EventPacket, origin_2d: Vector2, radius: float, is_isometric_xz: bool = false) -> int:
	if packet == null:
		return 0
	_seq_counter += 1
	packet.event_id = _seq_counter
	packet.set_spatial_header_2d(origin_2d, radius, is_isometric_xz)
	return _spatial_grid.dispatch_packet(packet)

## 配置 getter（供测试与遥测断言）
func get_spatial_cell_size() -> float: return _spatial_grid._cell_size
func get_frame_buffer_capacity() -> int: return _ring_buffer.capacity()
func get_max_dispatch_depth() -> int: return _max_depth
func get_pool() -> EventPool: return _pool
```

```gdscript
class_name EventBusSubscriptionToken
extends RefCounted

var bus: EventBusCore = null
var channel_id: int = 0
var category_mask: int = 0
var callback: Callable = Callable()
var is_spatial: bool = false
var token_id: int = 0
var is_active: bool = false

func _init(p_bus: EventBusCore, p_channel: int, p_mask: int, p_cb: Callable, p_spatial: bool, p_tid: int = 0) -> void:

	channel_id = p_channel

## 幂等注销（重复调用返回 false，无副作用）
func unbind() -> bool:

		return false
	bus._unregister_token(self)
	is_active = false
	return true
```
