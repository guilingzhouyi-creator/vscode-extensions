## 📌 第一性原理溯源指针

* **精准上游规范指针**：
  - `config/domains/combat.json:L120-L138`：既有 `draw_per_round: 3` 写死数量与手牌上限；
  - `config/domains/combat.json:L169-L192`：既有 `tertiary_timeline` 与 `round_lifecycle` 缺少加发牌点与预排期分布配置；
  - `backend/domains/physics_thermodynamics/tertiary_timeline_engine.gd:L49-L73`：`advance_time` 实时临时重排期事件点，缺乏整回合计划；
  - `backend/domains/physics_thermodynamics/combat_round_coordinator.gd:L96-L108`：手牌为空且 AP 耗尽时直接提前掐断回合，阻塞时间轴。

* **核心不变量约束断言**：
  - `Inv-TR-1`（双随机完全独立不变量）：发牌数量生成（Quantity Random）与具体卡牌实例/牌型抽取（Card-Type Random）属于两个相互独立的随机过程，数量随机分布不依赖牌型池，牌型抽取概率严格遵循 PRD 权重；
  - `Inv-TR-2`（零手牌合法性与流水线守恒）：普通回合默认基础发牌数量下限为 1（$MinimumHandCount \ge 1$），但最终发牌数量严格由 $BaseCount \to SpecialModifier \to FinalCount$ 决定；特殊影响导致 $FinalHandCount = 0$ 属于合法业务状态，严禁使用 `max(1, count)` 强行覆盖；
  - `Inv-TR-3`（时间轴零手牌绝对不阻塞）：即使当前回合 $FinalHandCount == 0$ 且玩家无可用 AP，第三时间轴必须正常推进至本回合时间上限或紧张点强制切轮，严禁因手牌为空暂停、跳过或提前销毁时间轴；
  - `Inv-TR-4`（回合开局预排期不变量）：本回合所有第三时间轴事件（普通发牌点、加发牌点、紧张点）必须在回合启动时一次性离散分布生成，禁止在时间到达时临时判定是否存在事件；
  - `Inv-TR-5`（跨回合修饰显式消费契约）：上一回合对下一回合时间轴的干预必须通过 `PendingTimelineModifierDTO` 显式承载，包含来源、生命周期与单次消费标记，严禁通过隐式全局内存变量泄漏；
  - `Inv-TR-6`（前端信息安全边界不变量）：前端/客户端快照仅包含当前推进进度比例（`progress_ratio`）、已触发事件历史（`resolved_events`）与当前已入手牌，严禁向客户端暴露未触发事件的精确时间戳、剩余秒数倒计时及未来卡牌内容。

* **防漂移最高指示**：
  - 严格保持与既有 `CombatRoundCoordinator`、`ActionCardStagingBar` 和 `DeterministicRNG` 的接口兼容，采用增量 DTO 扩展，杜绝破坏既有 82 套测试套件；
  - 严格遵循零硬编码规范，所有数量分布、时间区间、间距约束与事件权重收敛于 `config/domains/combat.json`。

## 一、 核心数据模型与代码契约 (Data Structures & Contracts)

### 1.1 手牌抽取规格与结果数据模型 (`HandDrawSpecDTO` / `HandDrawResultDTO`)

```gdscript
# 模块路径: res://backend/domains/physics_thermodynamics/combat_hand_draw_dto.gd
class_name CombatHandDrawDTO
extends RefCounted

## 手牌抽取输入规格
class HandDrawSpecDTO extends RefCounted:
	var is_first_encounter: bool = false       # 是否为初次遇敌首回合
	var current_round: int = 1                 # 当前小回合编号
	var current_hand_count: int = 0            # 待发栏已有手牌数
	var max_capacity: int = 5                  # 手牌待发栏上限
	var special_modifiers: Array[Dictionary] = [] # 施加于抽牌的特殊影响清单

	func to_dto() -> Dictionary:
		return {
			"is_first_encounter": is_first_encounter,
			"current_round": current_round,
			"current_hand_count": current_hand_count,
			"max_capacity": max_capacity,
			"special_modifiers": special_modifiers.duplicate(true),
		}

## 手牌抽取最终裁决结果
class HandDrawResultDTO extends RefCounted:
	var base_random_count: int = 0             # 基础随机决定的发牌数量
	var modifier_delta: int = 0                # 特殊修正总量 (可为负数)
	var final_draw_count: int = 0              # 最终实际抽牌数量 (合法支持 0)
	var actual_cards: Array = []               # 最终生成的卡牌实体列表 (CombatActionCardEntity)
	var applied_modifiers: Array[String] = []  # 实际生效的特殊影响 ID 列表
	var is_zero_hand_state: bool = false       # 是否进入合法 0 手牌状态

	func to_dto() -> Dictionary:
		var cards_dto: Array = []
		for c in actual_cards:
			if c != null and c.has_method("to_dto"):
				cards_dto.append(c.to_dto())
			elif c != null:
				cards_dto.append({"card_id": c.card_id, "verb_type": c.verb_type})
		return {
			"base_random_count": base_random_count,
			"modifier_delta": modifier_delta,
			"final_draw_count": final_draw_count,
			"cards": cards_dto,
			"applied_modifiers": applied_modifiers.duplicate(),
			"is_zero_hand_state": is_zero_hand_state,
		}
```

### 1.2 第三时间轴离散事件排期模型 (`TimelineEventScheduleDTO`)

```gdscript
# 模块路径: res://backend/domains/physics_thermodynamics/timeline_event_schedule_dto.gd
class_name TimelineEventScheduleDTO
extends RefCounted

enum EventKind {
	NORMAL_DRAW,     # 普通随机发牌点 (依数量规则动态抽卡推入手牌)
	ADD_DRAW,        # 加发牌点 (局势奖励额外补发 1~2 张卡牌)
	TENSION_PULSE,   # 回内紧张点 (触发压力激增、环境脉冲或强行切轮)
}

class ScheduledPoint extends RefCounted:
	var point_id: String = ""
	var event_kind: EventKind = EventKind.NORMAL_DRAW
	var trigger_time_ms: int = 0               # 仅后端内部持有的绝对毫秒时间戳
	var payload: Dictionary = {}               # 事件专属参数 (如紧张类型、加牌参数)
	var is_executed: bool = false              # 是否已触发完成

	func to_internal_dict() -> Dictionary:
		return {
			"point_id": point_id,
			"event_kind": int(event_kind),
			"trigger_time_ms": trigger_time_ms,
			"payload": payload.duplicate(true),
			"is_executed": is_executed,
		}

var round_number: int = 1
var max_round_duration_ms: int = 10000
var scheduled_points: Array[ScheduledPoint] = []
var active_modifiers: Array[Dictionary] = []

func to_internal_dto() -> Dictionary:
	var pts: Array[Dictionary] = []
	for p in scheduled_points:
		pts.append(p.to_internal_dict())
	return {
		"round_number": round_number,
		"max_round_duration_ms": max_round_duration_ms,
		"scheduled_points": pts,
		"active_modifiers": active_modifiers.duplicate(true),
	}
```

```gdscript
# 模块路径: res://backend/domains/physics_thermodynamics/combat_timeline_client_dto.gd
class_name CombatTimelineClientDTO
extends RefCounted

## 专门面向表现层/前端暴露的只读安全视图，100% 消除未决事件精确倒计时与未来卡牌窥探
class PresenterSnapshotDTO extends RefCounted:
	var round_number: int = 1
	var progress_ratio: float = 0.0            # 时间轴推进比例 [0.0, 1.0]，供进度条渲染

	var is_zero_hand_active: bool = false      # 当前是否因特殊状态处于无牌可打姿态

	func to_client_view() -> Dictionary:
		return {
			"round_number": round_number,
			"progress_ratio": progress_ratio,
			"is_round_concluded": is_round_concluded,
			"current_hand_count": current_hand_count,
			"resolved_events": resolved_events.duplicate(true),
			"is_zero_hand_active": is_zero_hand_active,
		}
```
