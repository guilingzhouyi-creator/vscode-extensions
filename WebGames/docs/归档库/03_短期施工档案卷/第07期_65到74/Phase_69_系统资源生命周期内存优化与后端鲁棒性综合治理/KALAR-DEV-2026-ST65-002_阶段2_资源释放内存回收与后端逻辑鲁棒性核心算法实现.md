* **核心不变量约束断言**：
  - `Inv-RM-6 (安全释放零崩溃不变量)`：资源管理器销毁任何底层实例（如调用 `node.free()`）前，必须先验证其实例有效性（`is_instance_valid()`）且未入树或已从父级树移除，防止双重销毁（Double Free）与 Godot 引擎层崩溃。
  - `Inv-RM-7 (作用域订阅闭环不变量)`：通过 `ScopedEventSubscriber` 托管的所有信号连接，在宿主进入 `RELEASE` 或调用 `clear()` 时，其内部存活令牌数必须归零，任何后续广播不得触发失效闭包。
  - `Inv-RM-8 (有界缓存上限不变量)`：`BoundedResourceCache` 中存活的有效条目数在任何时刻严格小于等于 `max_capacity`；超出容量时必须依据指定淘汰策略（LRU/FIFO）立即驱逐超额旧数据。
  - `Inv-RM-9 (输入污染零渗透不变量)`：后端鲁棒性守卫对一切进入核心求解域的数据包执行前置清洗，遇到 null、越界、类型不匹配或坏值时，执行安全截断或回退默认值，严禁脏数据静默污染状态机。
  - `Inv-RM-10 (核心操作严格幂等不变量)`：受控于 `IdempotencyController` 的写操作（如连续触发多次保存、重复发放同一全局序列号奖励、重复初始化），仅首次执行产生状态副作用，后续重复调用安全透传返回首次结果，副作用计数严格恒等于 1。
* **防漂移最高指示**：严禁采用 `try...catch...ignore` 静默吞掉底层异常；严格区分可恢复异常、业务拒绝、配置错误与系统级崩溃，建立结构化分类隔离。

## 一、 算法实现设计与核心代码契约 (Algorithm Architecture & Code Contracts)

### 1. 统一资源生命周期管理器 (`resource_lifecycle_manager.gd`)

```gdscript
# 模块路径: res://backend/infrastructure/resource_governance/resource_lifecycle_manager.gd
class_name ResourceLifecycleManager extends RefCounted

## 已注册资源映射表: resource_id -> ResourceLifecycleDescriptor
var _tracked_resources: Dictionary = {}
## 待清理队列（延迟销毁安全缓冲）
var _pending_dispose_queue: Array[ResourceLifecycleDescriptor] = []

## 注册非 RefCounted 对象进入生命周期治理
func track_resource(
	p_res_id: String,
	p_target: Object,
	p_category: ResourceLifecycleDescriptor.ResourceCategory,
	p_owner_name: String
) -> ResourceLifecycleDescriptor:
	if p_res_id.is_empty() or p_target == null:
		return null
	if _tracked_resources.has(p_res_id):
		return _tracked_resources[p_res_id] as ResourceLifecycleDescriptor

	var desc := ResourceLifecycleDescriptor.new(p_res_id, p_target, p_category, p_owner_name)
	desc.transition_to(ResourceLifecycleDescriptor.LifecycleState.INITIALIZED)
	_tracked_resources[p_res_id] = desc
	return desc

## 标记资源使用开始
func mark_in_use(p_res_id: String) -> bool:
	if not _tracked_resources.has(p_res_id):
		return false
	var desc: ResourceLifecycleDescriptor = _tracked_resources[p_res_id]
	return desc.transition_to(ResourceLifecycleDescriptor.LifecycleState.IN_USE)

## 标记资源进入闲置
func mark_idle(p_res_id: String) -> bool:
	if not _tracked_resources.has(p_res_id):
		return false
	var desc: ResourceLifecycleDescriptor = _tracked_resources[p_res_id]
	return desc.transition_to(ResourceLifecycleDescriptor.LifecycleState.IDLE)

## 安全释放并物理销毁资源
func dispose_resource(p_res_id: String) -> bool:
	if not _tracked_resources.has(p_res_id):
		return false
	var desc: ResourceLifecycleDescriptor = _tracked_resources[p_res_id]
	_tracked_resources.erase(p_res_id)

	# 先转移至 RELEASED
	desc.transition_to(ResourceLifecycleDescriptor.LifecycleState.RELEASED)

	# 执行物理资源回收
	var target: Object = desc.get_target_instance()
	if target != null and is_instance_valid(target):
		if target is Node:
			var node: Node = target as Node
			if node.get_parent() != null:
				node.get_parent().remove_child(node)
			node.queue_free()
		elif not (target is RefCounted):
			target.free()

	desc.mark_disposed()
	return true

## 全量清理所有受控资源（退出、场景切换或测试重置时调用）
func dispose_all() -> int:
	var freed_count: int = 0
	var keys := _tracked_resources.keys()
	for key in keys:
		if dispose_resource(str(key)):
			freed_count += 1
	_tracked_resources.clear()
	_pending_dispose_queue.clear()
	return freed_count

## 获取当前存活且受控的资源统计字典
func get_live_metrics() -> Dictionary:
	var active_nodes: int = 0
	var active_others: int = 0
	for res_id in _tracked_resources:
		var desc: ResourceLifecycleDescriptor = _tracked_resources[res_id]
		if desc.target_category == ResourceLifecycleDescriptor.ResourceCategory.UI_CONTROL_NODE:
			active_nodes += 1
		else:
			active_others += 1
	return {
		"total_tracked": _tracked_resources.size(),
		"active_ui_nodes": active_nodes,
		"active_other_resources": active_others
	}
```

### 2. 作用域事件订阅代理 (`scoped_event_subscriber.gd`)

```gdscript
# 模块路径: res://backend/infrastructure/resource_governance/scoped_event_subscriber.gd
class_name ScopedEventSubscriber extends RefCounted

var _tokens: Array[EventSubscriptionToken] = []
var _is_disposed: bool = false

## 订阅指定信号，并纳管生成的注销令牌
func subscribe(p_sig: Signal, p_callable: Callable) -> bool:
	if _is_disposed or not p_sig.is_valid() or not p_callable.is_valid():
		return false
	if p_sig.is_connected(p_callable):
		return false # 防重复绑定

	var err: Error = p_sig.connect(p_callable)
	if err != OK:
		return false

	var token := EventSubscriptionToken.new(p_sig, p_callable)
	_tokens.append(token)
	return true

## 解绑指定信号
func unsubscribe_from(p_sig: Signal, p_callable: Callable) -> bool:
	for i in range(_tokens.size() - 1, -1, -1):
		var tok: EventSubscriptionToken = _tokens[i]
		if tok.source_signal == p_sig and tok.callable_target == p_callable:
			tok.unsubscribe()
			_tokens.remove_at(i)
			return true
	return false

## 释放全部托管订阅（宿主销毁或重置时调用）
func clear() -> int:
	var cleared_count: int = 0
	for tok in _tokens:
		if tok.unsubscribe():
			cleared_count += 1
	_tokens.clear()
	return cleared_count

## 获取当前存活订阅数
func get_active_subscription_count() -> int:
	var count: int = 0
	for tok in _tokens:
		if tok.is_active:
			count += 1
	return count

## 彻底销毁代理
func dispose() -> void:
	clear()
	_is_disposed = true
```

### 3. 有界资源淘汰缓存容器 (`bounded_resource_cache.gd`)

```gdscript
# 模块路径: res://backend/infrastructure/resource_governance/bounded_resource_cache.gd
class_name BoundedResourceCache extends RefCounted

var _spec: CachePolicySpec
var _store: Dictionary = {}
var _access_order: Array[String] = [] # 用于 LRU 与 FIFO 淘汰追踪
var _timestamps: Dictionary = {}     # key -> float (秒)

func _init(p_spec: CachePolicySpec = null) -> void:
	if p_spec != null:
		_spec = p_spec
	else:
		_spec = CachePolicySpec.new("default_cache", 100)

## 放入条目，超出容量时自动触发淘汰
func put(p_key: String, p_value: Variant) -> void:
	if p_key.is_empty():
		return

	# 若已存在，更新值与访问顺序
	if _store.has(p_key):
		_store[p_key] = p_value
		_timestamps[p_key] = Time.get_unix_time_from_system()
		_touch(p_key)
		return

	# 容量已满，执行驱逐
	while _store.size() >= _spec.max_capacity:
		_evict_one()

	_store[p_key] = p_value
	_timestamps[p_key] = Time.get_unix_time_from_system()
	_access_order.append(p_key)

## 获取条目，支持 TTL 过期淘汰
func get_val(p_key: String, p_default: Variant = null) -> Variant:
	if not _store.has(p_key):
		return p_default

	# TTL 过期检查
	if _spec.eviction_policy == CachePolicySpec.EvictionPolicy.TTL:
		var created_time: float = float(_timestamps.get(p_key, 0.0))
		var current_time: float = Time.get_unix_time_from_system()
		if current_time - created_time > _spec.ttl_seconds:
			_remove_internal(p_key)
			return p_default

	if _spec.eviction_policy == CachePolicySpec.EvictionPolicy.LRU:
		_touch(p_key)

	return _store[p_key]

## 判定条目是否存在且未过期
func has(p_key: String) -> bool:
	return get_val(p_key, null) != null

## 驱逐单个条目
func _evict_one() -> void:
	if _access_order.is_empty():
		return
	var victim_key: String = _access_order[0]
	_remove_internal(victim_key)

func _remove_internal(p_key: String) -> void:
	_store.erase(p_key)
	_timestamps.erase(p_key)
	_access_order.erase(p_key)

func _touch(p_key: String) -> void:
	_access_order.erase(p_key)
	_access_order.append(p_key)

## 热更/场景重置清理通道
func clear_on_hot_reload() -> void:
	if _spec.allow_hot_reload_clear:
		clear()

func clear() -> void:
	_store.clear()
	_timestamps.clear()
	_access_order.clear()

func size() -> int:
	return _store.size()
```

### 4. 后端鲁棒性守卫与幂等性控制器 (`backend_robustness_guard.gd` / `idempotency_controller.gd`)

```gdscript
# 模块路径: res://backend/infrastructure/robustness/backend_robustness_guard.gd
class_name BackendRobustnessGuard extends RefCounted

## 安全整数清洗（防御越界与 NaN/null）
static func sanitize_int(p_val: Variant, p_min: int, p_max: int, p_default: int) -> int:
	if p_val == null or not (p_val is int or p_val is float):
		return p_default
	var num: int = int(p_val)
	return clampi(num, p_min, p_max)

## 安全字符串清洗（防御空与非法转义）
static func sanitize_string(p_val: Variant, p_default: String = "") -> String:
	if p_val == null:
		return p_default
	return str(p_val).strip_edges()

## 安全字典清洗（确保非空与类型安全）
static func sanitize_dict(p_val: Variant) -> Dictionary:
	if p_val == null or not (p_val is Dictionary):
		return {}
	return (p_val as Dictionary).duplicate(true)

## 安全数组清洗
static func sanitize_array(p_val: Variant) -> Array:
	if p_val == null or not (p_val is Array):
		return []
	return (p_val as Array).duplicate(true)
```

```gdscript
# 模块路径: res://backend/infrastructure/robustness/idempotency_controller.gd
class_name IdempotencyController extends RefCounted

## 执行操作记录字典: operation_token -> { "executed_at": float, "result": Variant }
var _executed_tokens: Dictionary = {}
var _max_token_history: int = 200

## 检查并执行幂等性保护
## 返回: { "already_executed": bool, "result": Variant }
func check_and_record(p_token: String, p_result_if_new: Variant = true) -> Dictionary:
	if p_token.is_empty():
		return { "already_executed": false, "result": null }

	if _executed_tokens.has(p_token):
		var entry: Dictionary = _executed_tokens[p_token]
		return {
			"already_executed": true,
			"result": entry.get("result", null)
		}

	# 淘汰超额历史令牌
	if _executed_tokens.size() >= _max_token_history:
		var first_key: String = str(_executed_tokens.keys()[0])
		_executed_tokens.erase(first_key)

	_executed_tokens[p_token] = {
		"executed_at": Time.get_unix_time_from_system(),
		"result": p_result_if_new
	}
	return {
		"already_executed": false,
		"result": p_result_if_new
	}

func reset() -> void:
	_executed_tokens.clear()
```

## 三、 核心算法验收矩阵 (DoD Matrix)

| 检验项 ID | 校验目标 | 验证输入 | 预期输出断言 |
| :--- | :--- | :--- | :--- |
| `TC-RM-S2-01` | ResourceLifecycleManager 节点纳管与销毁 | 追踪动态 Control 节点并触发 dispose | 节点退出树并被释放，`_tracked_resources` 计数归零，不抛出 Double Free 异常 |
| `TC-RM-S2-02` | ScopedEventSubscriber 自动批量解绑 | 绑定 3 个信号后触发 clear | 全部信号 `is_connected` 变更为 false，有效订阅计数归零 |
| `TC-RM-S2-03` | BoundedResourceCache LRU 容量硬约束 | 设容量为 3，依次 put A, B, C, D | 容器元素严格为 3，最久未访问的 A 被精准淘汰 |
| `TC-RM-S2-04` | BackendRobustnessGuard 脏数据清洗 | 传入 null, 越界值, 字符串型负数 | 正确截断在 `[p_min, p_max]` 范围，不产生 NaN 或类型崩溃 |
| `TC-RM-S2-05` | IdempotencyController 连续重复调用防重 | 相同 Token 连续调用 3 次 | 首次 `already_executed = false`，后两次为 true 且结果严格透传 |
