## 📌 第一性原理溯源指针

* **精准上游规范指针**：
  - `backend/infrastructure/log_ring_buffer.gd`：环形内存日志缓冲区（高效内存闭环，避免日志造成内存泄漏）；
  - `config/infrastructure/log.json` / `errors_catalog.json`：配置驱动单一真源，统一英文模板库；
  - `backend/domains/persistence_protocol/runtime_mode_gate.gd`：P68 确立的单机/联机四层隔离门禁；
  - `backend/infrastructure/telemetry_aggregate.gd`：轻量级遥测聚合器。
* **核心不变量约束断言**：
  - `Inv-RM-11 (日志文本解耦不变量)`：业务代码中严禁直接包含自然语言英文句子硬编码（如 `Log("Player inventory update failed")`）；所有日志必须由 `event_code` 经由配置驱动模板目录解析。
  - `Inv-RM-12 (结构化六元组不变量)`：流经日志中枢的每条记录必须完整包含 `event_code`, `domain`, `severity`, `timestamp`, `correlation_id`, `message` 六元组；严禁无来源、无等级的匿名打印。
  - `Inv-RM-13 (日志内存有界不变量)`：日志在内存中暂存时必须严格受限（如最大 500 条 LogRingBuffer），超出条目自动覆盖最旧条目，严禁日志容器随运行时间无限膨胀。
  - `Inv-RM-14 (单机联机日志隔离不变量)`：在 `SP_STANDALONE` 模式下的本地详细 TRACE/DEBUG 日志与本地异常堆栈，严禁打包至联机 RPC 同步信道中，杜绝网络带宽浪费与客户端隐私泄露。
  - `Inv-RM-15 (性能分析零开销扩展不变量)`：遥测与性能分析钩子在未启用或生产 Release 模式下，其开销严格控制在纳秒级条件分支跳过（No-Op），且底层采集策略（内存/执行时延/CPU）与业务解耦。
* **防漂移最高指示**：严禁机械增加难以维护的深层抽象；日志与遥测必须轻量高效，兼顾面向程序的机器可读性与面向开发者的人类可读英文语义。

## 一、 日志系统与性能钩子代码契约 (System Contracts & Architecture)

### 1. 统一结构化日志条目与服务 (`unified_logger_service.gd`)

```gdscript
# 模块路径: res://backend/infrastructure/logging/unified_logger_service.gd
class_name UnifiedLoggerService extends RefCounted

## 日志严重等级标准枚举
enum Severity {
	TRACE = 0,
	DEBUG = 1,
	INFO = 2,
	WARNING = 3,
	ERROR = 4,
	CRITICAL = 5
}

## 结构化日志记录实体
class LogRecord extends RefCounted:
	var event_code: String = ""
	var domain: String = ""
	var severity: Severity = Severity.INFO
	var timestamp: float = 0.0
	var correlation_id: String = ""
	var context: Dictionary = {}
	var message: String = ""

	func to_dict() -> Dictionary:
		return {
			"event_code": event_code,
			"domain": domain,
			"severity": severity,
			"timestamp": timestamp,
			"correlation_id": correlation_id,
			"context": context,
			"message": message
		}

## 英文消息模板目录: event_code -> format_template
var _message_catalog: Dictionary = {}
## 过滤等级阈值
var _min_severity: Severity = Severity.INFO
## 内存环形缓冲区引用
var _ring_buffer: RefCounted = null
## 性能遥测汇集钩子
var _metric_sink: Object = null

func _init(p_catalog: Dictionary = {}, p_min_level: Severity = Severity.INFO) -> void:
	_message_catalog = p_catalog.duplicate(true)
	_min_severity = p_min_level

## 加载配置表模板库
func load_catalog_from_dict(p_config: Dictionary) -> void:
	if p_config.has("events") and p_config["events"] is Dictionary:
		for code in p_config["events"]:
			_message_catalog[code] = str(p_config["events"][code])

## 记录日志统一核心 API
func log_event(
	p_code: String,
	p_domain: String,
	p_severity: Severity,
	p_params: Dictionary = {},
	p_correlation_id: String = ""
) -> LogRecord:
	if p_severity < _min_severity:
		return null

	var template: String = str(_message_catalog.get(p_code, "[{code}] Event occurred in domain {domain}"))
	var formatted_msg: String = _format_message(template, p_params, p_code, p_domain)

	var rec := LogRecord.new()
	rec.event_code = p_code
	rec.domain = p_domain
	rec.severity = p_severity
	rec.timestamp = Time.get_unix_time_from_system()
	rec.correlation_id = p_correlation_id
	rec.context = p_params.duplicate(true)
	rec.message = formatted_msg

	_dispatch_record(rec)
	return rec

## 模板占位符格式化
func _format_message(p_tpl: String, p_params: Dictionary, p_code: String, p_domain: String) -> String:
	var msg: String = p_tpl
	msg = msg.replace("{code}", p_code)
	msg = msg.replace("{domain}", p_domain)
	for k in p_params:
		msg = msg.replace("{" + str(k) + "}", str(p_params[k]))
	return msg

## 多路分发输出 Sink
func _dispatch_record(p_rec: LogRecord) -> void:
	# 1. 内存有界环形缓冲
	if _ring_buffer != null and _ring_buffer.has_method("push_record"):
		_ring_buffer.call("push_record", p_rec.to_dict())

	# 2. 控制台输出格式化
	var sev_name: String = _get_severity_name(p_rec.severity)
	var formatted_line: String = "[%s][%s][%s] %s" % [
		sev_name,
		p_rec.domain,
		p_rec.event_code,
		p_rec.message
	]
	if p_rec.severity >= Severity.ERROR:
		printerr(formatted_line)
	else:
		print(formatted_line)

func _get_severity_name(p_sev: Severity) -> String:
	match p_sev:
		Severity.TRACE: return "TRACE"
		Severity.DEBUG: return "DEBUG"
		Severity.INFO: return "INFO"
		Severity.WARNING: return "WARN"
		Severity.ERROR: return "ERROR"
		Severity.CRITICAL: return "CRIT"
		_: return "UNKNOWN"
```

### 2. 性能与资源遥测钩子规范 (`profiling_metric_hook.gd`)

```gdscript
# 模块路径: res://backend/infrastructure/profiling/profiling_metric_hook.gd
class_name ProfilingMetricHook extends RefCounted

## 性能指标采样包
class PerformanceSample extends RefCounted:
	var metric_name: String = ""
	var category: String = "" # MEMORY, EXECUTION_TIME, RESOURCE_COUNT, TICK_LATENCY
	var value: float = 0.0
	var timestamp: float = 0.0
	var tags: Dictionary = {}

var _enabled: bool = false
var _samples: Array[PerformanceSample] = []
var _max_samples: int = 500

func set_enabled(p_enabled: bool) -> void:
	_enabled = p_enabled

func record_metric(p_name: String, p_category: String, p_val: float, p_tags: Dictionary = {}) -> void:
	if not _enabled:
		return
	if _samples.size() >= _max_samples:
		_samples.pop_front()

	var sample := PerformanceSample.new()
	sample.metric_name = p_name
	sample.category = p_category
	sample.value = p_val
	sample.timestamp = Time.get_unix_time_from_system()
	sample.tags = p_tags.duplicate(true)
	_samples.append(sample)

func get_samples_by_category(p_cat: String) -> Array[PerformanceSample]:
	var res: Array[PerformanceSample] = []
	for s in _samples:
		if s.category == p_cat:
			res.append(s)
	return res

func clear() -> void:
	_samples.clear()
```
