## 一、 核心算法与业务实现 (Core Algorithms & Implementation)

### 1.1 统一数据访问层 (`SaveDataAccessLayer`)

```gdscript
# 模块路径: res://backend/domains/persistence_protocol/save_data_access_layer.gd
class_name SaveDataAccessLayer extends RefCounted:

    static var _domain_contracts: Dictionary = {}   # domain_id -> SaveDomainContract
    static var _dirty_domains: Dictionary = {}      # domain_id -> true（脏标记）
    static var _config_version: int = -1

    ## 域契约新鲜度守卫：domains.json save 段装载，热重载版本推进重建
    static func _ensure_contracts_fresh() -> void:
        if _config_version == GameConfig.config_reload_version():
            return
        _config_version = GameConfig.config_reload_version()
        _domain_contracts.clear()
        for entry in GameConfig.get_array("infrastructure.domains", "domains", []):
            var id := String(entry.get("id", ""))
            var save_cfg := entry.get("save", {})
            if not bool(save_cfg.get("enabled", false)):
                continue
            _domain_contracts[id] = SaveDomainContract.from_config(id, save_cfg)

    ## 统一写入入口：按域契约调度 serialize，装配进统一 payload（Inv-SV-6）
    static func build_save_payload(domain_ids: Array = []) -> Dictionary:
        _ensure_contracts_fresh()
        var providers: Dictionary = {}
        for domain_id in (domain_ids if not domain_ids.is_empty() else _domain_contracts.keys()):
            var contract: SaveDomainContract = _domain_contracts.get(domain_id, null)
            if contract == null or not contract.is_provider_registered():
                continue
            var agg := contract.get_provider_aggregate()
            if agg == null:
                continue
            providers[domain_id] = agg
        return GameSaveAssembler.build_save_payload(providers)

    ## 脏标记写入：仅序列化脏域（增量保存，Inv-SV-10）
    static func mark_dirty(domain_id: String) -> void:
        if _domain_contracts.has(domain_id):
            _dirty_domains[domain_id] = true

    static func flush_dirty() -> Dictionary:
        var dirty_ids: Array = _dirty_domains.keys()
        if dirty_ids.is_empty():
            return { "success": true, "written": 0 }
        var payload := build_save_payload(dirty_ids)
        _dirty_domains.clear()
        # 写入经统一介质（生命周期服务持有的 SaveManager 会话）
        return _persist_via_session(payload)

    ## 统一读取入口：应用存档数据到各域 consumer（Inv-SV-6 反向流）
    static func apply_save_data(data: Dictionary, consumer_domain_ids: Array = []) -> Dictionary:
        _ensure_contracts_fresh()
        var consumers: Dictionary = {}
        for domain_id in (consumer_domain_ids if not consumer_domain_ids.is_empty() else data.keys()):
            var contract: SaveDomainContract = _domain_contracts.get(domain_id, null)
            if contract == null or not contract.is_consumer_registered():
                continue
            consumers[domain_id] = contract.get_consumer_callable()
        return GameSaveAssembler.apply_save_data(data, consumers)
```

**算法要点**：
- **统一入口**（Inv-SV-6）：`build_save_payload/apply_save_data` 为唯一读写门面，内部委托既有 `GameSaveAssembler`（复用 47 域装配，不重写）；
- **脏标记增量**（Inv-SV-10）：`mark_dirty(domain_id)` 只对高频域开放（inventory/economy 契约 `save.strategy="dirty"`），`flush_dirty` 仅序列化脏域；
- **域契约新鲜度**：复用 Phase 64 `config_reload_version()` 模式，domains.json save 段热重载自动重建。

### 1.2 版本迁移引擎 (`SaveMigrationEngine`)

```gdscript
# 模块路径: res://backend/domains/persistence_protocol/save_migration_engine.gd
class_name SaveMigrationEngine extends RefCounted:

    ## 迁移执行：复制 → 逐级迁移 → 校验 → 提交（Inv-SV-7 可回滚）
    static func migrate(data: Dictionary, detected_version: String) -> Dictionary:
        var working := data.duplicate(true)          # 复制（绝不原地改原档）
        var ver := detected_version
        var steps: Array = []
        while ver != GameConfig.get_string("infrastructure.persistence", "format_version", "1.0.0"):
            if not GameSaveAssembler.has_migration(ver):
                return { "success": false, "error_code": "MIGRATION_PATH_BROKEN",
                         "at_version": ver, "partial": working }
            var step := GameSaveAssembler.get_migration(ver)
            working = step.call(working)             # 迁移一步
            steps.append(ver)
            ver = GameSaveAssembler.next_version(ver)
        # 迁移后校验（缺字段回退默认值来源）
        var validation := SaveValidationService.validate_domains(working)
        if not validation.is_empty():
            return { "success": false, "error_code": "MIGRATION_VALIDATION_FAIL",
                     "issues": validation, "partial": working }
        return { "success": true, "data": working, "steps": steps }

    ## 域级迁移：单域 Domain Version 提升（不影响整档 Save Version）
    static func migrate_domain(domain_id: String, domain_data: Dictionary, from_ver: int) -> Dictionary:
        var contract: SaveDomainContract = SaveDataAccessLayer.get_contract(domain_id)
        var target_ver: int = contract.domain_version
        var working := domain_data.duplicate(true)
        var ver := from_ver
        while ver < target_ver:
            var mig_key := "%s:%d" % [domain_id, ver]
            if not GameSaveAssembler.has_domain_migration(mig_key):
                return { "success": false, "error_code": "DOMAIN_MIGRATION_PATH_BROKEN", "domain_id": domain_id }
            working = GameSaveAssembler.get_domain_migration(mig_key).call(working)
            ver += 1
        return { "success": true, "data": working }
```

**算法要点**：
- **复制-迁移-校验-提交**（Inv-SV-7）：`duplicate(true)` 起迁移，全程不动原档；断链（`MIGRATION_PATH_BROKEN`）或校验失败返回 partial + 错误码，调用方决定回滚（原档仍在磁盘）；
- **域级独立演进**：`migrate_domain` 提升单域 Domain Version，不触碰整档 Save Version——与阶段1 三级版本模型对应；
- **可测试可追踪**：迁移步骤列表返回（`steps`），供测试断言与审计留痕。

### 1.3 编辑器热更新管理器 (`EditorHotReloadManager`)

```gdscript
# 模块路径: res://backend/domains/persistence_protocol/editor_hot_reload_manager.gd
class_name EditorHotReloadManager extends RefCounted:

    static var _last_hashes: Dictionary = {}   # 配置表名 -> content hash
    static var _hot_reload_scope: Dictionary = {}  # domain_id -> { allow: bool, requires_rebuild: bool }

    ## 变更检测：对白名单配置表计算内容 hash，与上次比对（Inv-SV-9 领域级）
    static func detect_changes() -> Dictionary:
        var changed: Dictionary = {}
        for table_key in _hot_reload_scope.keys():
            var current := _compute_table_hash(table_key)
            if _last_hashes.has(table_key) and _last_hashes[table_key] != current:
                changed[table_key] = current
        return changed

    ## 配置表内容 SHA-256 哈希计算（复用 GameConfig.get_table 与 JSON 规范化序列化）
    static func _compute_table_hash(table_name: String) -> String:
        return JSON.stringify(GameConfig.get_table(table_name)).sha256_text()

    ## 领域级热更新：确定受影响域 → 重载 → 通知刷新（Inv-SV-9；仅单机，Inv-SV-8）
    static func apply_hot_reload(changed_tables: Dictionary) -> Dictionary:
        if not RuntimeModeGate.is_single_player():
            return { "success": false, "error_code": "HOT_RELOAD_BLOCKED_IN_ONLINE" }
        var affected_domains: Array = _map_tables_to_domains(changed_tables.keys())
        var results: Dictionary = {}
        for domain_id in affected_domains:
            var scope: Dictionary = _hot_reload_scope.get(domain_id, {})
            if not bool(scope.get("allow", false)):
                results[domain_id] = { "skipped": "HOT_RELOAD_DENIED" }
                continue
            if bool(scope.get("requires_rebuild", false)):
                results[domain_id] = { "skipped": "REQUIRES_RUNTIME_REBUILD" }
                continue
            # 重载该域配置并刷新业务系统（经事件总线通知）
            var reloaded := _reload_domain(domain_id)
            EventBus.get_instance().emit_domain_event(
                "persistence.domain_hot_reloaded", { "domain_id": domain_id, "result": reloaded })
            results[domain_id] = reloaded
        # 更新基线 hash（仅成功后，失败保持旧 hash 便于重试）
        if _all_succeeded(results):
            _last_hashes = _last_hashes.merged(changed_tables)
        return { "success": true, "results": results }
```

**算法要点**：
- **领域级、可控、可追踪**（Inv-SV-9）：只重载受影响域，逐个域记录 skip 原因（`HOT_RELOAD_DENIED`/`REQUIRES_RUNTIME_REBUILD`），每域广播 `persistence.domain_hot_reloaded` 事件供业务系统刷新；
- **失败安全**：仅全成功才推进 hash 基线（失败保持旧 hash 可重试）；`requires_rebuild` 域（如战斗参数）拒绝运行时热更，需重建运行时状态；
- **单机隔离**（Inv-SV-8）：入口先过 `RuntimeModeGate.is_single_player()`，联机模式直接阻断（见 1.4）。

### 1.4 运行模式隔离层 (`RuntimeModeGate`)

```gdscript
# 模块路径: res://backend/domains/persistence_protocol/runtime_mode_gate.gd
class_name RuntimeModeGate extends RefCounted:

    enum Mode { SINGLE_PLAYER, ONLINE }

    static var _mode: Mode = Mode.SINGLE_PLAYER
    static var _online_data_source: Variant = null   # 联机专用数据源（服务端权威）

    ## 运行模式切换（启动时由 GameBootstrap 按联机配置决定，一次性）
    static func set_mode(mode: Mode) -> void:
        _mode = mode

    static func is_single_player() -> bool:
        return _mode == Mode.SINGLE_PLAYER

    ## 数据源隔离：单机 = 本地 SaveManager；联机 = 服务端权威源（禁止本地编辑器数据注入）
    static func get_active_data_source() -> Variant:
        if _mode == Mode.ONLINE:
            return _online_data_source   # 联机专用（server_broadcaster/reconciliation 权威源）
        return SaveManager              # 单机本地持久化

    ## 生命周期隔离：热更新/编辑器修改仅单机生命周期注册；联机生命周期无热更新钩子
    static func is_hot_reload_enabled() -> bool:
        return _mode == Mode.SINGLE_PLAYER
```

**算法要点**：
- **四层隔离而非单 `if`**（Inv-SV-8）：运行模式（Mode 枚举，启动一次性设定）+ 数据源（`get_active_data_source` 按模式返回权威源）+ 生命周期（热更新钩子仅单机注册）+ 服务入口（`EditorHotReloadManager.apply_hot_reload` 首道闸）——联机模式四层全部不触发热更新路径；
- **数据源权威**：联机数据源为服务端权威（既有 `server_broadcaster`/`reconciliation_solver`），本地编辑器修改永远无法注入联机数据流。
