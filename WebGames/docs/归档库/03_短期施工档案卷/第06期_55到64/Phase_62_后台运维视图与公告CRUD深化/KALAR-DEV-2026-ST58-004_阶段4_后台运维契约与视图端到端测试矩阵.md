### 2. TestContractTransactionCoordinator 事务化套件（10 tests）

```gdscript
# 文件：tests/test_contract_transaction_coordinator.gd
# 覆盖：事务化派发四步流水线 + 失败路径原子性

extends GdUnit4TestSuite

var _eventbus_events: Array = []

func _enter_suite() -> void:
	EventBus.connect("contract.transaction_committed", Callable(self, "_on_txn_committed"))

func _exit_suite() -> void:
	EventBus.disconnect("contract.transaction_committed", Callable(self, "_on_txn_committed"))

func _on_txn_committed(payload: Dictionary) -> void:
	_eventbus_events.append(payload)

func test_txn_begin_returns_transaction() -> void:
	var txn := ContractTransactionCoordinator.begin({"contract_id": "CT-AS-AC-03", "params": {}, "initiated_by": "gm_001"})
	assert(txn is ContractTransaction).is_ok()
	assert(txn.is_committed() == false).is_ok()

func test_txn_commit_broadcasts_eventbus() -> void:
	var before_count := _eventbus_events.size()
	var txn := ContractTransactionCoordinator.begin({"contract_id": "CT-AS-AC-03", "params": {}, "initiated_by": "gm_001"})
	txn.commit({"success": true})
	await get_tree().process_frame
	assert(_eventbus_events.size() == before_count + 1).is_ok()
	var payload := _eventbus_events.back()
	assert(payload.contract_id == "CT-AS-AC-03").is_ok()
	assert(payload.initiated_by == "gm_001").is_ok()
	assert("timestamp" in payload).is_ok()

func test_txn_dispatch_and_commit_full_pipeline() -> void:
	# 完整事务化流水线
	var params := {"command_id": "cmd_spawn_npc", "payload": {}, "audit_reason": "补偿 GM 测试操作"}
	var txn := ContractTransactionCoordinator.begin({"contract_id": "CT-AS-AC-03", "params": params, "initiated_by": "gm_001"})
	var result := ContractParser.dispatch_ui_command("CT-AS-AC-03", params)

	assert(txn.is_committed()).is_ok()

	var txn := ContractTransactionCoordinator.begin({"contract_id": "CT-AS-AC-03", "params": {}, "initiated_by": "gm_001"})

	var err: ContractError = null
	try:
		txn.commit({"success": true})
	except Error as e:
		err = e

	await get_tree().process_frame
	assert(_eventbus_events.size() == before_count).is_ok()
	assert(err is ContractValidationError).is_ok()

	var before_count := _eventbus_events.size()
```
