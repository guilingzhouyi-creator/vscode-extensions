  - `backend/domains/inventory/wearable_inventory.gd:L46-L60`：入包判定多重循环无缓存计算。

* **核心不变量约束断言**：
  - `Inv-PF-1`（物品统计增量守恒）：任一物品数量变动时，仅对根路径节点执行 `category_node["stats"][key] += delta` 增量变更，任何单个物品操作的时间复杂度严格为 $O(\text{TreeDepth}) = O(1)$，绝对禁止遍历同级未变动叶节点；
  - `Inv-PF-2`（方位解析常数时间）：方位动词解析通过内存静态倒排索引（`String -> Vector2`）完成，查找时间复杂度为 $O(1)$，热路径零临时对象分配；
  - `Inv-PF-3`（触发器实体完全隔离）：空间触发器区域对象（`SpatialTriggerArea`）自身为不可变无状态几何体，触发状态必须由 `(trigger_id, entity_id)` 复合键唯一标识并由注册表集中追踪，不同实体进出事件绝对互不干扰；
  - `Inv-PF-4`（DAG 邻接表索引完备）：剧情因果图初始化时必须构建基于出边源节点的哈希邻接表，推进节点动作时取下游边的复杂度严格为 $O(\text{OutDegree})$，杜绝 $O(|E|)$ 全边遍历；
  - `Inv-PF-5`（DAG 会话可恢复）：运行时会话必须实现 `to_dto()` 与 `from_dto()` 契约，支持将活跃节点集、已完成集与运行时上下文完备持久化与恢复；
  - `Inv-PF-6`（仓储度量增量闭环）：随身背包必须在装载与卸下时同步增量维护 `current_volume` 与 `current_weight`，入包容量判定复杂度严格为 $O(1)$。

* **防漂移最高指示**：
  - 严格保持各领域已有公共 API 的对外签名与返回值兼容性，内部实施透明性能优化，杜绝上层业务代码感知破坏；
  - 严禁在治理过程中引入无界内存泄漏，所有缓存结构必须具备显式清空与热重载刷新机制。

### 1. 物品统计增量数据契约 (`ItemStatDeltaDTO`)

```gdscript
# 模块路径: res://backend/domains/item_statistics/item_stat_delta_dto.gd
class_name ItemStatDeltaDTO extends RefCounted:

	var canonical_id: String = ""
	var category_major: String = ""
	var category_minor: String = ""
	var stat_key: String = ""
	var delta_quantity: int = 0
	var actual_applied: int = 0
	var current_leaf_quantity: int = 0

	func to_dto() -> Dictionary:
		return {
			"canonical_id": canonical_id,
			"category_major": category_major,
			"category_minor": category_minor,
			"stat_key": stat_key,
			"delta_quantity": delta_quantity,
			"actual_applied": actual_applied,
			"current_leaf_quantity": current_leaf_quantity
		}

	static func from_dto(data: Dictionary) -> ItemStatDeltaDTO:
		var dto := ItemStatDeltaDTO.new()
		dto.canonical_id = str(data.get("canonical_id", ""))
		dto.category_major = str(data.get("category_major", ""))
		dto.category_minor = str(data.get("category_minor", ""))
		dto.stat_key = str(data.get("stat_key", ""))
		dto.delta_quantity = int(data.get("delta_quantity", 0))
		dto.actual_applied = int(data.get("actual_applied", 0))
		dto.current_leaf_quantity = int(data.get("current_leaf_quantity", 0))
		return dto
```

### 2. 空间方位倒排索引数据结构 (`DirectionAliasIndex`)

```gdscript
# 模块路径: res://backend/domains/spatial_movement/direction_alias_index.gd
class_name DirectionAliasIndex extends RefCounted:

	## 倒排归一化映射字典：UPPER_CASE_ALIAS -> Vector2.normalized()
	var alias_to_vector_map: Dictionary = {}
	var is_initialized: bool = false
	var last_updated_utc: int = 0

	func to_dto() -> Dictionary:
		var serialized_map := {}
		for k in alias_to_vector_map.keys():
			var v: Vector2 = alias_to_vector_map[k]
			serialized_map[k] = [v.x, v.y]
		return {
			"alias_count": alias_to_vector_map.size(),
			"last_updated_utc": last_updated_utc,
			"map": serialized_map
		}
```

### 3. 多实体空间触发器状态映射实体 (`MultiEntityTriggerRegistry`)

```gdscript
# 模块路径: res://backend/domains/spatial_movement/multi_entity_trigger_registry.gd
class_name MultiEntityTriggerRegistry extends RefCounted:

	## 复合键映射：复合键格式 "{trigger_id}::{entity_id}" -> int (SpatialTriggerFSM.TriggerState)
	var _state_table: Dictionary = {}

	func get_state(trigger_id: String, entity_id: String) -> int:
		var key := _build_composite_key(trigger_id, entity_id)
		return _state_table.get(key, SpatialTriggerFSM.TriggerState.OUTSIDE)

	func update_state(trigger_id: String, entity_id: String, new_state: int) -> int:
		var key := _build_composite_key(trigger_id, entity_id)
		var prev: int = _state_table.get(key, SpatialTriggerFSM.TriggerState.OUTSIDE)
		_state_table[key] = new_state
		return prev

	func remove_entity(entity_id: String) -> void:
		var prefix := "::" + entity_id
		var to_remove := []
		for k in _state_table.keys():
			if str(k).ends_with(prefix):
				to_remove.append(k)
		for rk in to_remove:
			_state_table.erase(rk)

	func clear() -> void:
		_state_table.clear()

	static func _build_composite_key(trigger_id: String, entity_id: String) -> String:
		return "%s::%s" % [trigger_id, entity_id]
```

### 4. 剧情 DAG 会话持久化快照契约 (`NarrativeDagSessionDTO`)

```gdscript
# 模块路径: res://backend/domains/narrative_orchestration/narrative_dag_session_dto.gd
class_name NarrativeDagSessionDTO extends RefCounted:

	var graph_id: String = ""
	var active_node_ids: Array[String] = []
	var completed_node_ids: Array[String] = []
	var runtime_context: Dictionary = {}
	var is_terminated: bool = false
	var snapshot_timestamp_utc: int = 0

	func to_dto() -> Dictionary:
		return {
			"graph_id": graph_id,
			"active_node_ids": active_node_ids.duplicate(),
			"completed_node_ids": completed_node_ids.duplicate(),
			"runtime_context": runtime_context.duplicate(true),
			"is_terminated": is_terminated,
			"snapshot_timestamp_utc": snapshot_timestamp_utc
		}

	static func from_dto(data: Dictionary) -> NarrativeDagSessionDTO:
		var dto := NarrativeDagSessionDTO.new()
		dto.graph_id = str(data.get("graph_id", ""))
		dto.active_node_ids.assign(data.get("active_node_ids", []))
		dto.completed_node_ids.assign(data.get("completed_node_ids", []))
		dto.runtime_context = (data.get("runtime_context", {}) as Dictionary).duplicate(true)
		dto.is_terminated = bool(data.get("is_terminated", false))
		dto.snapshot_timestamp_utc = int(data.get("snapshot_timestamp_utc", 0))
		return dto
```
