### 1. 移动计算参数结构化与热重载监听

```gdscript
# 模块路径: res://backend/domains/spatial_movement/movement_vector_bridge_solver.gd
class MovementConfigParameters extends RefCounted:
	var agi_ref: float = 10.0
	var agi_factor_per: float = 0.005
	var agi_min: float = 0.5
	var load_threshold: float = 30.0
	var load_divisor: float = 100.0
	var load_min: float = 0.2
	var impedance_floor: float = 0.1
	var facing_thresh_sq: float = 0.0001

static var _movement_params: MovementConfigParameters = null

static func reload_movement_config() -> void:
	var p := MovementConfigParameters.new()
	p.agi_ref = GameConfig.get_float("domains.spatial_movement", "movement/agi_reference_stat", 10.0)
	p.agi_factor_per = GameConfig.get_float("domains.spatial_movement", "movement/agi_factor_per_point", 0.005)
	p.agi_min = GameConfig.get_float("domains.spatial_movement", "movement/agi_factor_min", 0.5)
	p.load_threshold = GameConfig.get_float("domains.spatial_movement", "movement/load_threshold_kg", 30.0)
	p.load_divisor = GameConfig.get_float("domains.spatial_movement", "movement/load_excess_divisor", 100.0)
	p.load_min = GameConfig.get_float("domains.spatial_movement", "movement/load_penalty_min", 0.2)
	p.impedance_floor = GameConfig.get_float("domains.spatial_movement", "movement/impedance_floor", 0.1)
	p.facing_thresh_sq = GameConfig.get_float("domains.spatial_movement", "movement/facing_threshold_sq", 0.0001)
	_movement_params = p
	build_direction_cache()

## 参数化有效速度计算（从本地结构读取，零实时读配置开销）
static func calculate_effective_speed(
	base_speed: float,
	agi_stat: float,
	total_load_kg: float,
	terrain_impedance_multiplier: float = 1.0
) -> float:
	if _movement_params == null:
		reload_movement_config()
	var p := _movement_params
	var agi_factor := maxf(p.agi_min, 1.0 + (agi_stat - p.agi_ref) * p.agi_factor_per)
	var load_penalty := 1.0
	if total_load_kg > p.load_threshold:
		var excess := total_load_kg - p.load_threshold
		load_penalty = maxf(p.load_min, 1.0 - (excess / p.load_divisor))
	var safe_impedance := maxf(p.impedance_floor, terrain_impedance_multiplier)
	return (base_speed * agi_factor * load_penalty) / safe_impedance
```

### 2. CDKey 频控状态字典有界清理与 TTL 淘汰防线

```gdscript
# 模块路径: res://backend/domains/cdkey_voucher/cdkey_redemption_solver.gd
## 具有最大容量守卫的失败频控记录（防内存无界增长）
static func _record_failure_bounded(
	account_id: String,
	current_time_utc: int,
	rate_limit_state: Dictionary,
	max_capacity: int = 5000
) -> void:
	if rate_limit_state.size() >= max_capacity:
		_purge_expired_rate_limits(rate_limit_state, current_time_utc)
		# 若清理后仍满，强制淘汰最前一项
		if rate_limit_state.size() >= max_capacity:
			var first_key = rate_limit_state.keys()[0]
			rate_limit_state.erase(first_key)

	var info: Dictionary = rate_limit_state.get(account_id, { "count": 0, "lock_until_utc": 0 })
	var cnt: int = int(info.get("count", 0)) + 1
	var lock_until: int = 0
	if cnt >= _max_failed_attempts():
		lock_until = current_time_utc + _lockout_duration_seconds()
		cnt = 0
	rate_limit_state[account_id] = { "count": cnt, "lock_until_utc": lock_until }

static func _purge_expired_rate_limits(rate_limit_state: Dictionary, current_time_utc: int) -> void:
	var expired_keys := []
	for acc in rate_limit_state.keys():
		var item: Dictionary = rate_limit_state[acc]
		if int(item.get("lock_until_utc", 0)) < current_time_utc and int(item.get("count", 0)) == 0:
			expired_keys.append(acc)
	for k in expired_keys:
		rate_limit_state.erase(k)
```

### 3. 剧情因果 DAG 引擎会话持久化与恢复对接

```gdscript
# 模块路径: res://backend/domains/narrative_orchestration/narrative_dag_execution_engine.gd
## 导出运行时会话状态快照
func export_session_dto() -> NarrativeDagSessionDTO:
	var dto := NarrativeDagSessionDTO.new()
	dto.graph_id = graph.graph_id if graph else ""
	dto.active_node_ids = active_node_ids.duplicate()
	dto.completed_node_ids = completed_node_ids.duplicate()
	dto.runtime_context = runtime_context.duplicate(true)
	dto.is_terminated = is_terminated
	dto.snapshot_timestamp_utc = int(Time.get_unix_time_from_system())
	return dto

## 从快照恢复运行时会话状态
func restore_session_dto(dto: NarrativeDagSessionDTO, p_graph: NarrativeDAGGraphDTO) -> Dictionary:
	if dto == null or p_graph == null or dto.graph_id != p_graph.graph_id:
		return { "success": false, "error_code": "INVALID_RESTORE_PAYLOAD" }
	graph = p_graph
	_build_graph_indices()
	active_node_ids.assign(dto.active_node_ids)
	completed_node_ids.assign(dto.completed_node_ids)
	runtime_context = dto.runtime_context.duplicate(true)
	is_terminated = dto.is_terminated
	_completed_set.clear()
	for cid in completed_node_ids:
		_completed_set[cid] = true
	return { "success": true, "restored_nodes": active_node_ids.size() }
```
