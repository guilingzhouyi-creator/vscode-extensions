### 1. 生命周期有限状态机 (GameLifecycleManager)

```gdscript
# 模块路径: res://backend/domains/lifecycle/game_lifecycle_manager.gd
class_name GameLifecycleManager
extends RefCounted

signal lifecycle_state_changed(previous_phase: int, new_phase: int)

static var _instance: GameLifecycleManager = null
var _current_phase: int = GameLifecycleModel.LifecyclePhase.UNINITIALIZED
var _shutdown_in_progress: bool = false
var _active_shutdown_dto: GameShutdownRequestDTO = null

static func get_instance() -> GameLifecycleManager:
	if _instance == null:
		_instance = GameLifecycleManager.new()
	return _instance

func get_current_phase() -> int:
	return _current_phase

## 推进状态跃迁（写权限内部收敛）
func transition_to(target_phase: int) -> bool:
	if not _is_valid_transition(_current_phase, target_phase):
		EventBus.get_instance().emit_log("error", "非法生命周期状态跃迁: %d -> %d" % [_current_phase, target_phase])
		return false

	var prev := _current_phase
	_current_phase = target_phase
	lifecycle_state_changed.emit(prev, target_phase)
	EventBus.get_instance().emit_domain_event("lifecycle.phase_changed", {
		"args": [prev, target_phase],
		"summary": {"previous": prev, "current": target_phase}
	})
	return true

func _is_valid_transition(from_phase: int, to_phase: int) -> bool:
	match from_phase:
		GameLifecycleModel.LifecyclePhase.UNINITIALIZED:
			return to_phase == GameLifecycleModel.LifecyclePhase.BOOT_INITIALIZING
		GameLifecycleModel.LifecyclePhase.BOOT_INITIALIZING:
			return to_phase == GameLifecycleModel.LifecyclePhase.RUNNING or to_phase == GameLifecycleModel.LifecyclePhase.STOPPED
		GameLifecycleModel.LifecyclePhase.RUNNING:
			return to_phase == GameLifecycleModel.LifecyclePhase.PAUSED or to_phase == GameLifecycleModel.LifecyclePhase.STOPPING
		GameLifecycleModel.LifecyclePhase.PAUSED:
			return to_phase == GameLifecycleModel.LifecyclePhase.RUNNING or to_phase == GameLifecycleModel.LifecyclePhase.STOPPING
		GameLifecycleModel.LifecyclePhase.STOPPING:
			return to_phase == GameLifecycleModel.LifecyclePhase.STOPPED or to_phase == GameLifecycleModel.LifecyclePhase.STOP_FAILED
		GameLifecycleModel.LifecyclePhase.STOP_FAILED:
			return to_phase == GameLifecycleModel.LifecyclePhase.FORCE_STOPPING
		GameLifecycleModel.LifecyclePhase.FORCE_STOPPING:
			return to_phase == GameLifecycleModel.LifecyclePhase.STOPPED
		GameLifecycleModel.LifecyclePhase.STOPPED:
			return false  ## 终态不可再跃迁
	return false
```

### 2. 三阶段停机编排门面服务 (GameLifecycleService)

```gdscript
# 模块路径: res://backend/domains/lifecycle/game_lifecycle_service.gd
class_name GameLifecycleService
extends RefCounted

## 执行安全停机总管线
static func execute_graceful_shutdown(request: GameShutdownRequestDTO) -> GameShutdownResponseDTO:
	var fsm := GameLifecycleManager.get_instance()
	var start_time := Time.get_ticks_msec()
	var resp := GameShutdownResponseDTO.new()
	resp.current_phase = fsm.get_current_phase()

	# 1. 状态守卫：防止重复停机或非法状态调用
	if fsm.get_current_phase() == GameLifecycleModel.LifecyclePhase.STOPPING:
		resp.success = false
		resp.error_code = "ALREADY_STOPPING"
		return resp
	if fsm.get_current_phase() == GameLifecycleModel.LifecyclePhase.STOPPED:
		resp.success = true
		resp.error_code = "ALREADY_STOPPED"
		return resp

	# 2. 跃迁至 STOPPING 态
	if not fsm.transition_to(GameLifecycleModel.LifecyclePhase.STOPPING):
		resp.success = false
		resp.error_code = "TRANSITION_REJECTED"
		return resp

	# 3. 阶段一：冻结所有外来输入与调度
	_freeze_world_and_inputs()

	# 4. 阶段二：执行全域数据持久化强制落盘
	var auto_save: bool = GameConfig.get_bool("infrastructure.lifecycle", "policies/auto_save_on_exit", true)
	if auto_save:
		var save_result := _flush_domain_saves(request.target_save_slot)
		resp.is_save_completed = save_result.get("success", false)
		resp.save_sha256 = String(save_result.get("sha256", ""))
		if not resp.is_save_completed:
			EventBus.get_instance().emit_log("error", "退出存盘失败: " + str(save_result.get("message", "")))

	# 5. 阶段三：注销会话与清理外部连接
	if not request.session_token.is_empty():
		AuthService.revoke_token(request.session_token)

	# 6. 阶段四：调用反装配清理单例与注册表
	GameBootstrap.teardown()

	# 7. 跃迁至 STOPPED 终态
	fsm.transition_to(GameLifecycleModel.LifecyclePhase.STOPPED)
	resp.success = true
	resp.error_code = "OK"
	resp.current_phase = GameLifecycleModel.LifecyclePhase.STOPPED
	resp.elapsed_milliseconds = Time.get_ticks_msec() - start_time
	return resp

static func _freeze_world_and_inputs() -> void:
	EventBus.get_instance().emit_domain_event("lifecycle.inputs_frozen", {})

static func _flush_domain_saves(target_slot: String) -> Dictionary:
	var slot_name := target_slot if not target_slot.is_empty() else "quicksave_exit"
	# 收集各已注册领域的序列化聚合
	var providers := {} # 由当前运行上下文中的活跃领域实例填充
	var payload := GameSaveAssembler.build_save_payload(providers)
	return SaveManager.save_game(slot_name, payload)
```

### 3. 前端退出确认弹窗 (ExitConfirmDialog)

```gdscript
# 模块路径: res://frontend/views/common/exit_confirm_dialog.gd
class_name ExitConfirmDialog
extends PanelContainer

signal confirmed_exit(reason: int)
signal cancelled_exit

@onready var _label_title: Label = %LabelTitle
@onready var _label_message: Label = %LabelMessage
@onready var _btn_confirm: Button = %BtnConfirm
@onready var _btn_cancel: Button = %BtnCancel
@onready var _spinner_progress: ProgressBar = %SpinnerProgress

var _is_processing: bool = false

func _ready() -> void:
	_btn_confirm.pressed.connect(_on_confirm_pressed)
	_btn_cancel.pressed.connect(_on_cancel_pressed)
	_spinner_progress.visible = false

func popup_confirm(reason_text_key: String = "ui.dialog.exit.confirm_message") -> void:
	visible = true
	_is_processing = false
	_btn_confirm.disabled = false
	_btn_cancel.disabled = false
	_spinner_progress.visible = false
	UIIntermediary.resolve(_label_title, "ui.dialog.exit.title")
	UIIntermediary.resolve(_label_message, reason_text_key)

func _on_confirm_pressed() -> void:
	if _is_processing:
		return
	_is_processing = true
	_btn_confirm.disabled = true
	_btn_cancel.disabled = true
	_spinner_progress.visible = true
	confirmed_exit.emit(GameLifecycleModel.ExitReason.USER_DESKTOP_QUIT)

func _on_cancel_pressed() -> void:
	if _is_processing:
		return
	visible = false
	cancelled_exit.emit()
```
