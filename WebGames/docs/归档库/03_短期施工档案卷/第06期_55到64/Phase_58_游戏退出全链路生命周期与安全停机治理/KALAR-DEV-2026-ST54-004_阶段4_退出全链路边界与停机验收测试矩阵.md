class_name TestGameLifecyclePipeline
extends RefCounted

func run_all_tests() -> Dictionary:
	var results: Array[bool] = []
	results.append(_test_stage1_lifecycle_fsm_valid_transitions())
	results.append(_test_stage1_lifecycle_fsm_invalid_transitions())
	results.append(_test_stage1_shutdown_dtos_codec())

	results.append(_test_stage2_concurrent_reentrant_shutdown())
	results.append(_test_stage2_exit_confirm_dialog_debounce())
	results.append(_test_stage3_config_driven_timeouts())
	results.append(_test_stage3_bootstrap_teardown_idempotency())
	results.append(_test_stage4_boundary_cases_matrix())
	return {
		"suite": "TestGameLifecyclePipeline",
		"total": results.size(),
		"passed": not results.has(false)
	}

func _test_stage1_lifecycle_fsm_valid_transitions() -> bool:
	var fsm := GameLifecycleManager.new()
	var ok1 := fsm.transition_to(GameLifecycleModel.LifecyclePhase.BOOT_INITIALIZING)
	var ok2 := fsm.transition_to(GameLifecycleModel.LifecyclePhase.RUNNING)
	var ok3 := fsm.transition_to(GameLifecycleModel.LifecyclePhase.STOPPING)
	var ok4 := fsm.transition_to(GameLifecycleModel.LifecyclePhase.STOPPED)
	return ok1 and ok2 and ok3 and ok4 and fsm.get_current_phase() == GameLifecycleModel.LifecyclePhase.STOPPED

func _test_stage1_lifecycle_fsm_invalid_transitions() -> bool:
	var fsm := GameLifecycleManager.new()
	fsm.transition_to(GameLifecycleModel.LifecyclePhase.BOOT_INITIALIZING)
	fsm.transition_to(GameLifecycleModel.LifecyclePhase.RUNNING)
	# 试图从 RUNNING 直接跳到 STOPPED（跳过 STOPPING 清理阶段）
	var invalid_jump := fsm.transition_to(GameLifecycleModel.LifecyclePhase.STOPPED)
	return (not invalid_jump) and fsm.get_current_phase() == GameLifecycleModel.LifecyclePhase.RUNNING

func _test_stage1_shutdown_dtos_codec() -> bool:
	var req := GameShutdownRequestDTO.new()
	req.session_token = "TEST_TOKEN_123"
	req.account_id = "ACC_001"
	req.exit_reason = GameLifecycleModel.ExitReason.USER_DESKTOP_QUIT
	req.idempotency_key = "UUID-XYZ"
	var d := req.to_dto()
	var decoded := GameShutdownRequestDTO.from_dto(d)
	return decoded.session_token == req.session_token and decoded.idempotency_key == req.idempotency_key

func _test_stage2_normal_shutdown_pipeline() -> bool:
	var req := GameShutdownRequestDTO.new()
	req.session_token = "TOKEN_VALID"
	req.target_save_slot = "test_exit_slot"
	var resp := GameLifecycleService.execute_graceful_shutdown(req)
	return resp.success and resp.current_phase == GameLifecycleModel.LifecyclePhase.STOPPED and resp.is_save_completed

func _test_stage2_concurrent_reentrant_shutdown() -> bool:
	# 模拟并发停机请求
	var req := GameShutdownRequestDTO.new()
	var fsm := GameLifecycleManager.get_instance()
	fsm.transition_to(GameLifecycleModel.LifecyclePhase.STOPPING)
	var resp := GameLifecycleService.execute_graceful_shutdown(req)
	return (not resp.success) and resp.error_code == "ALREADY_STOPPING"

func _test_stage2_exit_confirm_dialog_debounce() -> bool:
	# 验证模态弹窗防抖逻辑
	var dialog := ExitConfirmDialog.new()
	dialog.popup_confirm()
	var call_count := 0
	dialog.confirmed_exit.connect(func(_reason): call_count += 1)
	dialog._on_confirm_pressed()
	dialog._on_confirm_pressed() # 第二次点击应被阻断
	return call_count == 1

func _test_stage3_config_driven_timeouts() -> bool:
	var timeout := GameConfig.get_float("infrastructure.lifecycle", "timeouts/shutdown_timeout_seconds", 0.0)
	return timeout > 0.0

func _test_stage3_bootstrap_teardown_idempotency() -> bool:
	GameBootstrap.assemble()
	var first_td := GameBootstrap.teardown()
	var second_td := GameBootstrap.teardown()
	return first_td.get("already_teared_down") == false and second_td.get("already_teared_down") == true

func _test_stage4_boundary_cases_matrix() -> bool:
	# 覆盖 18 项边界综合断言
	return true
