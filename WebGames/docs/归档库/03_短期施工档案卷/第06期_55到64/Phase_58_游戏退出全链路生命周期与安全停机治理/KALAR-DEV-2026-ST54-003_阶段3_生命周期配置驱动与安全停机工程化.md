### 2. GameBootstrap 对称反装配实现 (Teardown)

```gdscript
# 模块路径: res://backend/infrastructure/game_bootstrap.gd
## 全域反装配（幂等）：释放静态常驻单例、解绑全局事件监听器并重置状态
static func teardown() -> Dictionary:
	if not _assembled:
		return {"success": true, "already_teared_down": true}

	# 1. 释放地面掉落监听器与事件总线解绑
	if _ground_loot_listener != null:
		if _ground_loot_listener.has_method("dispose"):
			_ground_loot_listener.dispose()
		_ground_loot_listener = null

	# 2. 清理各注册表单例引用
	_catalog = null
	_magic_registry = null
	_magic_rules = null
	_quality_registry = null

	# 3. 复位全局装配标记
	_assembled = false

	# 4. 广播引擎反装配完成事件
	EventBus.get_instance().emit_domain_event("engine.teardown_completed", {
		"summary": {"timestamp_utc": int(Time.get_unix_time_from_system())}
	})

	return {
		"success": true,
		"already_teared_down": false,
		"catalog_released": true,
		"listeners_disposed": true
	}
```

### 3. audit_config 静态值域校验规则 (VD_LC_*)

```python
# 脚本路径: scripts/py/audit_config.py
def _validate_lifecycle_config(data: dict) -> list[str]:
    errors = []
    timeouts = data.get("timeouts", {})
    shutdown_to = timeouts.get("shutdown_timeout_seconds", 0.0)
    save_to = timeouts.get("save_flush_timeout_seconds", 0.0)
    force_to = timeouts.get("force_kill_timeout_seconds", 0.0)

    # VD_LC_01: shutdown_timeout 必须大于 save_flush_timeout
    if shutdown_to <= save_to:
        errors.append(f"VD_LC_01: shutdown_timeout_seconds({shutdown_to}) 必须大于 save_flush_timeout_seconds({save_to})")

    # VD_LC_02: force_kill_timeout 必须在 [1.0, 10.0] 区间内
    if not (1.0 <= force_to <= 10.0):
        errors.append(f"VD_LC_02: force_kill_timeout_seconds({force_to}) 超出合法区间 [1.0, 10.0]")

    # VD_LC_03: poll_interval_ms 必须在 [10, 1000] 之间
    poll_ms = timeouts.get("poll_interval_ms", 0)
    if not (10 <= poll_ms <= 1000):
        errors.append(f"VD_LC_03: poll_interval_ms({poll_ms}) 超出合法区间 [10, 1000]")

    return errors
```
