### 5. 反序列化类型守卫复用（Phase 56 S1 范式）

```gdscript
# 位置：frontend/views/base_view_contract_binder.gd（追加反序列化守卫）

func _sanitize_contract_payload(payload: Dictionary, contract_id: String) -> Dictionary:
	# 复用 Phase 56 S1 _sanitize_section 范式
	var entry := ContractRegistryIndex.get_contract(contract_id)
	if entry == null:
		return payload
	var sanitized := {}
	for field in entry.field_mapping:
		var backend_val: Variant = payload.get(field.backend_field)
		if backend_val == null:
			sanitized[field.backend_field] = null
			continue
		# 类型守卫：按 transform_config.to 校验类型
		var expected_type: String = field.transform_config.get("to", "identity")
		if not _check_type(backend_val, expected_type):
			PushWarning("Contract field type mismatch in %s: %s" % [contract_id, field.backend_field])
			sanitized[field.backend_field] = null
			continue
		sanitized[field.backend_field] = backend_val
	return sanitized

func _check_type(value: Variant, expected: String) -> bool:
	match expected:
		"int": return value is int
		"float": return value is int or value is float
		"String": return value is String
		"bool": return value is bool
		"Array": return value is Array
		"Dictionary": return value is Dictionary
		_: return true
```
