## 一、 核心算法与求解器设计

### 1. 泛化路径寻址算法 (JsonPathResolver)

支持点分/斜杠路径、数组通配符 `*` 以及嵌套属性展开：

```python
class JsonPathResolver:
    @staticmethod
    def resolve(data: Any, path: str) -> List[Tuple[str, Any]]:
        """
        解析路径并返回 [(具体路径字符串, 取值)] 列表。
        支持语法：
          - "point_cost/tier_size" -> 单值寻址
          - "entries/*/contract_id" -> 数组通配展开
          - "attributes/*/school" -> 字典通配展开
        """
        parts = path.replace(".", "/").strip("/").split("/")
        return JsonPathResolver._traverse(data, parts, "")

    @staticmethod
    def _traverse(node: Any, parts: List[str], current_path: str) -> List[Tuple[str, Any]]:
        if not parts:
            return [(current_path, node)]
        head, *tail = parts
        results: List[Tuple[str, Any]] = []

        if head == "*":
            if isinstance(node, list):
                for idx, item in enumerate(node):
                    elem_path = f"{current_path}[{idx}]" if current_path else f"[{idx}]"
                    results.extend(JsonPathResolver._traverse(item, tail, elem_path))
            elif isinstance(node, dict):
                for k, v in node.items():
                    elem_path = f"{current_path}/{k}" if current_path else k
                    results.extend(JsonPathResolver._traverse(v, tail, elem_path))
        elif isinstance(node, dict) and head in node:
            next_path = f"{current_path}/{head}" if current_path else head
            results.extend(JsonPathResolver._traverse(node[head], tail, next_path))
        return results
```

### 2. 声明式规则求值引擎 (DeclarativeRuleEvaluator)

分派执行 9 大规则原语：

```python
class DeclarativeRuleEvaluator:
    def __init__(self, all_configs: Dict[str, dict]):
        self.all_configs = all_configs  # 用于外键跨表检索

    def evaluate_rule(self, rule: dict, data: dict, rel_posix: str) -> List[Finding]:
        primitive = rule.get("primitive")
        handler = getattr(self, f"_eval_{primitive}", None)
        if not handler:
            return [Finding(
                rule_id="RULE_ENGINE_UNKNOWN_PRIMITIVE",
                category="config",
                file_path=rel_posix,
                line_number=0,
                message=f"未知的规则原语类型: {primitive}",
                severity="ERROR"
            )]
        return handler(rule, data, rel_posix)

    def _eval_positive_num(self, rule: dict, data: dict, rel_posix: str) -> List[Finding]:
        findings = []
        allow_float = rule.get("params", {}).get("allow_float", True)
        min_exclusive = rule.get("params", {}).get("min_exclusive", 0)
        matches = JsonPathResolver.resolve(data, rule["key_path"])

        for loc, val in matches:
            if val is None:
                continue
            if isinstance(val, bool) or not isinstance(val, (int, float)):
                findings.append(Finding(
                    rule_id=rule["rule_id"], category="config", file_path=rel_posix, line_number=0,
                    message=f"{loc} 期望数值类型，实测为 {type(val).__name__}",
                    severity=rule.get("severity", "ERROR")
                ))
            elif not allow_float and isinstance(val, float):
                findings.append(Finding(
                    rule_id=rule["rule_id"], category="config", file_path=rel_posix, line_number=0,
                    message=f"{loc} 必须为整数，实测为浮点数 {val}",
                    severity=rule.get("severity", "ERROR")
                ))
            elif val <= min_exclusive:
                findings.append(Finding(
                    rule_id=rule["rule_id"], category="config", file_path=rel_posix, line_number=0,
                    message=f"{loc} 必须 > {min_exclusive}，实测为 {val}",
                    severity=rule.get("severity", "ERROR")
                ))
        return findings

    def _eval_bounded_range(self, rule: dict, data: dict, rel_posix: str) -> List[Finding]:
        findings = []
        p = rule.get("params", {})
        min_v, max_v = p.get("min_val"), p.get("max_val")
        inclusive = p.get("inclusive", True)
        matches = JsonPathResolver.resolve(data, rule["key_path"])

        for loc, val in matches:
            if val is None:
                continue
            if isinstance(val, bool) or not isinstance(val, (int, float)):
                findings.append(Finding(
                    rule_id=rule["rule_id"], category="config", file_path=rel_posix, line_number=0,
                    message=f"{loc} 必须为数值，实测为 {type(val).__name__}", severity=rule.get("severity", "ERROR")
                ))
            else:
                out_of_bound = (val < min_v or val > max_v) if inclusive else (val <= min_v or val >= max_v)
                if out_of_bound:
                    op = "[]" if inclusive else "()"
                    findings.append(Finding(
                        rule_id=rule["rule_id"], category="config", file_path=rel_posix, line_number=0,
                        message=f"{loc} 越界: 值 {val} 不在区间 {op[0]}{min_v}, {max_v}{op[1]} 内",
                        severity=rule.get("severity", "ERROR")
                    ))
        return findings

    def _eval_range_ordered(self, rule: dict, data: dict, rel_posix: str) -> List[Finding]:
        findings = []
        lefts = JsonPathResolver.resolve(data, rule["key_path_left"])
        rights = JsonPathResolver.resolve(data, rule["key_path_right"])
        if lefts and rights:
            l_val, r_val = lefts[0][1], rights[0][1]
            if isinstance(l_val, (int, float)) and isinstance(r_val, (int, float)) and not isinstance(l_val, bool) and not isinstance(r_val, bool):
                strict_less = rule.get("params", {}).get("strict_less", True)
                violates = (l_val >= r_val) if strict_less else (l_val > r_val)
                if violates:
                    op = "<" if strict_less else "<="
                    findings.append(Finding(
                        rule_id=rule["rule_id"], category="config", file_path=rel_posix, line_number=0,
                        message=f"区间倒置: {rule['key_path_left']}({l_val}) 必须 {op} {rule['key_path_right']}({r_val})",
                        severity=rule.get("severity", "ERROR")
                    ))
        return findings

    def _eval_enum_whitelist(self, rule: dict, data: dict, rel_posix: str) -> List[Finding]:
        findings = []
        whitelist = set(rule.get("params", {}).get("whitelist", []))
        arr_path = rule.get("array_path")
        key_path = rule.get("item_key_path", "")
        full_path = f"{arr_path}/*/{key_path}" if arr_path else key_path
        matches = JsonPathResolver.resolve(data, full_path)

        for loc, val in matches:
            if val is not None and val not in whitelist:
                findings.append(Finding(
                    rule_id=rule["rule_id"], category="config", file_path=rel_posix, line_number=0,
                    message=f"{loc} 非法枚举值 '{val}'，有效白名单为: {sorted(list(whitelist))}",
                    severity=rule.get("severity", "ERROR")
                ))
        return findings

    def _eval_unique_key(self, rule: dict, data: dict, rel_posix: str) -> List[Finding]:
        findings = []
        full_path = f"{rule['array_path']}/*/{rule['item_key_path']}"
        matches = JsonPathResolver.resolve(data, full_path)
        seen = {}
        for loc, val in matches:
            if val is not None:
                if val in seen:
                    findings.append(Finding(
                        rule_id=rule["rule_id"], category="config", file_path=rel_posix, line_number=0,
                        message=f"唯一键冲突: 键 '{val}' 在 {seen[val]} 与 {loc} 重复",
                        severity=rule.get("severity", "ERROR")
                    ))
                else:
                    seen[val] = loc
        return findings

    def _eval_foreign_key_reference(self, rule: dict, data: dict, rel_posix: str) -> List[Finding]:
        findings = []
        src_path = f"{rule['source_dict_path']}/*/{rule['source_item_key']}"
        target_dict_path = rule["target_dict_path"]
        target_file = rule.get("target_file_override", rel_posix)
        target_data = self.all_configs.get(target_file, data)

        targets = JsonPathResolver.resolve(target_data, target_dict_path)
        valid_keys = set(targets[0][1].keys()) if targets and isinstance(targets[0][1], dict) else set()

        sources = JsonPathResolver.resolve(data, src_path)
        for loc, ref in sources:
            if ref is not None and ref not in valid_keys:
                findings.append(Finding(
                    rule_id=rule["rule_id"], category="config", file_path=rel_posix, line_number=0,
                    message=f"{loc} 引用未定义外键 '{ref}'（目标 {target_file}:{target_dict_path}）",
                    severity=rule.get("severity", "ERROR")
                ))
        return findings
```

```python
DOMAIN_PRIVATE_ACCESS_RE = re.compile(r"\b([a-z][a-z0-9_]*_service|[a-z][a-z0-9_]*_manager|[a-z][a-z0-9_]*_aggregate)\._([a-z][a-z0-9_]+)\b")

def check_cross_domain_boundary(path: Path, lines: List[str], violations: List[Finding]) -> None:
    rel_posix = path.relative_to(ROOT).as_posix()
    # 仅约束 backend/domains/ 生产代码，排除 tests/
    if not rel_posix.startswith("backend/domains/"):
        return
    current_domain = rel_posix.split("/")[2] if len(rel_posix.split("/")) > 2 else ""

    for line_idx, line in enumerate(lines, 1):
        # 忽略注释
        code = line.split("#")[0].strip()
        m = DOMAIN_PRIVATE_ACCESS_RE.search(code)
        if m:
            target_var, target_private = m.group(1), m.group(2)
            # 若变量名暗示跨域（如其他域的服务），判定违规
            violations.append(Finding(
                rule_id="ADV-BND-001",
                category="gdscript",
                file_path=rel_posix,
                line_number=line_idx,
                message=f"检测到跨对象直接访问私有成员 '{target_var}._{target_private}'，严禁跨边界私有穿透",
                severity="ERROR",
                suggestion="改用目标对象的公共接口或通过事件广播交互"
            ))
```
