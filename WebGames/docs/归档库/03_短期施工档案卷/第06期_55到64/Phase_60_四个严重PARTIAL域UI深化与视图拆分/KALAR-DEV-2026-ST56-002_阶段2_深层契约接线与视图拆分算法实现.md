## 一、 数据结构设计与代码契约 (Data Structures & Code Contracts)

### 1. AccountEntry 视图接线（承接 character_creation）

```gdscript
# 位置：frontend/views/account_entry.gd（追加接线代码）

# 追加字段
@onready var _creation_progress_bar: ProgressBar = %CreationProgressBar
@onready var _creation_option_buttons: VBoxContainer = %CreationOptionButtons
@onready var _attribute_preview_panel: Control = %AttributePreviewPanel

func _contract_init() -> void:
	# 绑定 character_creation 5 条契约（Phase 57 S1 首批 1 条 + Phase 60 S1 增量 4 条）
	var entries := ContractRegistryIndex.find_by_domain_view("character_creation", "account_entry")
	_binder.bind_contracts(entries)

# FSM 状态可见回调
func _on_creation_state_changed(event: Dictionary) -> void:
	var state := event.get("current_state", "")
	var progress := event.get("progress_percentage", 0.0)
	_creation_progress_bar.value = progress
	_creation_progress_bar.tooltip_text = _translate_state(state)
	_refresh_option_buttons(event.get("available_options", []))

func _refresh_option_buttons(options: Array) -> void:
	for btn in _creation_option_buttons.get_children():
		btn.queue_free()
	for opt in options:
		var btn := Button.new()
		btn.text = opt.get("label", "")
		btn.pressed.connect(_on_creation_option_pressed.bind(opt.get("value")))
		_creation_option_buttons.add_child(btn)

func _on_creation_option_pressed(value: String) -> void:
	_dispatch_contract_command("CT-CC-AE-04", {"target_state": value})

func _on_race_changed(race: String) -> void:
	var result := _dispatch_contract_command("CT-CC-AE-05", {
		"race": race,
		"gender": _selected_gender,
		"role": _selected_role
	})
	if result.get("accepted"):
		_refresh_contract_field("CT-CC-AE-03", result.get("response", {}))
```

### 4. InventoryView 新视图完整接线（承接 inventory）

```gdscript
# 位置：frontend/views/inventory_view.gd（新视图完整实现）

class_name InventoryView
extends BaseViewContractBinder

@onready var _backpack_panel: Control = %BackpackPanel
@onready var _item_grid: GridContainer = %ItemGrid
@onready var _capacity_bar: ProgressBar = %CapacityBar
@onready var _appraise_btn: Button = %AppraiseBtn

func _contract_init() -> void:
	var entries := ContractRegistryIndex.find_by_domain_view("inventory", "inventory_view")
	_binder.bind_contracts(entries)
	_refresh_backpack()

func _view_name() -> String:
	return "inventory_view"

func _refresh_backpack() -> void:
	_dispatch_contract_command("CT-IN-IV-03", {
		"item_id": "",  # 空查询获取背包快照
		"target_slot": "",
		"target_container": "backpack"
	})

func _on_backpack_updated(event: Dictionary) -> void:
	_refresh_contract_field("CT-IN-IV-01", event.get("snapshot", {}))

func _on_item_dropped(drop_event: Dictionary) -> void:
	var item_id := drop_event.get("item_id", "")
	var target_slot := drop_event.get("target_slot", "")
	var target_container := drop_event.get("target_container", "backpack")
	var result := _dispatch_contract_command("CT-IN-IV-03", {
		"item_id": item_id,
		"target_slot": target_slot,
		"target_container": target_container
	})
	if not result.get("accepted"):
		_show_toast(_translate_error(result.get("error_code", "")))

func _on_appraise_btn_pressed() -> void:
	var selected_item_id := _item_grid.get_selected_item_id()
	var result := _dispatch_contract_command("CT-IN-IV-04", {"item_id": selected_item_id})
	if result.get("accepted"):
		_show_appraisal_result(result.get("response", {}))
```
