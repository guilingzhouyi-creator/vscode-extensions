class TestContractErrorFactory:
    """契约错误工厂（5 类错误路径覆盖）"""

    def test_err_missing_contract(self, contract_index, event_bus):
        """E-01: 引用不存在的契约"""
        with pytest.raises(ValueError, match="contract not found"):
            ContractParser.dispatch_ui_command("foo", {}, "CT-XX-YY-99", contract_index, event_bus)

    def test_err_invalid_params_schema(self, contract_index, event_bus):
        """E-02: 参数缺失必需字段"""
        with pytest.raises(ValueError, match="missing field"):
            ContractParser.dispatch_ui_command("advance_fsm", {"event": "APPROVED"}, "CT-CC-AE-03", contract_index, event_bus)

    def test_err_deprecated_contract(self, contract_index, event_bus):
        """E-03: 引用已废弃契约"""
        deprecated = MagicMock()
        deprecated.deprecated = True
        deprecated.replacement_contract_id = "CT-CC-AE-02"
        with pytest.raises(ValueError, match="deprecated"):
            ContractParser.dispatch_ui_command("foo", {}, "CT-OLD-01", contract_index, event_bus, override_entry=deprecated)

    def test_err_infra_exempt_only(self, contract_index, event_bus):
        """E-04: INFRA 豁免域不得走 UI 契约路径"""
        with pytest.raises(ValueError, match="infra exempt"):
            ContractParser.dispatch_ui_command("foo", {}, "CT-GW-WM-01", contract_index, event_bus, force_infra_check=True)

    def test_err_field_mapping_mismatch(self, contract_index):
        """E-05: DTO 字段与 field_mapping 不匹配"""
        dto = {"unknown_field": 42}
        with pytest.raises(ValueError, match="field mapping mismatch"):
            ContractParser.resolve_dto_mapping(dto, "CT-IN-IV-01", contract_index)
