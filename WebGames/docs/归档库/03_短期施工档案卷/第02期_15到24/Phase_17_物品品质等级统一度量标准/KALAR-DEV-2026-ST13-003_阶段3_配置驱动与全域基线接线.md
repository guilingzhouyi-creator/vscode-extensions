```gdscript
# 模块路径: res://backend/infrastructure/game_config.gd（_required_tables 登记）
# 新增必需表条目（audit_config.py 自动比对 config/ 目录，缺表即报错）：
#   "domains.quality_tiers",

# 模块路径: res://backend/domains/item_namespace_registry/quality_tier_registry.gd
class_name QualityTierRegistry extends RefCounted:
	var _config_table: String = "domains.quality_tiers"

	func reload_configuration() -> void:
		# 严格类型化取值器 + 安全兜底（禁 get_value）
		var tiers_cfg: Dictionary = GameConfig.get_dict(_config_table, "tiers", {})
		var intervals_cfg: Dictionary = GameConfig.get_dict(_config_table, "mythic_intervals", {})
		# 端点连续语义（INV-4）：真神器强度 > 准神器强度，配置缺失/倒置 → 校验失败
		var quasi: float = GameConfig.get_float(_config_table, "mythic_intervals/quasi_artifact/strength_weight", 4.0)
		var true_w: float = GameConfig.get_float(_config_table, "mythic_intervals/true_artifact/strength_weight", 4.8)
		_ready = (tiers_cfg.size() == 6) and (true_w > quasi) and _validate_monotonic_levels(tiers_cfg)
```

**全域接线矩阵（只读引用 + 扩展钩子）**：

| 接线点 | 动作 | 规范约束落地 |
| :--- | :--- | :--- |
| 物品注册库（Phase 11 事实源） | 物品声明新增 `quality_tier` / `mythic_interval` 字段（可选尾部，既有条目默认普通档） | 新物品必须可解析；未登记品质拒入账 |
| gacha 稀有度 | 掉落权重改经 `rarity_weight` 基线查询 | 稀有度设计服从等级基线 |
| 锻造强化/重铸 | `strength_weight` 作为收益上限基准 | 防数值膨胀越级，特殊物品仅既有标准内扩展 |
| 前端 Theme 品阶色（Phase 15 S3 已建 1~6 阶色） | 颜色令牌与后端 `color_token` 同源对齐 | 品阶色与等级一一对应，杜绝双份定义漂移 |
| 物品统计（item_statistics） | 递归树归档维度扩展 `tier/interval` | 效果评估按统一基线分层 |
