## 一、 核心业务细类与算法实现 (Business Implementation & Solver)

```gdscript
# 模块路径: res://backend/domains/item_namespace_registry/quality_tier_registry.gd
class_name QualityTierRegistry extends RefCounted:
	# 品质登记注册表：加载配置表、启动校验四项不变量、提供基线查询
	var _tiers: Dictionary = {}            # tier(int) -> {level, name_key, strength_weight, rarity_weight, color_token}
	var _mythic_intervals: Dictionary = {} # interval(int) -> {strength_weight, ...}
	var _ready: bool = false

	func reload_configuration() -> void:
		# 严格使用 GameConfig 类型化取值器（阶段 3 细化表路径），禁止 get_value
		var table: Dictionary = GameConfig.get_dict("domains.quality_tiers", "tiers", {})
		_tiers.clear()
		for tier_key in table:
			var entry: Dictionary = table[tier_key]
			_tiers[int(entry.get("tier", 0))] = entry
		_mythic_intervals = GameConfig.get_dict("domains.quality_tiers", "mythic_intervals", {})
		_ready = _validate_invariants()

	# INV-1 单调等级序：tier 与等级值一一映射且严格递增（禁止跨级倒置）
	func _validate_invariants() -> bool:
		var levels: Array[int] = []
		for tier in _tiers:
			levels.append(int(_tiers[tier].get("level", 0)))
		levels.sort()
		for i in range(levels.size() - 1):
			if levels[i] + 1 != levels[i + 1]:
				return false  # 等级序断裂/倒置
		return true

	func get_tier_baseline(tier: int) -> Dictionary:
		return _tiers.get(tier, {})

	func get_mythic_interval_baseline(interval: int) -> Dictionary:
		return _mythic_intervals.get(interval, {})

# 模块路径: res://backend/domains/item_namespace_registry/item_quality_resolver.gd
class_name ItemQualityResolver extends RefCounted:
	# 物品品质解析器：canonical_id（或物品声明）→ 唯一主档 + 神话区间 + 基线快照
	var _registry: QualityTierRegistry

	func resolve(canonical_id: String, declared_tier: int, declared_interval: int) -> Dictionary:
		# INV-3 区间端点从属：非神话级不得携带区间端点
		if declared_interval != ItemQualitySnapshot.MythicInterval.NONE \
				and declared_tier != ItemQualitySnapshot.QualityTier.MYTHIC:
			return {"success": false, "code": "INTERVAL_OUTSIDE_MYTHIC", "canonical_id": canonical_id}
		var baseline: Dictionary = _registry.get_tier_baseline(declared_tier)
		if baseline.is_empty():
			return {"success": false, "code": "UNREGISTERED_TIER", "canonical_id": canonical_id}
		# 解析成功：主档 + 区间 + 数值基线（强度/稀有度权重）
		return {
			"success": true,
			"canonical_id": canonical_id,
			"tier": declared_tier,
			"mythic_interval": declared_interval,
			"strength_weight": baseline.get("strength_weight", 1.0),
			"rarity_weight": baseline.get("rarity_weight", 1.0),
		}
```

**与现有体系衔接（只读引用，不破坏既有行为）**：

| 消费方 | 接入点 | 基线用途 |
| :--- | :--- | :--- |
| gacha 抽卡 | 稀有度权重 | 掉落稀有度基线 `rarity_weight` 统一收敛到品质等级（替换零散权重表） |
| 锻造/强化 | 成功率与品阶 | `strength_weight` 作为强化收益上限基准，防数值膨胀越级 |
| 掉落表 / GM 发放 | 发放前校验 | 新物品声明必须可解析（唯一映射），未登记拒发（对齐 Phase 11 注册库事实源） |
| 物品统计 | 归档维度 | `tier/interval` 作为递归树新增统计维度（可与 item_statistics 扩展） |

**扩展边界（规范约束落地）**：新增物品类型/新域引用基线必须经 `ItemQualityResolver` 查询；特殊物品仅允许在主品质等级既有标准内（含神话区间端点区间）扩展，禁止新增第七档或重定义区间语义。
