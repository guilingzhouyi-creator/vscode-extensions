## 一、 数据结构设计与代码契约 (Data Structures & Code Contracts)

### 1. 静态配置默认值缓存与版本感知
在 `item_entity.gd` 中：

```gdscript
# 模块路径: res://backend/domains/inventory/item_entity.gd
static var _defaults_cached: bool = false
static var _cached_config_version: int = -1
static var _cached_custom_name: String = "未命名物品"
static var _cached_category: String = CATEGORY_QUEST_MISC
static var _cached_mass_kg: float = 1.0
static var _cached_volume_slots: int = 1
static var _cached_durability: float = 100.0
static var _cached_hardness: float = 10.0
static var _cached_edge_sharpness: float = 1.0
static var _cached_reach_meters: float = 1.0
static var _cached_effective_armor: float = 0.0
static var _cached_total_sockets: int = 2
static var _cached_market_price: int = 10

static func _ensure_defaults() -> void:
	var cur_ver := GameConfig.config_reload_version()
	if _defaults_cached and _cached_config_version == cur_ver:
		return
	_cached_custom_name = GameConfig.get_string("domains.inventory", "item_defaults/custom_name", "未命名物品")
	_cached_category = GameConfig.get_string("domains.inventory", "item_defaults/category", CATEGORY_QUEST_MISC)
	_cached_mass_kg = GameConfig.get_float("domains.inventory", "item_defaults/mass_kg", 1.0)
	_cached_volume_slots = GameConfig.get_int("domains.inventory", "item_defaults/volume_slots", 1)
	_cached_durability = GameConfig.get_float("domains.inventory", "item_defaults/durability", 100.0)
	_cached_hardness = GameConfig.get_float("domains.inventory", "item_defaults/material_hardness", 10.0)
	_cached_edge_sharpness = GameConfig.get_float("domains.inventory", "item_defaults/combat_metrics/edge_sharpness", 1.0)
	_cached_reach_meters = GameConfig.get_float("domains.inventory", "item_defaults/combat_metrics/reach_meters", 1.0)
	_cached_effective_armor = GameConfig.get_float("domains.inventory", "item_defaults/combat_metrics/effective_armor", 0.0)
	_cached_total_sockets = GameConfig.get_int("domains.inventory", "item_defaults/total_sockets", 2)
	_cached_market_price = GameConfig.get_int("domains.inventory", "item_defaults/market_base_price", 10)
	_cached_config_version = cur_ver
	_defaults_cached = true
```

### 2. 构造函数初始化与 reset_state 契约
```gdscript
func _init() -> void:
	reset_state()

func reset_state() -> void:
	_ensure_defaults()
	item_id = ""
	item_uid = ""
	template_id = ""
	custom_name = _cached_custom_name
	category = _cached_category
	mass_kg = _cached_mass_kg
	volume_slots = _cached_volume_slots
	durability_current = _cached_durability
	durability_max = _cached_durability
	material_hardness = _cached_hardness
	combat_metrics = {
		"edge_sharpness": _cached_edge_sharpness,
		"reach_meters": _cached_reach_meters,
		"effective_armor": _cached_effective_armor,
		"elemental_resistances": {}
	}
	affix_sockets = {
		"total_sockets": _cached_total_sockets,
		"imprinted_runes": [],
		"dynamic_affixes": []
	}
	market_base_price = _cached_market_price
	quality_snapshot = {}
	container_state = "UNOWNED"
	owner_account_id = ""
	attribute_mounts = []
```
