## 一、 数据结构设计与代码契约 (Data Structures & Code Contracts)

### 1. AuthService 过期 Token 淘汰与写回切除
在 `auth_service.gd` 中，彻底消除过期分支的反向写回：

```gdscript
# 模块路径: res://backend/domains/account/auth_service.gd
static func validate_token(token: String, account_id: String = "", device_fingerprint: String = "", now_utc: int = -1) -> Dictionary:
	if token.is_empty():
		return {"success": false, "error_code": "TOKEN_UNKNOWN"}
	var token_hash := token.sha256_text()
	var session: Dictionary = _sessions.get(token_hash, {})
	if session.is_empty():
		return {"success": false, "error_code": "TOKEN_UNKNOWN"}
	if bool(session.get("revoked", false)):
		return {"success": false, "error_code": "TOKEN_REVOKED"}
	var now := now_utc if now_utc >= 0 else int(Time.get_unix_time_from_system())
	_prune_sessions(now)
	if now < int(session.get("issued_at", 0)):
		return {"success": false, "error_code": "TOKEN_FUTURE"}
	if now > int(session.get("expires_at", 0)):
		# 彻底从 _sessions 擦除，杜绝写回复活
		_sessions.erase(token_hash)
		return {"success": false, "error_code": "TOKEN_EXPIRED"}
	# ... 正常校验逻辑 ...
```

### 2. 鉴权 Hook 对称注销 API
补齐注销入口：

```gdscript
# 模块路径: res://backend/domains/account/auth_service.gd
static func unregister_hook(stage_key: String, callback: Callable) -> bool:
	if stage_key.is_empty() or not _auth_hooks.has(stage_key):
		return false
	var hooks_list: Array = _auth_hooks[stage_key]
	var idx := hooks_list.find(callback)
	if idx >= 0:
		hooks_list.remove_at(idx)
		if hooks_list.is_empty():
			_auth_hooks.erase(stage_key)
		return true
	return false
```

### 3. VersionedRuntimeContext 对称释放补齐
在 `version_runtime_context.gd` 中：

```gdscript
# 模块路径: res://backend/domains/version_governance/version_runtime_context.gd
func dispose() -> void:
	if _disposed:
		return
	_disposed = true
	config_snapshot.clear()
	activation_token = ""
```
