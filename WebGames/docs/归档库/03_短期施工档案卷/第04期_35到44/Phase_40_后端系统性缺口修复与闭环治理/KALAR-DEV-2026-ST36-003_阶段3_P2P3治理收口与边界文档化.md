class_name CommissionFSM

    commission: CommissionAggregate,

    if _completed_transactions.has(tx_id):

## 三、GAP-05 修复实现设计

**新增文件**：`backend/domains/ground_loot/ground_loot_event_listener.gd`

```gdscript
# ==============================================================================
# 领域：怪物击杀事件→地面掉落声明监听器 (Ground Loot Event Listener)
# 文件路径: res://backend/domains/ground_loot/ground_loot_event_listener.gd
# 职责: 监听 EventBus 的 monster.killed 事件，将 drop_declarations 转化为
#       GroundDroppedItemAggregate 实例（由战斗结算方触发，解耦掉落声明）
# ==============================================================================
class_name GroundLootEventListener extends RefCounted

var _catalog: ItemRegistryCatalog = null
var _active_drops: Array[GroundDroppedItemAggregate] = []

func _init(catalog: ItemRegistryCatalog) -> void:
    _catalog = catalog
    EventBus.get_instance().connect("domain_event_emitted", _on_domain_event)

func _on_domain_event(event_name: String, payload: Dictionary) -> void:
    if event_name != "monster.killed":
        return
    var declarations: Array = payload.get("drop_declarations", [])
    for decl in declarations:
        var canonical_id := str(decl.get("canonical_id", ""))
        if canonical_id.is_empty():
            continue
        var result := GroundDroppedItemAggregate.create_from_declaration(
            canonical_id, _catalog, str(decl.get("display_name", ""))
        )
        if not result.get("success", false):
            continue
        var drop := GroundDroppedItemAggregate.new(
            UniqueIdGenerator.next_id("DRP_"),
            result["item_entity"],
            Vector2(float(decl.get("x", 0.0)), float(decl.get("y", 0.0))),
            str(decl.get("killer_id", "")),
            int(decl.get("decay_cat", GroundDroppedItemAggregate.LootDecayCategory.STANDARD_MINERAL_EQUIP))
        )
        _active_drops.append(drop)

## 当前所有活跃掉落物（供地面拾取求解器消费）
func get_active_drops() -> Array[GroundDroppedItemAggregate]:
    return _active_drops

## 清理已衰变掉落物
func cleanup_decayed() -> void:
    _active_drops = _active_drops.filter(func(d): return not d.is_decayed)
```

**装配接入**：`GameBootstrap.assemble()` 调用后创建 `GroundLootEventListener` 实例（注入 catalog），与 `world_clock_master` 的 tick 联动调用 `cleanup_decayed()`。

**战斗结算方约定**（`monster.killed` payload 格式）：

```gdscript
EventBus.get_instance().emit_domain_event("monster.killed", {
    "args": [monster.monster_id],
    "summary": { "monster_id": monster.monster_id, "killer_id": killer_id },
    "drop_declarations": [
        {
            "canonical_id": "KALAR:EQUIPMENT:SWORD:IRON_SWORD",
            "display_name": "铁剑",
            "x": drop_pos.x,
            "y": drop_pos.y,
            "killer_id": killer_id,
            "decay_cat": GroundDroppedItemAggregate.LootDecayCategory.STANDARD_MINERAL_EQUIP
        }
    ]
})
```
