## 一、 审计引擎核心实现 (Header Audit Engine)

### 1.1 新增 `scripts/py/audit_script_comment.py`

```python
#!/usr/bin/env python3
# =============================================================================
# audit_script_comment.py — 脚本库三语言头注释规范审计（Phase 75 门禁）
# -----------------------------------------------------------------------------
# 职能域：audit
# 触发方：audit_runner（task_id="script_comment"）/ 本地 CLI 直跑
# 文件路径: scripts/py/audit_script_comment.py
# 职责: 对 scripts/sh/*.sh、scripts/ps1/*.ps1、scripts/py/*.py 逐文件解析头注释，
#       校验阶段1 契约：MUST 字段（文件路径/职责）、SHOULD 字段（职能域/触发方/
#       用法/依赖/退出码）、78 字符分隔线、未决标记禁令（TODO/FIXME/XXX/HACK）。
# 依赖：python3（stdlib：re/pathlib）；audit_common（ensure_utf8_stdout/resolve_repo_root）
# 退出码：0=全部合规；1=存在违规（阻断）；2=用法错误
# =============================================================================
import re
import sys
from pathlib import Path

from audit_common import ensure_utf8_stdout, resolve_repo_root

ensure_utf8_stdout()

ROOT = resolve_repo_root()
TARGET_DIRS = {".sh": ROOT / "scripts" / "sh",
               ".ps1": ROOT / "scripts" / "ps1",
               ".py": ROOT / "scripts" / "py"}

HEADER_MUST = ("文件路径:", "职责")
HEADER_SHOULD = ("职能域", "触发方", "用法", "依赖", "退出码")
PENDING_RE = re.compile(r"\b(TODO|FIXME|XXX|HACK)\b")

def audit_file(path: Path) -> list[dict]:
    """解析单文件头注释（前 30 行），返回违规清单。"""
    violations: list[dict] = []
    head = "\n".join(path.read_text(encoding="utf-8", errors="ignore").splitlines()[:30])
    for marker in HEADER_MUST:
        if marker not in head:
            violations.append({"file": str(path), "rule": "HEADER-MISSING-MUST", "detail": f"缺少必填字段「{marker}」"})
    for marker in HEADER_SHOULD:
        if marker not in head:
            violations.append({"file": str(path), "rule": "HEADER-MISSING-SHOULD", "detail": f"缺少建议字段「{marker}」"})
    if path.suffix != ".py":  # py 已有 shebang+分隔线，等宽线仅校验 sh/ps1 头块
        lines = path.read_text(encoding="utf-8", errors="ignore").splitlines()
        for i, line in enumerate(lines[:30]):
            if re.match(r"^#\s*=+$", line) and len(line.strip("#").strip()) != 78:
                violations.append({"file": str(path), "rule": "HEADER-SEPARATOR-WIDTH", "detail": f"L{i+1} 分隔线非 78 字符"})
    for i, line in enumerate(head.splitlines()):
        if PENDING_RE.search(line) and not line.strip().startswith("#"):
            violations.append({"file": str(path), "rule": "COMMENT-PENDING-MARKER", "detail": f"L{i+1} 含未决标记"})
    return violations

def main() -> int:
    all_violations: list[dict] = []
    for suffix, directory in TARGET_DIRS.items():
        for path in sorted(directory.glob(f"*{suffix}")):
            all_violations.extend(audit_file(path))
    for v in all_violations:
        print(f"{v['file']} [{v['rule']}] {v['detail']}")
    print(f"【audit-script-comment】共扫描 {sum(1 for _ in _iter_targets())} 文件，违规 {len(all_violations)} 处")
    return 1 if all_violations else 0
```
