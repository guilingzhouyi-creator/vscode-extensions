## 一、 配置驱动与工程化改造 (Generalization)

```json
{
  "reward_items": [
    { "canonical_id": "KALAR:EQUIP:WEAPON_BLADE:MITHRIL_LONGSWORD", "count": 1 }
  ]
}
```

```gdscript
# 模块路径: res://backend/domains/quest_causality/quest_reward_dispatch.gd
class_name QuestRewardDispatch extends RefCounted:
    static func dispatch_item_rewards(reward_items: Array, catalog: ItemRegistryCatalog, inventory: WearableInventoryAggregate, library: AccountItemLibraryAggregate = null) -> Dictionary:
        var granted := 0
        for entry in reward_items:
            var canonical_id := str(entry.get("canonical_id", ""))
            var proto = catalog.get_prototype(canonical_id)
            if proto == null:
                return {"success": false, "error_code": "ITEM_NOT_REGISTERED", "canonical_id": canonical_id}
            var count := int(entry.get("count", 1))
            for i in range(count):
                var item := ItemInstanceFactory.build_instance(proto, proto.english_name, "QST_")
                if not inventory.add_item(item):
                    return {"success": false, "error_code": "INVENTORY_FULL", "granted": granted}
                granted += 1
                if library != null:
                    ItemStatisticsSolver.record_item_event(library, ItemStatisticsSolver.EVENT_ITEM_GRANTED, canonical_id, 1, catalog, {"uid": item.item_uid})
        return {"success": true, "granted": granted}
```

* 新增 UID 前缀 `QST_`（任务奖励）登记入 `item_namespace_registry.json:uid.prefixes`
* `quest.json:reward_items` 键级入 `audit_item_keys.py` 审计范围
