## 一、 核心业务逻辑实现 (Core Business Logic)

```gdscript
# 模块路径: res://backend/domains/account/character_death_solver.gd
class_name CharacterDeathSolver extends RefCounted:
    # 角色本体需清理的精确白名单（禁止删全档）
    const CHARACTER_OWNED_KEYS: Array[String] = [
        "attributes", "growth_state", "inventory", "equipment",
        "personal_resources", "personal_quest_progress",
        "relationship_state", "character_exclusive_progress"
    ]
    static func handle_character_death(account: AccountProfileAggregate, slot_id: String, world: WorldInstance) -> Dictionary:
        if account == null or slot_id.is_empty():
            return {"success": false, "error_code": "INVALID_SLOT"}
        var dto: SaveSlotSummaryDTO = account.get_summary_by_slot(slot_id)
        if dto == null:
            return {"success": false, "error_code": "SLOT_EMPTY"}
        # 1. 精确清理角色本体子集（白名单）
        var cleared: Array[String] = []
        for k in CHARACTER_OWNED_KEYS:
            if dto.has_character_payload(k):
                dto.clear_character_payload(k)
                cleared.append(k)
        dto.is_fallen = true
        # 2. 世界零回退：时间与状态继续（仅追加历史影响片段，不重置）
        if world != null:
            world.advance_world_time(1) # 文字版：WorldClockMaster 推进，不重置
            world.append_history_fragment({"source": dto.character_name, "slot_id": slot_id, "event": "character_death"})
        # 3. 新角色零继承：清空后槽位可重新绑定全新 CharacterProfile
        return {"success": true, "cleared_keys": cleared, "world_time": world.world_time if world else {}}
```

| 死亡操作 | 角色数据 | 世界数据 |
| :--- | :--- | :--- |
| 清理 | 属性/背包/装备/个人资源/任务/关系/专属进度（白名单） | 否 |
| 保留 | 账号容器/权限/其他槽位角色 | 时间/ NPC/ 环境/ 事件/ 资源/ 社会/历史变化 |
