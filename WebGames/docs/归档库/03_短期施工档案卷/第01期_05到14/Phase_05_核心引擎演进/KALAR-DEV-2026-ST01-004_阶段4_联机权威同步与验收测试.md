# 模块路径: res://tests/unit/test_phase05_core_engine.gd
class_name TestPhase05CoreEngine extends RefCounted:

    static func run_all_tests() -> Dictionary:
        var results := []
        results.append(test_game_loop_stack_transitions())
        results.append(test_tilemap_viewport_clamp())
        results.append(test_world_evolution_and_clock())

        var all_passed := true
        for r in results:
            if not r.get("passed", false):
                all_passed = false
                break
        return { "domain": "Phase 05: 核心引擎演进", "all_passed": all_passed, "results": results }

    static func test_game_loop_stack_transitions() -> Dictionary:
        var stack := GameLoopStateStack.new()
        stack.push_state(GameLoopStateStack.EngineState.LOGIN_AUTH)
        stack.push_state(GameLoopStateStack.EngineState.WORLD_EXPLORATION)
        var passed = (stack.current_state == GameLoopStateStack.EngineState.WORLD_EXPLORATION)
        return { "test": "TC-P05-01: 游戏全局状态机入栈与出栈流转", "passed": passed }

    static func test_tilemap_viewport_clamp() -> Dictionary:
        var bounds = Rect2(0, 0, 1000, 1000)
        var pos = TilemapViewportSolver.calculate_camera_target(Vector2(500, 500), Vector2(1500, 500), 1.0, 5.0, bounds)
        var passed = (pos.x <= 1000.0)
        return { "test": "TC-P05-02: 相机视口边界钳制与平滑跟随", "passed": passed }

    static func test_world_evolution_and_clock() -> Dictionary:
        var clock := WorldClockMaster.new()
        WorldEvolutionFSM.advance_offline_evolution(clock, 24)
        var passed = (clock.calendar_days >= 1)
        return { "test": "TC-P05-03: 离线世界自发演化与时钟推移", "passed": passed }
