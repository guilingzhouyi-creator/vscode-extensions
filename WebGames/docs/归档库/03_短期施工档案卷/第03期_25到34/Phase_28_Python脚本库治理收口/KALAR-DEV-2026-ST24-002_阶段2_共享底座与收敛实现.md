### 1.1 audit_common.py（共享底座，新文件）

```python
# audit_common.py — 脚本库共享底座（Phase 28 治理收口）
from dataclasses import dataclass
from pathlib import Path
import json, re, sys

## 唯一仓库根解析（消除 parents[2]/parent.parent.parent/parent 三变体漂移）
def resolve_repo_root() -> Path:
    return Path(__file__).resolve().parents[2]

## utf-8 输出样板（Windows 控制台安全）
def ensure_utf8_stdout() -> None:
    if sys.platform == "win32" and hasattr(sys.stdout, "reconfigure"):
        try:
            sys.stdout.reconfigure(encoding="utf-8", errors="replace")
            sys.stderr.reconfigure(encoding="utf-8", errors="replace")
        except Exception:
            pass

## 容错 JSON 读取
def load_json(path: Path) -> dict:
    return json.loads(path.read_text(encoding="utf-8", errors="ignore"))

@dataclass
class Finding:
    rule: str
    file: str
    line: int
    message: str
    suggestion: str = ""
    severity: str = "error"

## 统一结论打印（「审查结论」分级一致性）
def print_conclusion(violations: list, hints: list, name: str) -> int:
    print(f"违规总数: {len(violations)} / 提示: {len(hints)}")
    for v in violations:
        print(f"  ✗ {v}")
    for h in hints:
        print(f"  · {h}")
    print("-" * 80)
    if violations:
        print(f"【审查结论】未通过（{name} 违规，已阻断！）")
        return 1
    print(f"【审查结论】通过（{name} 合规）")
    return 0

## 三检查项共享助手（收敛跨脚本重写）
LEGACY_PLACEHOLDER_RE = re.compile(r"%[sd]|%\.\d*f|\{\}")
CJK_RE = re.compile(r"[\u4e00-\u9fff]")
def scan_legacy_placeholders(text: str) -> list:
    return LEGACY_PLACEHOLDER_RE.findall(text)
def contains_cjk(text: str) -> bool:
    return bool(CJK_RE.search(text))
def key_contract_re() -> re.Pattern:
    return re.compile(r"[a-z][a-z0-9_]*(\.[a-z][a-z0-9_]*)+")   # 三段式点分小写键契约
```
