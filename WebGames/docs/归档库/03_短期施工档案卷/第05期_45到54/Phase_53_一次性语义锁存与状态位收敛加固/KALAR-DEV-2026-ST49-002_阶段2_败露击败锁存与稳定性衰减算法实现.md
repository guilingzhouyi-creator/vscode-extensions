### 1. M6 修复：败露锁存 + 合并归零 + 记账路由守卫

```gdscript
# 模块路径: res://backend/domains/identity_disguise/identity_exposure_pipeline.gd
static func trigger_unmask_exposure(identity, compromised_alias_id: String) -> Dictionary:
    if not identity.alias_slots.has(compromised_alias_id):
        return {"success": false, "error_code": "ALIAS_NOT_FOUND"}

    var profile := identity.alias_slots[compromised_alias_id]
    # Inv-ON-1 新增：锁存入口——已败露直接幂等返回（0 搬运，防 double count）
    if profile.is_compromised:
        return {
            "success": true,
            "error_code": "ALREADY_COMPROMISED",
            "character_uuid": identity.character_uuid,
            "transferred_infamy_bounty": 0,
            "total_true_infamy": identity.base_true_infamy_bounty,
        }

    profile.is_compromised = true

    # Inv-ON-2 改造：先搬运、后归零（残值清零，杜绝二次搬运现场）
    var merged_infamy: int = profile.accumulated_infamy_bounty
    identity.base_true_infamy_bounty += merged_infamy
    profile.accumulated_infamy_bounty = 0

    # 若当前正穿戴该假名，自动脱下恢复真身（原语义保留）
    if identity.active_alias_id == compromised_alias_id:
        identity.active_alias_id = ""

    # reputation 侧同样搬运归零（对齐：声望也是 alias 累积量）
    identity.base_true_reputation += profile.accumulated_reputation
    profile.accumulated_reputation = 0

    return {
        "success": true,
        "error_code": "OK",
        "character_uuid": identity.character_uuid,
        "true_canonical_name": identity.true_canonical_name,
        "compromised_alias": profile.display_name,
        "transferred_infamy_bounty": merged_infamy,
        "total_true_infamy": identity.base_true_infamy_bounty,
    }
```

```gdscript
# record_deed 路由守卫（同文件）：目标 alias 已败露 → 直接记真身（锁存后通道不再向 alias 累积）
static func record_deed(identity, reputation_delta: int, infamy_bounty_delta: int) -> void:
    var aid := identity.active_alias_id
    if aid != "" and identity.alias_slots.has(aid):
        var profile := identity.alias_slots[aid]
        if profile.is_compromised:                       # 新增守卫
            identity.base_true_reputation += reputation_delta
            identity.base_true_infamy_bounty += infamy_bounty_delta
            return
        profile.accumulated_reputation += reputation_delta
        profile.accumulated_infamy_bounty += infamy_bounty_delta
    else:
        identity.base_true_reputation += reputation_delta
        identity.base_true_infamy_bounty += infamy_bounty_delta
```

### 2. M7 修复：稳定性单调衰减公式替换

```gdscript
# 模块路径: res://backend/domains/sovereignty_realm/nation_sovereignty_fsm.gd（61/69 行替换）
# 新：单调不升（Inv-ON-3）——下限只约束从上方下跌的止点，绝不抬升现值
# 成功分支：nation.stability_percent = _apply_stability_drop(nation.stability_percent, success_drop, success_floor)
# 失败分支：nation.stability_percent = _apply_stability_drop(nation.stability_percent, failure_drop, failure_floor)

static func _apply_stability_drop(cur: float, drop: float, floor: float) -> float:
    if cur <= floor:
        return cur
    return maxf(floor, cur - drop)
```

**行为对照（修复前 → 修复后）**：stability=20、失败 drop=15、floor=40：旧 `max(40, 20-15)=40`（抬升）→ 新 `20`（不变，等下次从上方衰减）；stability=60：旧 `45` → 新 `45`（无变化）。成功路径在 cur>floor 时数值不变，仅在「已低于下限」样本上收敛为不抬升。
