## 一、 阶段目标与算法定位 (对齐 P10.2 / 10.5 / 10.7)

本阶段实现文字版角色序章的两大计算求解器与装配流水线：
1. **`ProloguePlaceholderResolver`（占位符解析引擎）**：高可靠参数替换器，全面消除代码内的业务魔数与固定字符串；
2. **`StarterLoadoutDispatcher`（新手装备与资源发放流水线）**：配置驱动的初始化发放器，实现防重复发放与幂等保护；
3. **`PrologueExecutionKernel`（序章执行内核）**：负责编排初始化事件与状态流转，产出纯结构化事件（`CharacterCreated`, `EquipmentGranted`, `LocationAssigned`, `DialogueReady`）。

## 二、 占位符解析引擎算法实现 (`ProloguePlaceholderResolver`)

解析器严禁在代码中写死任何具体角色名、出生点或道具名，全部通过上下文提取与正则安全替换：

```gdscript
class_name ProloguePlaceholderResolver
extends RefCounted

static var _regex: RegEx = RegEx.create_from_string(r"\{([A-Z0-9_]+)\}")

## 从角色序章上下文动态收集运行时占位符字典
static func extract_placeholders(ctx: CharacterPrologueContext) -> Dictionary:
	var dict: Dictionary = {}
	if ctx == null:
		return dict

	dict["PLAYER_NAME"] = ctx.character_name
	dict["CHARACTER_NAME"] = ctx.character_name
	dict["CHARACTER_ID"] = ctx.character_id
	dict["RACE_ID"] = ctx.race_id
	dict["GENDER"] = ctx.gender

	# 查配置映射出生地与新手装备的国际化展示名
	var loc_table := GameConfig.get_dict("domains.world", "node_defaults", {})
	var loc_name := GameConfig.get_string("narratives.prologue", "locations/" + ctx.assigned_start_location_id, ctx.assigned_start_location_id)
	dict["START_LOCATION"] = loc_name

	var kit_cfg := GameConfig.get_dict("domains.starter_loadout", "starter_kits/" + ctx.assigned_starter_kit_id, {})
	var weapon_canonical: String = kit_cfg.get("display_weapon", "KALAR:EQUIP:WEAPON_BLADE:IRON_SWORD")
	dict["INITIAL_EQUIPMENT"] = GameConfig.get_string("narratives.prologue", "items/" + weapon_canonical, weapon_canonical)

	var npc_target: String = GameConfig.get_string("narratives.prologue", "npcs/default_guide", "FORTRESS_GUARD")
	dict["NPC_TARGET"] = GameConfig.get_string("narratives.prologue", "npcs/" + npc_target, npc_target)

	dict["EVENT_RESULT"] = "SUCCESS"
	return dict

## 将模板中的 {KEY} 替换为运行时实值，并检测未匹配占位符
static func resolve_template(template_str: String, placeholders: Dictionary) -> Dictionary:
	if template_str.is_empty():
		return { "success": true, "text": "", "missing_placeholders": [] }

	var missing: Array[String] = []
	var result_text := template_str

	var matches := _regex.search_all(template_str)
	for m in matches:
		var key := m.get_string(1)
		if placeholders.has(key):
			var val := str(placeholders[key])
			result_text = result_text.replace("{%s}" % key, val)
		else:
			if not missing.has(key):
				missing.append(key)
			# 遇缺失占位符注入安全降级标记，避免抛出异常
			result_text = result_text.replace("{%s}" % key, "[UNKNOWN:%s]" % key)

	return {
		"success": missing.is_empty(),
		"text": result_text,
		"missing_placeholders": missing
	}
```

## 三、 配置驱动新手装备发放算法 (`StarterLoadoutDispatcher`)

```gdscript
class_name StarterLoadoutDispatcher
extends RefCounted

## 为角色序章发放新手初始物资 (幂等防重复保护)
static func dispatch_starter_kit(ctx: CharacterPrologueContext) -> Dictionary:
	if ctx == null or ctx.character_id.is_empty():
		return { "success": false, "error_code": "INVALID_CONTEXT" }

	# 1. 幂等防护：检查是否已发放过新手装备
	if bool(ctx.runtime_placeholders.get("STARTER_KIT_GRANTED", false)):
		return {
			"success": true,
			"code": "ALREADY_GRANTED",
			"message": "该角色已领取过新手装备，触发幂等跳过"
		}

	# 2. 读取配置决定套件 ID (按种族映射，若无则取全局默认)
	var kit_id := ctx.assigned_starter_kit_id
	if kit_id.is_empty():
		kit_id = GameConfig.get_string("domains.starter_loadout", "race_kit_map/" + ctx.race_id, "KIT_HUMAN_DEFAULT")
		ctx.assigned_starter_kit_id = kit_id

	var kit_cfg := GameConfig.get_dict("domains.starter_loadout", "starter_kits/" + kit_id, {})
	if kit_cfg.is_empty():
		# 安全兜底配置
		kit_cfg = {
			"initial_gold": GameConfig.get_int("domains.starter_loadout", "defaults/initial_gold", 50),
			"initial_items": []
		}

	var gold: int = int(kit_cfg.get("initial_gold", 50))
	var items: Array = kit_cfg.get("initial_items", [])

	# 3. 标记已发放
	ctx.runtime_placeholders["STARTER_KIT_GRANTED"] = true
	ctx.runtime_placeholders["STARTER_GOLD"] = gold

	return {
		"success": true,
		"code": "KIT_DISPATCHED",
		"gold": gold,
		"items": items,
		"kit_id": kit_id
	}
```

## 四、 序章执行内核 (`PrologueExecutionKernel`) 与状态解耦

执行内核严格产出解耦的结构化事件，不直接操纵 UI：

```gdscript
class_name PrologueExecutionKernel
extends RefCounted

## 启动角色序章，产出初始化结构化事件流
static func boot_character_prologue(opening: OpeningEventStreamDTO) -> Dictionary:
	var ctx := CharacterPrologueContext.new()
	ctx.account_id = opening.account_id
	ctx.character_id = opening.character_id
	ctx.character_name = opening.character_name
	ctx.race_id = GameConfig.get_string("domains.character_creation", "defaults/race", "HUMAN")
	ctx.assigned_start_location_id = opening.starting_location_id
	ctx.assigned_opening_quest_id = opening.opening_quest_line_id
	ctx.created_timestamp_utc = int(Time.get_unix_time_from_system())
	ctx.is_active = true

	# 1. 提取并缓存初始占位符
	var ph := ProloguePlaceholderResolver.extract_placeholders(ctx)
	ctx.runtime_placeholders = ph

	# 2. 发放初始装备
	var kit_res := StarterLoadoutDispatcher.dispatch_starter_kit(ctx)

	# 3. 结构化事件列表 (事件与状态彻底解耦)
	var structured_mutations: Array[Dictionary] = [
		{ "type": "CharacterCreated", "character_id": ctx.character_id, "name": ctx.character_name },
		{ "type": "LocationAssigned", "location_id": ctx.assigned_start_location_id },
		{ "type": "EquipmentGranted", "kit_id": kit_res.get("kit_id", ""), "gold": kit_res.get("gold", 0) },
		{ "type": "DialogueReady", "npc_target": ph.get("NPC_TARGET", "") }
	]

	# 4. 生成第一步事件数据包
	var template := GameConfig.get_string("narratives.prologue", "narrative.prologue.awakening", "【苏醒】晨曦降临，开拓者 {PLAYER_NAME} 缓缓睁开双眼……")
	var text_res := ProloguePlaceholderResolver.resolve_template(template, ctx.runtime_placeholders)

	var packet := PrologueEventPacketDTO.new()
	packet.account_id = ctx.account_id
	packet.character_id = ctx.character_id
	packet.character_name = ctx.character_name
	packet.current_stage = 1
	packet.narrative_text = text_res.get("text", "")
	packet.available_actions = ["WAKE_UP", "NEXT"]
	packet.reserved_payload = { "mutations": structured_mutations }

	return { "success": true, "context": ctx, "packet": packet }
```
