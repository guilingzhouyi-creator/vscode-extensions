### 2.1 综合先手判定求解器 (`CombatInitiativeSolver`)

```gdscript
# 模块路径: res://backend/domains/physics_thermodynamics/combat_initiative_solver.gd
class_name CombatInitiativeSolver
extends RefCounted

static func evaluate_initiative(
	player: ParticipantInitiativeSpec,
	enemy: ParticipantInitiativeSpec,
	rng: DeterministicRNG
) -> BattleInitiativeResultDTO:
	var result := BattleInitiativeResultDTO.new()
	result.player_spec = player
	result.enemy_spec = enemy
	result.seed_used = rng.get_current_seed() if rng else 0
	result.start_utc = int(Time.get_unix_time_from_system())

	# 1. 权重由 combat.json initiative_model 读取
	var w_agi := GameConfig.get_float("domains.combat", "initiative_model/weight_agility", 1.0)
	var w_per := GameConfig.get_float("domains.combat", "initiative_model/weight_perception", 0.8)
	var w_lvl := GameConfig.get_float("domains.combat", "initiative_model/weight_level", 0.5)
	var jitter_min := GameConfig.get_int("domains.combat", "initiative_model/jitter_min", -2)
	var jitter_max := GameConfig.get_int("domains.combat", "initiative_model/jitter_max", 2)

	# 2. 双方基础综合评分
	var p_base := player.base_agility * w_agi + player.base_perception * w_per + float(player.level) * w_lvl
	var e_base := enemy.base_agility * w_agi + enemy.base_perception * w_per + float(enemy.level) * w_lvl

	# 3. 乘算修正 (装备轻灵度 + 状态增益)
	var p_mod := 1.0 + player.equip_speed_modifier + player.temporary_buff_modifier
	var e_mod := 1.0 + enemy.equip_speed_modifier + enemy.temporary_buff_modifier

	# 4. 确定性随机抖动注入
	var p_jitter := float(rng.randi_range(jitter_min, jitter_max)) if rng else 0.0
	var e_jitter := float(rng.randi_range(jitter_min, jitter_max)) if rng else 0.0

	player.calculated_score = (p_base * p_mod) + p_jitter
	enemy.calculated_score = (e_base * e_mod) + e_jitter

	# 5. 胜负与平局决胜判定 (Tie-Breaker)
	var ap_lead_bonus := GameConfig.get_int("domains.combat", "initiative_model/lead_ap_bonus", 2)
	if abs(player.calculated_score - enemy.calculated_score) > 0.001:
		if player.calculated_score > enemy.calculated_score:
			result.first_mover_id = player.participant_id
			result.tie_broken_by = "SCORE"
			player.is_first_mover = true
			player.initial_ap_bonus = ap_lead_bonus
			result.initial_ap_diff = ap_lead_bonus
		else:
			result.first_mover_id = enemy.participant_id
			result.tie_broken_by = "SCORE"
			enemy.is_first_mover = true
			enemy.initial_ap_bonus = ap_lead_bonus
			result.initial_ap_diff = -ap_lead_bonus
	else:
		# 二级决胜：纯敏捷比对
		if abs(player.base_agility - enemy.base_agility) > 0.001:
			if player.base_agility > enemy.base_agility:
				result.first_mover_id = player.participant_id
				result.tie_broken_by = "AGI_FALLBACK"
				player.is_first_mover = true
				result.initial_ap_diff = 1
			else:
				result.first_mover_id = enemy.participant_id
				result.tie_broken_by = "AGI_FALLBACK"
				enemy.is_first_mover = true
				result.initial_ap_diff = -1
		else:
			# 三级决胜：50/50 确定性随机硬币
			var coin := rng.randi_range(0, 1) if rng else 0
			if coin == 0:
				result.first_mover_id = player.participant_id
				result.tie_broken_by = "RNG_TIEBREAKER"
				player.is_first_mover = true
			else:
				result.first_mover_id = enemy.participant_id
				result.tie_broken_by = "RNG_TIEBREAKER"
				enemy.is_first_mover = true

	return result
```

### 2.2 第三时间轴推演引擎 (`TertiaryTimelineEngine`)

```gdscript
# 模块路径: res://backend/domains/physics_thermodynamics/tertiary_timeline_engine.gd
class_name TertiaryTimelineEngine
extends RefCounted

var current_timeline_ms: int = 0
var scheduled_events: Array[TimelineEventPointDTO] = []
var rng: DeterministicRNG = null
var event_counter: int = 0

func initialize(start_ms: int, in_rng: DeterministicRNG) -> void:
	current_timeline_ms = start_ms
	scheduled_events.clear()
	rng = in_rng
	event_counter = 0
	# 初始化首次事件排期
	_schedule_next_random_dispatch()
	_schedule_next_tension_pulse()

## 时间流逝驱动：推进 timeline_ms 并弹出所有到期的离散事件点
func advance_time(delta_ms: int) -> Array[TimelineEventPointDTO]:
	if delta_ms <= 0:
		return []

	current_timeline_ms += delta_ms
	var fired_events: Array[TimelineEventPointDTO] = []
	var remaining_events: Array[TimelineEventPointDTO] = []

	for evt in scheduled_events:
		if evt.trigger_time_ms <= current_timeline_ms:
			evt.is_processed = true
			fired_events.append(evt)
			# 触发后排期下一次对应事件点
			if evt.event_type == TimelineEventType.RANDOM_DISPATCH:
				_schedule_next_random_dispatch()
			elif evt.event_type == TimelineEventType.TENSION_PULSE:
				_schedule_next_tension_pulse()
		else:
			remaining_events.append(evt)

	scheduled_events = remaining_events
	return fired_events

func _schedule_next_random_dispatch() -> void:
	var min_int := GameConfig.get_int("domains.combat", "tertiary_timeline/dispatch_interval_min_ms", 1200)
	var max_int := GameConfig.get_int("domains.combat", "tertiary_timeline/dispatch_interval_max_ms", 2800)
	var delta := rng.randi_range(min_int, max_int) if rng else min_int

	event_counter += 1
	var evt := TimelineEventPointDTO.new()
	evt.event_id = "EVT_DISPATCH_%d" % event_counter
	evt.event_type = TimelineEventType.RANDOM_DISPATCH
	evt.trigger_time_ms = current_timeline_ms + delta
	scheduled_events.append(evt)

func _schedule_next_tension_pulse() -> void:
	var min_int := GameConfig.get_int("domains.combat", "tertiary_timeline/tension_interval_min_ms", 4500)
	var max_int := GameConfig.get_int("domains.combat", "tertiary_timeline/tension_interval_max_ms", 8500)
	var delta := rng.randi_range(min_int, max_int) if rng else min_int

	var tension_types: Array = GameConfig.get_array("domains.combat", "tertiary_timeline/tension_types", [
		"PRESSURE_SURGE", "ENVIRONMENT_PULSE", "FORCE_ROUND_ADVANCE"
	])
	var pick_idx := rng.randi_range(0, tension_types.size() - 1) if (rng and not tension_types.is_empty()) else 0
	var selected_type: String = tension_types[pick_idx] if not tension_types.is_empty() else "PRESSURE_SURGE"

	event_counter += 1
	var evt := TimelineEventPointDTO.new()
	evt.event_id = "EVT_TENSION_%d" % event_counter
	evt.event_type = TimelineEventType.TENSION_PULSE
	evt.trigger_time_ms = current_timeline_ms + delta
	evt.payload = { "tension_type": selected_type }
	scheduled_events.append(evt)
```

### 2.3 待发栏满载溢出与消费管理 (`ActionCardStagingBar`)

```gdscript
# 模块路径: res://backend/domains/physics_thermodynamics/action_card_staging_bar.gd
class_name ActionCardStagingBar
extends RefCounted

var owner_id: String = ""
var max_capacity: int = 5
var cards: Array[PhysicalVerbRegistry.CombatActionCardEntity] = []
var total_overflow_count: int = 0
var accumulated_overflow_energy: int = 0

func initialize(in_owner_id: String, capacity: int) -> void:
	owner_id = in_owner_id
	max_capacity = capacity
	cards.clear()
	total_overflow_count = 0
	accumulated_overflow_energy = 0

## 将抽取到的卡牌推入待发栏（满载时执行配置策略，绝不阻塞时间轴）
func enqueue_card(card: PhysicalVerbRegistry.CombatActionCardEntity) -> Dictionary:
	if card == null:
		return {"success": false, "reason": "NULL_CARD"}

	var overflow_policy := GameConfig.get_string("domains.combat", "staging_bar/overflow_policy", "DISCARD_OLDEST")
	var was_full := cards.size() >= max_capacity

	if was_full:
		total_overflow_count += 1
		match overflow_policy:
			"DISCARD_OLDEST":
				var discarded := cards.pop_front() as PhysicalVerbRegistry.CombatActionCardEntity
				cards.append(card)
				return {
					"success": true,
					"action": "OVERFLOW_DISCARD_OLDEST",
					"discarded_card_id": discarded.card_id if discarded else "",
					"inserted_card_id": card.card_id
				}
			"CONVERT_AP":
				var bonus_ap := GameConfig.get_int("domains.combat", "staging_bar/overflow_ap_bonus", 1)
				accumulated_overflow_energy += bonus_ap
				return {
					"success": true,
					"action": "OVERFLOW_CONVERT_AP",
					"bonus_ap": bonus_ap,
					"rejected_card_id": card.card_id
				}
			_:
				# 默认丢弃最新
				return {
					"success": false,
					"action": "OVERFLOW_REJECTED",
					"rejected_card_id": card.card_id
				}

	cards.append(card)
	return {"success": true, "action": "ENQUEUED", "card_id": card.card_id}

## 消耗/打出手牌
func pop_card(card_id: String) -> PhysicalVerbRegistry.CombatActionCardEntity:
	for i in range(cards.size()):
		if cards[i].card_id == card_id:
			var card = cards[i]
			cards.remove_at(i)
			return card
	return null
```

### 2.4 三轴联动与小回合生命周期编排器 (`CombatRoundCoordinator`)

```gdscript
# 模块路径: res://backend/domains/physics_thermodynamics/combat_round_coordinator.gd
class_name CombatRoundCoordinator
extends RefCounted

var timeline_engine: TertiaryTimelineEngine = null
var player_staging: ActionCardStagingBar = null
var enemy_staging: ActionCardStagingBar = null
var current_round: int = 1
var round_elapsed_ms: int = 0
var max_round_ms: int = 10000

## 步进循环：时间推进 -> 派发事件 -> 待发栏入队 -> 回合结束条件判定
func step_coordinator(
	delta_ms: int,
	player_participant: CombatPipelineFSM.CombatParticipant,
	enemy_participant: CombatPipelineFSM.CombatParticipant
) -> Dictionary:
	round_elapsed_ms += delta_ms
	var fired_events := timeline_engine.advance_time(delta_ms)
	var processed_events: Array[Dictionary] = []
	var should_end_round := false
	var end_reason := ""

	for evt in fired_events:
		if evt.event_type == TertiaryTimelineEngine.TimelineEventType.RANDOM_DISPATCH:
			# 触发一次卡牌抽取 (以玩家待发栏为例)
			var drawn_card := _draw_random_card_from_pool()
			var queue_res := player_staging.enqueue_card(drawn_card)
			processed_events.append({
				"event_id": evt.event_id,
				"type": "CARD_DISPATCHED",
				"time_ms": evt.trigger_time_ms,
				"queue_res": queue_res
			})
		elif evt.event_type == TertiaryTimelineEngine.TimelineEventType.TENSION_PULSE:
			var t_type: String = evt.payload.get("tension_type", "PRESSURE_SURGE")
			processed_events.append({
				"event_id": evt.event_id,
				"type": "TENSION_APPLIED",
				"tension_type": t_type,
				"time_ms": evt.trigger_time_ms
			})
			if t_type == "FORCE_ROUND_ADVANCE":
				should_end_round = true
				end_reason = "TENSION_FORCE_ADVANCE"

	# 小回合结束条件检查：AP耗尽且无手牌 / 超过最大动次时限 / 紧张点打断
	if not should_end_round:
		if player_participant.current_ap <= 0 and player_staging.cards.is_empty():
			should_end_round = true
			end_reason = "AP_AND_HAND_EXHAUSTED"
		elif round_elapsed_ms >= max_round_ms:
			should_end_round = true
			end_reason = "ROUND_TIME_LIMIT_REACHED"

	if should_end_round:
		_transition_to_next_round(end_reason)

	return {
		"round": current_round,
		"timeline_ms": timeline_engine.current_timeline_ms,
		"round_elapsed_ms": round_elapsed_ms,
		"events_fired": processed_events,
		"round_ended": should_end_round,
		"end_reason": end_reason
	}

func _transition_to_next_round(reason: String) -> void:
	current_round += 1
	round_elapsed_ms = 0
	# 核心铁律：timeline_engine 绝不重新初始化，继续沿用当前绝对时间轴前进！

func _draw_random_card_from_pool() -> PhysicalVerbRegistry.CombatActionCardEntity:
	# 简明确定性发牌工厂
	var card := PhysicalVerbRegistry.CombatActionCardEntity.new()
	card.card_id = "CARD_AUTO_%d" % timeline_engine.event_counter
	card.card_name = "动态发放行动卡"
	card.verb_type = "SLASH"
	card.ap_cost = -2
	return card
```
