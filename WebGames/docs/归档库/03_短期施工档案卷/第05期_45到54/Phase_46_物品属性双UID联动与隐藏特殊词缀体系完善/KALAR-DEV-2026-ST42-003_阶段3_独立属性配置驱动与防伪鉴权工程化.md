* **核心不变量约束断言**：
  1. **属性配置独立声明不变量**：所有属性定义（类型/上下界/公式/词缀评分/互斥标签）100% 由 `config/domains/item_attributes.json` 驱动，新增/微调属性零核心代码修改；
  2. **服务端权威签名防伪不变量**：物品已鉴定状态必须包含由服务端生成的哈希签名 `audit_signature`，客户端直接修改本地 `is_appraised = true` 必然因签名校验失败被安全重置为未鉴定状态；
  3. **配置热重载原子隔离不变量**：配置表热更若发生格式损坏，自动保留既有有效快照，不产生 Null 崩溃。
* **工程化重构最高指示**：严格使用 `GameConfig.get_*` 类型化取值器，杜绝 Variant `get_value`；循环内部避免重复寻路。

### 3.2 属性定义注册表 (`ItemAttributeRegistry`)

```gdscript
# 模块路径: res://backend/domains/item_attributes/item_attribute_registry.gd
class_name ItemAttributeRegistry
extends RefCounted

var _definitions: Dictionary = {}

func reload_configuration() -> void:
	_definitions.clear()
	var raw_attrs: Dictionary = GameConfig.get_dict("domains.item_attributes", "attributes", {})
	for attr_uid in raw_attrs.keys():
		var d: Dictionary = raw_attrs[attr_uid]
		d["attribute_uid"] = attr_uid
		var def := ItemAttributeDefinition.from_dto(d)
		_definitions[attr_uid] = def

func get_definition(attribute_uid: String) -> ItemAttributeDefinition:
	return _definitions.get(attribute_uid, null)

func has_definition(attribute_uid: String) -> bool:
	return _definitions.has(attribute_uid)

func get_all_definitions() -> Dictionary:
	return _definitions.duplicate()
```

### 3.3 客户端伪造鉴定防御守卫 (`ItemAttributeSecurityGuard`)

```gdscript
# 模块路径: res://backend/domains/item_attributes/item_attribute_security_guard.gd
class_name ItemAttributeSecurityGuard
extends RefCounted

## 验证挂载实例鉴定状态的真实性（防作弊）
static func verify_mount_integrity(mount: ItemAttributeMountInstance) -> bool:
	if mount == null:
		return false

	# 非特殊词缀无需校验防伪签名
	if mount.category != ItemAttributeDefinition.AttributeCategory.SPECIAL_AFFIX:
		return true

	# 处于未鉴定态，合法
	if mount.visibility == ItemAttributeMountInstance.VisibilityState.HIDDEN:
		return not mount.is_active

	# 声明为已鉴定态，必须包含有效签名
	var audit := mount.appraisal_audit
	if not bool(audit.get("is_appraised", false)):
		return false

	var ts := int(audit.get("appraised_timestamp", 0))
	var actor := str(audit.get("appraiser_actor_id", ""))
	var sig := str(audit.get("audit_signature", ""))

	var expected := ItemAttributeAppraisalSolver._generate_audit_signature(
		mount.item_uid, mount.attribute_uid, ts, actor
	)
	return (sig == expected)
```
