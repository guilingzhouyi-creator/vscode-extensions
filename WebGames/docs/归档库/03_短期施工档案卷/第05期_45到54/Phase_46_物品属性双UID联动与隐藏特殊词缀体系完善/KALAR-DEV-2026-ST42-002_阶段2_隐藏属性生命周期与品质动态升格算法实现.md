  2. **数值结算非负硬截断不变量**：A 类增量加成与 B 类减量削弱在结算目标属性时，若相减产生负数，算法必须严格执行 $\text{EffectiveStat} = \max(0.0, \text{CalculatedStat})$；

### 2.1 C 类隐藏属性鉴定求解器 (`ItemAttributeAppraisalSolver`)

```gdscript
# 模块路径: res://backend/domains/item_attributes/item_attribute_appraisal_solver.gd
class_name ItemAttributeAppraisalSolver
extends RefCounted

## 鉴定结果 DTO
class AppraisalResultDTO extends RefCounted:
	var success: bool = false
	var error_code: String = ""                  # "NOT_HIDDEN" / "SKILL_INSUFFICIENT" / "LORE_MISSING" / "ALREADY_APPRAISED"
	var mount_instance: ItemAttributeMountInstance = null
	var is_promoted_tier: bool = false
	var new_tier_rank: int = 1
	var audit_token: String = ""

	func to_dto() -> Dictionary:
		return {
			"success": success,
			"error_code": error_code,
			"mount": mount_instance.to_dto() if mount_instance else {},
			"is_promoted_tier": is_promoted_tier,
			"new_tier_rank": new_tier_rank,
			"audit_token": audit_token,
		}

static func appraise_mount_attribute(
	mount: ItemAttributeMountInstance,
	attr_def: ItemAttributeDefinition,
	appraiser_ctx: Dictionary,
	rng: DeterministicRNG
) -> AppraisalResultDTO:
	var result := AppraisalResultDTO.new()
	result.mount_instance = mount

	if mount == null or attr_def == null:
		result.success = false
		result.error_code = "INVALID_MOUNT_OR_DEF"
		return result

	if mount.category != ItemAttributeDefinition.AttributeCategory.SPECIAL_AFFIX:
		result.success = false
		result.error_code = "NOT_SPECIAL_AFFIX"
		return result

	if mount.visibility == ItemAttributeMountInstance.VisibilityState.APPRAISED or mount.is_active:
		result.success = false
		result.error_code = "ALREADY_APPRAISED"
		return result

	# 1. 鉴定前置条件校验 (技能等级 / 必备知识 / 鉴定卷轴)
	var req := attr_def.appraisal_requirement
	var min_skill: int = int(req.get("min_skill_level", 1))
	var required_lore: String = str(req.get("required_lore", ""))
	var appraiser_skill: int = int(appraiser_ctx.get("appraisal_skill_level", 0))
	var appraiser_lores: Array = appraiser_ctx.get("unlocked_lores", [])

	if appraiser_skill < min_skill:
		result.success = false
		result.error_code = "SKILL_LEVEL_INSUFFICIENT"
		return result

	if not required_lore.is_empty() and not appraiser_lores.has(required_lore):
		result.success = false
		result.error_code = "REQUIRED_LORE_MISSING"
		return result

	# 2. 状态跃迁与激活
	mount.visibility = ItemAttributeMountInstance.VisibilityState.APPRAISED
	mount.is_active = true

	# 3. 构造服务端防伪留痕凭证
	var timestamp := int(Time.get_unix_time_from_system())
	var actor_id := str(appraiser_ctx.get("actor_id", "ANONYMOUS"))
	var signature := _generate_audit_signature(mount.item_uid, mount.attribute_uid, timestamp, actor_id)

	mount.appraisal_audit = {
		"is_appraised": true,
		"appraised_timestamp": timestamp,
		"appraiser_actor_id": actor_id,
		"appraiser_skill_id": str(appraiser_ctx.get("skill_id", "DEFAULT_APPRAISAL")),
		"audit_signature": signature
	}

	result.success = true
	result.audit_token = signature
	return result

static func _generate_audit_signature(item_uid: String, attr_uid: String, ts: int, actor: String) -> String:
	var raw := "%s:%s:%d:%s:KALAR_SECURE_SALT" % [item_uid, attr_uid, ts, actor]
	return raw.sha256_text().substr(0, 16)
```

### 2.2 属性综合结算与非负截断求解器 (`ItemAttributeEvaluationSolver`)

```gdscript
# 模块路径: res://backend/domains/item_attributes/item_attribute_evaluation_solver.gd
class_name ItemAttributeEvaluationSolver
extends RefCounted

## 综合计算物品面板生效属性（落实非负下界截断与互斥标签守卫）
static func evaluate_effective_stats(
	item_uid: String,
	base_stats: Dictionary,
	mounts: Array[ItemAttributeMountInstance],
	defs_catalog: Dictionary # attribute_uid -> ItemAttributeDefinition
) -> Dictionary:
	var final_stats := base_stats.duplicate(true)
	var active_increments: Dictionary = {}
	var active_decrements: Dictionary = {}
	var active_c_affixes: Array[String] = []
	var seen_conflict_tags: Dictionary = {}

	# 1. 遍历挂载列表，过滤有效且激活的属性
	for mount in mounts:
		if not mount.is_active or mount.visibility == ItemAttributeMountInstance.VisibilityState.HIDDEN:
			continue

		var def: ItemAttributeDefinition = defs_catalog.get(mount.attribute_uid, null)
		if def == null:
			continue

		# 冲突标签校验：若已存在相同冲突标签，拒绝重复生效
		var conflict_hit := false
		for ctag in def.conflict_tags:
			if seen_conflict_tags.has(ctag):
				conflict_hit = true
				break
		if conflict_hit:
			continue

		for ctag in def.conflict_tags:
			seen_conflict_tags[ctag] = true

		var stat_name := def.target_stat
		var val := mount.current_value if mount.current_value != 0.0 else def.base_value
		# 单属性上界钳制
		val = min(val, def.value_max)

		match def.category:
			ItemAttributeDefinition.AttributeCategory.INCREMENT:
				active_increments[stat_name] = active_increments.get(stat_name, 0.0) + val
			ItemAttributeDefinition.AttributeCategory.DECREMENT:
				active_decrements[stat_name] = active_decrements.get(stat_name, 0.0) + val
			ItemAttributeDefinition.AttributeCategory.SPECIAL_AFFIX:
				active_c_affixes.append(mount.attribute_uid)
				active_increments[stat_name] = active_increments.get(stat_name, 0.0) + val

	# 2. 合并计算：基础值 + A类/C类增量 - B类减量
	var all_stat_names: Dictionary = {}
	for k in final_stats.keys(): all_stat_names[k] = true
	for k in active_increments.keys(): all_stat_names[k] = true
	for k in active_decrements.keys(): all_stat_names[k] = true

	for sname in all_stat_names.keys():
		var base_val: float = float(final_stats.get(sname, 0.0))
		var inc_val: float = float(active_increments.get(sname, 0.0))
		var dec_val: float = float(active_decrements.get(sname, 0.0))
		var calc_val := base_val + inc_val - dec_val

		# 核心铁律：属性下界统一不得低于 0.0，强制截断
		final_stats[sname] = max(0.0, calc_val)

	return {
		"item_uid": item_uid,
		"final_stats": final_stats,
		"increments": active_increments,
		"decrements": active_decrements,
		"active_c_affixes": active_c_affixes
	}
```

### 2.3 品质动态升格规则引擎 (`ItemQualityPromotionSolver`)

```gdscript
# 模块路径: res://backend/domains/item_attributes/item_quality_promotion_solver.gd
class_name ItemQualityPromotionSolver
extends RefCounted

## 依据已鉴定的特殊词缀评分，动态重估物品最终品阶
static func evaluate_quality_promotion(
	original_tier_rank: int,
	mounts: Array[ItemAttributeMountInstance],
	defs_catalog: Dictionary
) -> Dictionary:
	var total_affix_score := 0.0
	var active_c_count := 0
	var max_c_limit := GameConfig.get_int("domains.item_attributes", "global_limits/max_special_affixes_per_item", 3)

	for mount in mounts:
		if mount.is_active and mount.visibility == ItemAttributeMountInstance.VisibilityState.APPRAISED:
			var def: ItemAttributeDefinition = defs_catalog.get(mount.attribute_uid, null)
			if def and def.category == ItemAttributeDefinition.AttributeCategory.SPECIAL_AFFIX:
				active_c_count += 1
				if active_c_count <= max_c_limit:
					total_affix_score += def.affix_score

	# 从配置表读取品阶升级阈值阶梯
	var thresholds: Dictionary = GameConfig.get_dict("domains.item_attributes", "quality_upgrade_thresholds", {
		"tier_2": 50.0,   # 精良
		"tier_3": 120.0,  # 稀有
		"tier_4": 260.0,  # 史诗
		"tier_5": 500.0,  # 传说
		"tier_6": 1000.0  # 神圣
	})

	var target_tier := original_tier_rank
	if total_affix_score >= float(thresholds.get("tier_6", 1000.0)):
		target_tier = max(target_tier, 6)
	elif total_affix_score >= float(thresholds.get("tier_5", 500.0)):
		target_tier = max(target_tier, 5)
	elif total_affix_score >= float(thresholds.get("tier_4", 260.0)):
		target_tier = max(target_tier, 4)
	elif total_affix_score >= float(thresholds.get("tier_3", 120.0)):
		target_tier = max(target_tier, 3)
	elif total_affix_score >= float(thresholds.get("tier_2", 50.0)):
		target_tier = max(target_tier, 2)

	return {
		"original_tier_rank": original_tier_rank,
		"effective_tier_rank": target_tier,
		"is_promoted": (target_tier > original_tier_rank),
		"total_affix_score": total_affix_score,
		"active_c_affixes_count": active_c_count
	}
```
