## 二、 角色创建服务领域聚合根 (`CharacterCreationService`)

```gdscript
# 模块路径: res://backend/domains/character_creation/character_creation_service.gd
class_name CharacterCreationService
extends RefCounted

## 提交角色创建主入口
static func process_character_creation(
	request: CharacterCreationRequestDTO,
	slot: SaveSlotStateDTO,
	canary_flags: Dictionary
) -> Dictionary:
	if request == null or slot == null:
		return { "success": false, "error_code": "INVALID_INPUT", "message": "请求或档位数据为空" }

	# 1. 种族可用性与灰度校验
	var races_dict: Dictionary = GameConfig.get_dict("domains.character_creation", "races_catalog", {})
	var race_data: Dictionary = races_dict.get(request.selected_race_id, {})
	if race_data.is_empty():
		return { "success": false, "error_code": "RACE_NOT_FOUND", "message": "目标种族不存在: %s" % request.selected_race_id }

	var race_def := RaceDefinitionDTO.from_dto(race_data)

	# 灰度拦截非全量种族
	if not race_def.is_enabled:
		var tag := race_def.canary_feature_tag
		var flag = canary_flags.get(tag, null)
		var is_open: bool = CanaryRolloutSolver.is_feature_enabled_for_account(flag, request.account_id)
		if not is_open:
			return {
				"success": false,
				"error_code": "RACE_CANARY_RESTRICTED",
				"message": "该种族当前处于灰度封测阶段，未向当前账号开放"
			}

	# 2. 状态机校验（角色名、性别）
	var fsm := CharacterCreationFSM.new()
	fsm.start_creation(request.account_id, slot.slot_id, slot.bound_world_id)

	var name_res := fsm.set_character_name(request.character_name)
	if not name_res.success:
		return name_res

	var gender_res := fsm.set_gender(request.selected_gender, race_def)
	if not gender_res.success:
		return gender_res

	var ext_res := fsm.finalize_extensions(request.extra_custom_fields)
	if not ext_res.success:
		return ext_res

	# 3. 提交持久化与档位绑定
	var bind_res := SaveSlotCharacterBinder.commit_creation_and_bind_slot(request, slot, race_def)
	if not bind_res.success:
		return { "success": false, "error_code": bind_res.error_code, "message": bind_res.message }

	# 4. 发布事件总线流 (EventBus Broadcast)
	var bus := EventBus.get_instance()
	bus.emit_signal("domain_event", "character_creation.completed", {
		"account_id": request.account_id,
		"slot_id": slot.slot_id,
		"character_id": bind_res.created_character_profile.character_id,
		"character_name": bind_res.created_character_profile.character_name
	})

	# 5. 发布开局事件流
	bus.emit_signal("domain_event", "opening_narrative.triggered", bind_res.opening_event.to_dto())

	return {
		"success": true,
		"character_profile": bind_res.created_character_profile,
		"opening_event": bind_res.opening_event
	}
```
