    static func step_ecology(node: WorldMapGraphNode, delta_days: float) -> Dictionary:
        node.regional_mana_density = clamp(node.regional_mana_density + 0.02 * delta_days, 0.1, 2.5)
        var is_tide = node.regional_mana_density >= 2.0
        if is_tide: node.regional_mana_density = 0.8
