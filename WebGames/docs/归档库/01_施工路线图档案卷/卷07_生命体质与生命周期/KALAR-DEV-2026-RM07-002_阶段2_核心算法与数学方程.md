## 一、 核心算法与数学方程实现 (Algorithms & Equations)

```gdscript
# 模块路径: res://core/vol07/stage2_solvers.gd
class_name BioLifeCycleEngine extends RefCounted:
    static func evaluate_normalized_age(sheet: CharacterPhysiologySheet) -> float:
        return clamp(sheet.raw_chronological_age / max(0.1, sheet.lifespan_scale), 0.0, 100.0)
    static func get_apparent_stage(sheet: CharacterPhysiologySheet) -> String:
        var norm = evaluate_normalized_age(sheet)
        var sf_bonus = max(0.0, (sheet.somatic_function - 50.0) * 0.3)
        var app_age = max(18.0, norm - sf_bonus)
        if app_age < 50.0: return "ADULT"
        elif app_age < 85.0: return "MIDDLE_AGE"
        else: return "GREAT_LIMIT"
```
