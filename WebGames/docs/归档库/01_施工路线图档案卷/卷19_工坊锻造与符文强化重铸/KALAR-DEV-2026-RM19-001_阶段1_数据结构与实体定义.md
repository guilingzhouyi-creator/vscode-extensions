# 模块路径: res://core/vol19/stage1_数据结构与实体定义.gd
class_name EquipmentEnhancementSolver extends RefCounted:
    static func enhance(tier: int, protected: bool) -> Dictionary:
        if tier >= 15: return { "success": false, "tier": 15 }
        var rate = 1.0 if tier < 2 else 0.5
        var ok = randf() < rate
        return { "success": ok, "tier": (tier + 1) if ok else (tier if protected else max(0, tier - 1)) }
