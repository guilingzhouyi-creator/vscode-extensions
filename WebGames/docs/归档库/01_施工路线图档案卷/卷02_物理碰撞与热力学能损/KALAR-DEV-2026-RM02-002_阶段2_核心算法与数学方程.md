## 一、 核心算法与数学方程实现 (Algorithms & Equations)

```gdscript
# 模块路径: res://core/vol02/stage2_solvers.gd
class_name PhysicsThermoSolver extends RefCounted:
    static func resolve_impact(attacker_sf: float, m: float, v: float, theta: float, k_mom: float, k_edge: float, armor: float) -> float:
        var sf_scale = attacker_sf / 100.0
        var e_kinetic = 0.5 * m * pow(v, 2) * cos(theta) * k_mom * sf_scale
        return max(0.0, (e_kinetic * k_edge) - armor)

    static func resolve_mana_phase_loss(raw_input: float, sentence_count: int, is_primal: bool, env_density: float) -> Dictionary:
        if is_primal: return { "manifested": raw_input * 0.97, "loss": 0.03 }
        var step1_loss = clamp(0.50 - 0.25 * (env_density / 2.0), 0.25, 0.50)
        var step2_loss = clamp(0.10 + float(sentence_count) * 0.005, 0.10, 0.30)
        var out_energy = raw_input * (1.0 - step1_loss) * (1.0 - step2_loss)
        return { "manifested": out_energy, "loss": 1.0 - (out_energy / raw_input) }
```
