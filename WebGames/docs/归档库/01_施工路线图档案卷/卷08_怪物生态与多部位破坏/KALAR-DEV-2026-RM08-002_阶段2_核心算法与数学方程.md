class_name MonsterPartDamageSolver extends RefCounted:
    static func apply_damage(part: Dictionary, dmg: float) -> Dictionary:
        part.durability = max(0.0, part.durability - dmg)
        if part.durability <= 0.0: part.is_severed = true
        return { "is_severed": part.is_severed, "ap_penalty": -4 if part.is_severed else 0 }
