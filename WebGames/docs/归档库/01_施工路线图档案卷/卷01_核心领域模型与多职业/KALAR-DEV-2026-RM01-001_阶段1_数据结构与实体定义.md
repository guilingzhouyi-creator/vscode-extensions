```gdscript
# 模块路径: res://core/vol01/stage1_entities.gd
class_name ItemEntity extends RefCounted:
    var item_id: String
    var template_id: String
    var custom_name: String
    var category: String # WEAPON_BLADE / WEAPON_BLUNT / ARMOR_EQUIPMENT / CONSUMABLE_RATION / MANA_CRYSTAL / MANUSCRIPT_GRIMOIRE
    var mass_kg: float = 1.0
    var volume_slots: int = 1
    var durability_current: float = 100.0
    var durability_max: float = 100.0
    var material_hardness: float = 10.0
    var combat_metrics: Dictionary = { "edge_sharpness": 1.0, "reach_meters": 1.2, "effective_armor": 0.0 }
    var imprinted_runes: Array[String] = []
    var market_base_price: int = 10

class_name WearableInventoryAggregate extends RefCounted:
    var baseline_capacity: int = 0
    var equipped_slots: Dictionary = { "HEAD": null, "CHEST": null, "LEGS": null, "MAIN_HAND": null, "OFF_HAND": null }
    var storage_items: Array[ItemEntity] = []
    func calculate_total_capacity(race_bonus: float = 0.0) -> int:
        var total = baseline_capacity
        for slot in equipped_slots.values():
            if slot is ItemEntity and slot.category == "ARMOR_EQUIPMENT": total += slot.volume_slots
        return int(total * (1.0 + race_bonus))
```
