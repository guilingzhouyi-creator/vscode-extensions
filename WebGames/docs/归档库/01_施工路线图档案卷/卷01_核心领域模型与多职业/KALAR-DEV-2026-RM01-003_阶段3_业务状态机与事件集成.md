## 一、 业务状态机与事件总线实现 (State Machines & Event Bus)

```gdscript
# 模块路径: res://core/vol01/stage3_fsm.gd
class_name MultiProfessionEngine extends RefCounted:
    class ProfessionArchetype extends RefCounted:
        var profession_id: String
        var canonical_name: String
        var rank_tier: int = 1

    static func evaluate_active_professions(mastered_asts: Array) -> Array[ProfessionArchetype]:
        var res: Array[ProfessionArchetype] = []
        var phys_count = 0; var magic_count = 0
        for ast in mastered_asts:
            for node in ast.nodes.values():
                if node.node_type == "PHYSICAL_VERB": phys_count += 1
                elif node.node_type == "MAGIC_RUNE": magic_count += 1
        if phys_count >= 10:
            var p = ProfessionArchetype.new(); p.profession_id = "SWORD_MASTER"; p.canonical_name = "剑道宗师"; res.append(p)
        if magic_count >= 10:
            var p = ProfessionArchetype.new(); p.profession_id = "ARCHMAGE"; p.canonical_name = "圣魔导师"; res.append(p)
        if phys_count >= 8 and magic_count >= 8:
            var p = ProfessionArchetype.new(); p.profession_id = "SPELLBLADE"; p.canonical_name = "极效魔剑士"; res.append(p)
        return res
```
