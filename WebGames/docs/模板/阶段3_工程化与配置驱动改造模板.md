## 一、 配置驱动与泛化扩展实现 (Configuration-Driven & Refactoring)

```gdscript
# 模块路径: res://<domain_path>/<component_file>.gd
class_name <Domain>Component extends RefCounted:
    # 1. 配置驱动参数与默认兜底
    var max_capacity: int = 100
    var timeout_sec: float = 15.0
    var config_table_id: String = "domains.<domain>"

    func reload_configuration() -> void:
        # 严格使用类型化取值器，杜绝硬编码数值与文案
        max_capacity = GameConfig.get_int(config_table_id, "limits/max_capacity", 100)
        timeout_sec = GameConfig.get_float(config_table_id, "timers/timeout_sec", 15.0)

    # 2. 通用扩展与策略注入
    func process_with_strategy(input_item: Dictionary, strategy: Callable) -> Dictionary:
        if strategy.is_valid():
            return strategy.call(input_item)
        return input_item
```
