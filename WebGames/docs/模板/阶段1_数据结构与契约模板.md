## 一、 数据结构设计与代码契约 (Data Structures & Code Contracts)

```gdscript
# 模块路径: res://<domain_path>/<entity_file>.gd
class_name <Domain>Entity extends RefCounted:
    # 1. 核心属性与字段定义
    var id: String = ""
    var state: Dictionary = {}
    var version: int = 1

    # 2. DTO 数据交换契约
    func to_dto() -> Dictionary:
        return {
            "id": id,
            "state": state.duplicate(true),
            "version": version
        }

    static func from_dto(data: Dictionary) -> <Domain>Entity:
        var entity = <Domain>Entity.new()
        entity.id = data.get("id", "")
        entity.state = data.get("state", {}).duplicate(true)
        entity.version = data.get("version", 1)
        return entity
```
