### 3. refresh_boss_parts 循环内零瞬态分配改造
```gdscript
func refresh_boss_parts() -> void:
	var view: CombatView = _view
	if not view._boss_parts_list:
		return

	var existing_children := view._boss_parts_list.get_children()
	var parts_count := view.boss_parts.size()

	for i in range(parts_count):
		var part: Dictionary = view.boss_parts[i]
		var row: HBoxContainer
		if i < existing_children.size():
			row = existing_children[i] as HBoxContainer
			row.visible = true
		else:
			row = _create_part_row(view)
			view._boss_parts_list.add_child(row)
		_update_part_row(row, part)

	for i in range(parts_count, existing_children.size()):
		existing_children[i].visible = false
```
