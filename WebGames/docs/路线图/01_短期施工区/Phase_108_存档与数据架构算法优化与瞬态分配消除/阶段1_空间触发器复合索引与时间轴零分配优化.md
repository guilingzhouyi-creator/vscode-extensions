## 一、 数据结构设计与算法契约 (Data Structures & Algorithm Contracts)

### 1. 空间触发器二维嵌套索引结构
在 `multi_entity_trigger_registry.gd` 中，重构状态容器：

```gdscript
# 模块路径: res://backend/domains/spatial_movement/multi_entity_trigger_registry.gd
class_name MultiEntityTriggerRegistry
extends RefCounted

## 二维状态矩阵: trigger_id -> { entity_id: int(TriggerState) }
var _matrix: Dictionary = {}

func get_state(trigger_id: String, entity_id: String) -> int:
	var t_dict: Variant = _matrix.get(trigger_id, null)
	if t_dict == null:
		return SpatialTriggerFSM.TriggerState.OUTSIDE
	return t_dict.get(entity_id, SpatialTriggerFSM.TriggerState.OUTSIDE)

func update_state(trigger_id: String, entity_id: String, new_state: int) -> int:
	var t_dict: Dictionary
	if not _matrix.has(trigger_id):
		t_dict = {}
		_matrix[trigger_id] = t_dict
	else:
		t_dict = _matrix[trigger_id]
	var prev: int = t_dict.get(entity_id, SpatialTriggerFSM.TriggerState.OUTSIDE)
	t_dict[entity_id] = new_state
	return prev

func remove_entity(entity_id: String) -> void:
	var empty_triggers: Array = []
	for t_id in _matrix.keys():
		var t_dict: Dictionary = _matrix[t_id]
		t_dict.erase(entity_id)
		if t_dict.is_empty():
			empty_triggers.append(t_id)
	for empty_t in empty_triggers:
		_matrix.erase(empty_t)

func remove_trigger(trigger_id: String) -> void:
	_matrix.erase(trigger_id)

func clear() -> int:
	var count := get_total_tracked_count()
	_matrix.clear()
	return count

func get_total_tracked_count() -> int:
	var total := 0
	for t_dict in _matrix.values():
		total += t_dict.size()
	return total
```
