## 📌 第一性原理溯源指针

* **精准上游规范指针**：[AGENTS.md 工作区全局刚性公理](../../../../../AGENTS.md) §二.1、§二.2；[阶段1_全链路Agent遥测对象池与定长环形轨迹器及异步流控契约设计.md](阶段1_全链路Agent遥测对象池与定长环形轨迹器及异步流控契约设计.md)
* **核心不变量约束断言**：
  - `Inv-INPLACE-INJECTION-1 (原位注入零瞬态分配公理)`：`HomeActionSpaceSolver.resolve_action_space_into` 与 `HudCausalDeltaSolver.compute_deltas_into` 必须直接向传入的预分配池化集合注入数据，禁止在方法内部执行任何 `new()` 实例化或字面量字典构造；
  - `Inv-RING-SLOT-REUSE-2 (环形插槽就地覆盖公理)`：`AgentTrajectoryRingBuffer` 在满容量写入时，必须直接复用对应插槽中已有的 DTO 实例（调用 `reset_state()` 后原位写入），严禁丢弃旧对象创建新对象；
  - `Inv-WORKER-THREAD-ISOLATION-3 (Worker 线程数据气密隔离公理)`：后台 WorkerThreadPool 消费的遥测数据必须通过不可变快照或独立借出槽位隔离，严禁多线程并发读写同一活动的切片对象；
  - `Inv-DETERMINISTIC-SYNC-4 (确定性单线程保序断言)`：当 `_deterministic_mode == true` 时，派发队列必须立即在当前线程调用消费者，禁止产生跨帧与异步延迟。
* **防漂移最高指示**：控制流平铺，圈复杂度 $\text{CC} \le 12$，控制流嵌套深度 $\text{Depth} \le 3$，严禁敏捷工期黑话代号。

## 二、 定长环形缓冲轨迹器核心实现 (`AgentTrajectoryRingBuffer`)

文件路径：`res://backend/domains/world_state/services/agent_trajectory_ring_buffer.gd`

```gdscript
class_name AgentTrajectoryRingBuffer
extends RefCounted

var _capacity: int = 32
var _slots: Array[AgentHudTelemetryPacketDTO] = []
var _head: int = 0
var _count: int = 0

func _init(capacity: int = 32) -> void:
  _capacity = maxi(1, capacity)
  _slots.resize(_capacity)
  for i in range(_capacity):
    _slots[i] = AgentHudTelemetryPacketDTO.new()

## O(1) 取模写入，原位重置复用
func append_packet(packet: AgentHudTelemetryPacketDTO) -> void:
  if packet == null:
    return
  var target_slot: AgentHudTelemetryPacketDTO = _slots[_head]
  # 将传入数据深拷贝/原位同步到 target_slot
  _sync_packet_data(packet, target_slot)
  
  _head = (_head + 1) % _capacity
  _count = mini(_capacity, _count + 1)

## 导出最近 k 步保序时序窗口
func get_trajectory_window(steps: int = 8) -> Array[AgentHudTelemetryPacketDTO]:
  var out: Array[AgentHudTelemetryPacketDTO] = []
  var k: int = mini(steps, _count)
  if k <= 0:
    return out
  out.resize(k)
  var start_idx: int = (_head - k + _capacity) % _capacity
  for i in range(k):
    var slot_idx: int = (start_idx + i) % _capacity
    out[i] = _slots[slot_idx]
  return out
```

## 三、 异步背压派发器核心实现 (`AgentAsyncDispatcher`)

文件路径：`res://backend/domains/world_state/services/agent_async_dispatcher.gd`

```gdscript
class_name AgentAsyncDispatcher
extends RefCounted

var _queue: Array[AgentHudTelemetryPacketDTO] = []
var _capacity: int = 128
var _strategy: int = 0 # DROP_OLDEST
var _deterministic_mode: bool = true
var _sinks: Dictionary = {} # sink_id -> Callable

func dispatch_telemetry(packet: AgentHudTelemetryPacketDTO) -> bool:
  if packet == null:
    return false

  if _deterministic_mode:
    _notify_sinks(packet)
    return true

  # 背压流控：超过容量上限时按策略处理
  if _queue.size() >= _capacity:
    match _strategy:
      0: # DROP_OLDEST
        _queue.pop_front()
      1: # DROP_NEWEST
        return false
      2: # WARN_BLOCK
        # 触发流控告警
        pass

  _queue.append(packet)
  WorkerThreadPool.add_task(_process_queue_task, false, "AgentTelemetryDispatch")
  return true
```
