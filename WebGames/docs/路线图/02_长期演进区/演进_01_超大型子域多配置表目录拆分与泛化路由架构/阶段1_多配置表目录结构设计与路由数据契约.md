## 一、 数据结构设计与代码契约 (Data Structures & Code Contracts)

```gdscript
# 模块路径: res://backend/infrastructure/config_route_dto.gd
class_name ConfigRouteDTO extends RefCounted:

    enum RouteResolutionMode {
        EXACT_SUBTABLE,      # 精确子表匹配 (如 "domains.combat.damage_formulas")
        LEGACY_COMPAT_ALIAS, # 向后兼容单表别名重定向 (如 "domains.combat" -> "mechanics")
        FALLBACK_DEFAULT     # 默认缺省兜底
    }

    var root_layer: String = "domains"          # infrastructure / domains / frontend / narratives
    var domain_id: String = ""                  # 如 "combat"
    var sub_table_id: String = ""               # 如 "damage_formulas"
    var full_table_path: String = ""            # 如 "domains.combat.damage_formulas"
    var resolution_mode: RouteResolutionMode = RouteResolutionMode.EXACT_SUBTABLE

    func to_dto() -> Dictionary:
        return {
            "root_layer": root_layer,
            "domain_id": domain_id,
            "sub_table_id": sub_table_id,
            "full_table_path": full_table_path,
            "resolution_mode": resolution_mode
        }

    static func from_dto(data: Dictionary) -> ConfigRouteDTO:
        var route := ConfigRouteDTO.new()
        route.root_layer = data.get("root_layer", "domains")
        route.domain_id = data.get("domain_id", "")
        route.sub_table_id = data.get("sub_table_id", "")
        route.full_table_path = data.get("full_table_path", "")
        route.resolution_mode = data.get("resolution_mode", RouteResolutionMode.EXACT_SUBTABLE)
        return route
```

## 二、 目录拓扑与多级命名空间规范 (Directory Topology & Namespaces)

```
config/domains/
├── ground_loot.json                       # [常规单表领域] 表名: "domains.ground_loot"
├── organization_guild.json                # [常规单表领域] 表名: "domains.organization_guild"
└── combat/                                # [超大型多表子域] 专有子目录
    ├── mechanics.json                     # 表名: "domains.combat.mechanics"
    ├── damage_formulas.json               # 表名: "domains.combat.damage_formulas"
    └── buff_definitions.json              # 表名: "domains.combat.buff_definitions"
```

| 物理文件路径 | 映射逻辑表名 | 向后兼容别名 | 职责范畴 |
| :--- | :--- | :--- | :--- |
| `config/domains/combat/mechanics.json` | `domains.combat.mechanics` | `domains.combat` | 基础参战实体与动词配置 |
| `config/domains/combat/damage_formulas.json` | `domains.combat.damage_formulas` | - | 侵彻动量与两步相变能损公式 |
| `config/domains/combat/buff_definitions.json` | `domains.combat.buff_definitions` | - | 状态效果与抗性结算字典 |
