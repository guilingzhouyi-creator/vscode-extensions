## 一、 单元测试套件实现：TestConfigSubtablesDomain

### 1. 模块归属与路径
路径：`res://tests/unit/infrastructure/test_config_subtables.gd`  
基类：`TestCase` (`res://tests/test_case.gd`)

### 2. 代码实现方案

```gdscript
# ==============================================================================
# 单元测试：演进 01 多配置表路由与子表解析测试套件
# 文件路径: res://tests/unit/infrastructure/test_config_subtables.gd
# 职责: 验收多级点分命名空间解析、别名兼容重定向、跨子表键寻址与细粒度热重载。
# ==============================================================================

class_name TestConfigSubtablesDomain extends TestCase

func get_domain_name() -> String:
	return "Infrastructure: 多配置表路由与动态子表解析"

func run() -> Dictionary:
	setup()
	run_test(test_subtable_resolution_and_alias)
	run_test(test_subtable_value_access)
	run_test(test_subtable_get_many)
	run_test(test_subtable_hot_reload)
	run_test(test_subtable_key_builder)
	run_test(test_storage_catalog_subtables)
	run_test(test_table_path_multi_level)
	teardown()
	return pack_results()

func test_subtable_resolution_and_alias() -> void:
	# 1. 主表别名映射断言
	var resolved := ConfigRouterEngine.resolve_table_name("domains.combat")
	assert_true(resolved == "domains.combat.mechanics", "domains.combat 别名应重定向至 mechanics")

	# 2. 键前缀分流断言
	var r_kinetic := ConfigRouterEngine.resolve_route("domains.combat", "kinetic/energy_coef")
	assert_true(r_kinetic.resolved_table == "domains.combat.damage_formulas", "kinetic 前缀应路由至 damage_formulas")
	
	var r_buff := ConfigRouterEngine.resolve_route("domains.combat", "buff_types/bleed")
	assert_true(r_buff.resolved_table == "domains.combat.buff_definitions", "buff_types 前缀应路由至 buff_definitions")

	var r_mech := ConfigRouterEngine.resolve_route("domains.combat", "participant_defaults/hp")
	assert_true(r_mech.resolved_table == "domains.combat.mechanics", "常规键应回退至主子表 mechanics")

func test_subtable_value_access() -> void:
	# 验证通过子表全名直接读取
	var direct_coef: float = GameConfig.get_float("domains.combat.damage_formulas", "kinetic/energy_coef", 0.0)
	assert_true(direct_coef == 0.5, "子表全名直读 kinetic/energy_coef 期望为 0.5")

	# 验证通过旧表名兼容读取
	var compat_coef: float = GameConfig.get_float("domains.combat", "kinetic/energy_coef", 0.0)
	assert_true(compat_coef == 0.5, "旧表名兼容路由读取 kinetic/energy_coef 期望为 0.5")

	# 验证主机制键读取
	var hp: float = GameConfig.get_float("domains.combat", "participant_defaults/hp", 0.0)
	assert_true(hp == 100.0, "旧表名读取 participant_defaults/hp 期望为 100.0")

func test_subtable_get_many() -> void:
	var paths := PackedStringArray([
		"participant_defaults/hp",
		"kinetic/energy_coef",
		"nonexistent/deep/path"
	])
	var many := GameConfig.get_many("domains.combat", paths)
	assert_true(many.size() == 2, "get_many 跨子表检索应返回 2 个有效键")
	assert_true(float(many.get("participant_defaults/hp", 0.0)) == 100.0, "hp 应匹配")
	assert_true(float(many.get("kinetic/energy_coef", 0.0)) == 0.5, "energy_coef 应匹配")

func test_subtable_hot_reload() -> void:
	var res := GameConfig.reload_subtable("domains.combat.damage_formulas")
	assert_true(res.get("success", false) == true, "reload_subtable 应当成功")
	assert_true(res.get("subtable", "") == "domains.combat.damage_formulas", "返回子表名自洽")

func test_subtable_key_builder() -> void:
	var key := ConfigRouterEngine.build_subtable_key("domains", "combat", "damage_formulas")
	assert_true(key == "domains.combat.damage_formulas", "点分子表键构建自洽")

func test_storage_catalog_subtables() -> void:
	var catalog = StorageResourceCatalog.get_instance()
	var entry = catalog.resolve_config_table_entry("domains.combat.damage_formulas")
	assert_true(entry != null, "StorageResourceCatalog 应成功索引子表")
	assert_true(str(entry.physical_path) == "res://config/domains/combat/damage_formulas.json", "物理路径索引自洽")

func test_table_path_multi_level() -> void:
	var is_sub := ConfigRouterEngine.is_subtable("domains.combat.mechanics")
	assert_true(is_sub, "三段式表名应识别为子表")
	var not_sub := ConfigRouterEngine.is_subtable("domains.inventory")
	assert_true(not not_sub, "两段式表名不应识别为子表")
```
