## 一、 单元测试套件实现 (Unit Test Suite)

```gdscript
# 模块路径: res://tests/unit/test_config_subtables.gd
class_name TestConfigSubtablesDomain extends RefCounted:

    static func run_all_tests() -> Dictionary:
        var results := []
        results.append(test_subtable_resolution_and_alias())
        results.append(test_subtable_hot_reload())
        
        var all_passed := true
        for r in results:
            if not r.get("passed", false):
                all_passed = false
                break
        return { "domain": "演进 01: 多配置表路由架构", "all_passed": all_passed, "results": results }

    static func test_subtable_resolution_and_alias() -> Dictionary:
        var resolved = ConfigRouterEngine.resolve_table_name("domains.combat")
        var passed = (resolved == "domains.combat.mechanics")
        return { "test": "TC-SUB-01: 子表向后兼容别名重定向", "passed": passed }

    static func test_subtable_hot_reload() -> Dictionary:
        var key = ConfigRouterEngine.build_subtable_key("domains", "combat", "damage_formulas")
        var passed = (key == "domains.combat.damage_formulas")
        return { "test": "TC-SUB-02: 点分子表命名空间键构建", "passed": passed }
```
