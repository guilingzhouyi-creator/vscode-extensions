## 📌 第一性原理溯源指针

* **精准上游规范指针**：[阶段1 数据契约](阶段1_事件订阅契约与HUD快照数据模型设计.md) 一、节（DTO 与契约枚举）与 [`main_hud_view.gd` 公共入口](../../../../frontend/views/main_hud/main_hud_view.gd)（`update_status_snapshot`/`append_terminal_log`/`append_battle_log`）
* **核心不变量约束断言**：`任一 EventBusCore 信道在 HUD 侧至多一个订阅接入点；桥接器只做"事件 → HUD 公共入口"的纯转发与聚合，绝不绕过 HUD 公共 API 直写节点属性；快照聚合输入缺失时回退上一快照而非清零`。
* **业务实现最高指示**：分发必须按阶段3 配置路由表驱动（策略 handler 白名单），杜绝在桥接器中硬编码 `if channel ==` 链；订阅生命周期与视图 `_notification(NOTIFICATION_PREDELETE)` 解绑对齐，防悬挂连接。

## 一、 核心业务细类与算法实现 (Business Implementation & Solver)

```gdscript
# 模块路径: res://frontend/i18n/hud_event_bridge.gd
class_name HudEventBridge extends RefCounted:

    # 单例（与 ViewRouter/EventBusCore 同构：懒装配单例）
    static var _instance: HudEventBridge = null
    static func get_instance() -> HudEventBridge:
        if _instance == null:
            _instance = HudEventBridge.new()
        return _instance

    var _hud_view: MainHUDView = null          # 目标 HUD 视图（由宿主注入）
    var _subscribed: bool = false
    var _sub_tokens: Array[EventBusSubscriptionToken] = []   # EventBusCore 订阅令牌（幂等解绑）
    var _last_snapshot: HudStatusSnapshotDTO = HudStatusSnapshotDTO.new()   # 上一快照（缺输入回退）

    ## 绑定 HUD 视图并建立订阅（幂等；视图宿主 _ready 调用）
    func attach_hud(view: MainHUDView) -> Dictionary:
        if _hud_view == view and _subscribed:
            return { "success": true, "already_attached": true }
        _hud_view = view
        _ensure_subscribed()
        return { "success": true, "subscribed": _subscribed }

    func _ensure_subscribed() -> void:
        if _subscribed:
            return
        var bus := EventBusCore.get_instance()
        _sub_tokens.append(bus.on_channel(EventChannelDefinition.DOMAIN_EVENT_GENERIC, _on_domain_packet))
        _sub_tokens.append(bus.on_channel(EventChannelDefinition.NARRATIVE_EVENT_TEXT, _on_narrative_packet))
        _sub_tokens.append(bus.on_channel(EventChannelDefinition.WORLD_CLOCK_ADVANCED, _on_clock_packet))
        _sub_tokens.append(bus.on_channel(EventChannelDefinition.LOG_RECORD_POSTED, _on_log_packet))
        _subscribed = true

    ## 解绑订阅（HUD 视图 NOTIFICATION_PREDELETE 时调用，防悬挂连接）
    func detach() -> void:
        if not _subscribed:
            return
        for token in _sub_tokens:
            token.unbind()
        _sub_tokens.clear()
        _subscribed = false
        _hud_view = null

    ## 域事件入口：按阶段3 配置路由表分发（handler 白名单，未知通道降级 SYSTEM 战报行）
    func _on_domain_packet(packet: EventPacket) -> void:
        if _hud_view == null:
            return
        var payload: Dictionary = packet.payload_data if packet.payload_data is Dictionary else {}
        var channel := str(payload.get("channel", ""))          # emit_domain_event 通道名内嵌于 payload
        var inner: Dictionary = payload.get("payload", {}) if payload.get("payload", {}) is Dictionary else {}
        var route := _route_for(channel)
        match str(route.get("handler", "")):
            "snapshot_aggregate":
                _aggregate_snapshot(channel, inner)
            "route_to_main_hud":
                _hud_view.call_deferred("trigger_world_entry_route", inner)
            "terminal_narrative":
                _emit_terminal_line(channel, inner, "NARRATIVE")
            "quest_tracker":
                _hud_view.call_deferred("update_quest_tracker_from_event", inner)
            _:
                _emit_terminal_line(channel, inner, "SYSTEM")

    ## 快照聚合：事件载荷 → 上一快照合并 → 注入 HUD 顶栏（缺字段回退上一快照值）
    func _aggregate_snapshot(_channel: String, payload: Dictionary) -> void:
        var merged: Dictionary = _last_snapshot.to_dto()
        merged.merge(payload.get("snapshot", {}), true)   # 事件携带增量快照字段
        var snap := HudStatusSnapshotDTO.from_dto(merged)
        _last_snapshot = snap
        _hud_view.update_status_snapshot(
            snap.hp_current, snap.hp_max, snap.mp_current, snap.mp_max,
            snap.ap_current, snap.ap_max, snap.gold, snap.mana_crystals
        )

    ## 文案型叙事事件 → HUD 聊天/战报终端（复用既有公共入口）
    func _on_narrative_packet(packet: EventPacket) -> void:
        if _hud_view == null:
            return
        var payload: Dictionary = packet.payload_data if packet.payload_data is Dictionary else {}
        var category := str(payload.get("category", "SYSTEM"))
        _hud_view.append_battle_log(category, str(payload.get("text", "")))

    func _on_clock_packet(packet: EventPacket) -> void:
        # 时钟标签同步：委托 HUD 既有 _refresh_clock（本阶段预留，不新增节点直写）
        if _hud_view != null:
            _hud_view.call_deferred("refresh_clock_from_bus")

    func _on_log_packet(packet: EventPacket) -> void:
        if _hud_view != null:
            var record := packet.payload_dto as LogRecordDTO
            var level := record.level if record != null else "info"
            var message := record.message if record != null else str(packet.narrative_key)
            _hud_view.append_terminal_log("LOG", "[%s] %s" % [level, message])

    ## 通道 → 路由（阶段3 配置表读取；配置缺失回退 SYSTEM 战报行降级）
    func _route_for(channel: String) -> Dictionary:
        var routes: Dictionary = GameConfig.get_dict("frontend.views", "hud_event_bridge/channels", {})
        var route: Dictionary = routes.get(channel, {})
        if route.is_empty():
            route = GameConfig.get_dict("frontend.views", "hud_event_bridge/fallback_route", {})
        return route

    ## HUD 终端行输出（分类取色复用 ThemeManager，不新增硬编码色值）
    func _emit_terminal_line(_channel: String, payload: Dictionary, tag: String) -> void:
        var text := str(payload.get("summary", payload.get("text", "")))
        if text.is_empty():
            return
        _hud_view.append_terminal_log(tag, text)
```
