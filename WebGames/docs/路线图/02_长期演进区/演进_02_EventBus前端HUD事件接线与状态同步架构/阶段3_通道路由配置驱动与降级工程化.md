```gdscript
# 模块路径: res://frontend/i18n/hud_event_bridge.gd（阶段2 桥接器内工程化补充段）
# 1. Mock 降级开关：开启时订阅保持但 HUD 驱动置空（骨架演示零回归）
func is_mock_mode() -> bool:
    return GameConfig.get_bool("frontend.views", "hud_event_bridge/mock_mode/enabled", false)

func _on_domain_event(channel: String, payload: Dictionary) -> void:
    if _hud_view == null:
        return
    if is_mock_mode():
        # 降级：仅落可观测日志，不驱动 HUD（骨架态与接线态平滑互切）
        EventBusCore.get_instance().emit_log("info", "HudEventBridge mock: %s" % channel)
        return
    # ...（阶段2 分发逻辑不变）

# 2. 可观测性：桥接器启动/解绑/降级日志（统一经 EventBusCore.emit_log，不走 print）
func _log_bridge(msg: String) -> void:
    EventBusCore.get_instance().emit_log("info", "HudEventBridge: " + msg)
```
