## 一、 自动化测试套件与验收代码 (Test Suite & Verification)

```gdscript
# 模块路径: res://tests/unit/test_object_pool_and_batch_pipeline.gd
# 职责: 高承压对象池与无头批处理流水线端到端验收测试套件

class_name TestObjectPoolAndBatchPipeline extends RefCounted

class DummyPoolableItem extends RefCounted:
	var value: int = 100
	var tag: String = "dirty"
	var is_reset_called: bool = false

	func reset_state() -> void:
		value = 0
		tag = ""
		is_reset_called = true

func run_all_tests() -> Dictionary:
	var results: Array = []
	results.append(_test_pool_preallocation_and_capacity())
	results.append(_test_pool_acquire_reset_state())
	results.append(_test_pool_double_release_prevention())
	results.append(_test_pool_overflow_discard_policy())
	results.append(_test_high_frequency_churn_zero_leak())
	results.append(_test_soa_buffer_alignment())
	results.append(_test_headless_batch_pipeline_zero_allocation())
	results.append(_test_pool_manager_assembly_lifecycle())
	return {
		"suite": "TestObjectPoolAndBatchPipeline",
		"passed": not results.has(false),
		"results": results
	}

## 1. 预分配与容量模型测试
func _test_pool_preallocation_and_capacity() -> bool:
	var cfg := ObjectPoolConfigDTO.new()
	cfg.pool_name = "test_prealloc"
	cfg.preallocate_size = 16
	cfg.max_capacity = 32
	var pool := ObjectPool.new(func(): return DummyPoolableItem.new(), cfg)
	var stats := pool.get_stats()
	return stats.idle_in_pool == 16 and stats.total_allocated == 16

## 2. 借出时状态彻底重置断言
func _test_pool_acquire_reset_state() -> bool:
	var cfg := ObjectPoolConfigDTO.new()
	cfg.pool_name = "test_reset"
	cfg.preallocate_size = 2
	var pool := ObjectPool.new(func(): return DummyPoolableItem.new(), cfg)
	var item: DummyPoolableItem = pool.acquire() as DummyPoolableItem
	item.value = 999
	item.tag = "corrupted"
	pool.release(item)
	var item_reacquired: DummyPoolableItem = pool.acquire() as DummyPoolableItem
	return item_reacquired.value == 0 and item_reacquired.tag == "" and item_reacquired.is_reset_called

## 3. 防重复归还与非法归还拦截断言
func _test_pool_double_release_prevention() -> bool:
	var cfg := ObjectPoolConfigDTO.new()
	cfg.pool_name = "test_double_release"
	var pool := ObjectPool.new(func(): return DummyPoolableItem.new(), cfg)
	var item: DummyPoolableItem = pool.acquire() as DummyPoolableItem
	var first_release := pool.release(item)
	var second_release := pool.release(item) # 非法二次归还
	return first_release == true and second_release == false

## 4. 溢出丢弃机制断言
func _test_pool_overflow_discard_policy() -> bool:
	var cfg := ObjectPoolConfigDTO.new()
	cfg.pool_name = "test_overflow"
	cfg.preallocate_size = 0
	cfg.max_capacity = 2
	cfg.allow_overflow_alloc = true
	var pool := ObjectPool.new(func(): return DummyPoolableItem.new(), cfg)
	var a = pool.acquire()
	var b = pool.acquire()
	var c = pool.acquire() # 溢出第 3 个
	pool.release(a)
	pool.release(b)
	pool.release(c) # 超出 max_capacity=2，应被丢弃
	return pool.get_stats().idle_in_pool == 2

## 5. 高频吞吐 10,000 次借还零泄漏断言
func _test_high_frequency_churn_zero_leak() -> bool:
	var cfg := ObjectPoolConfigDTO.new()
	cfg.pool_name = "test_churn"
	cfg.preallocate_size = 64
	cfg.max_capacity = 128
	var pool := ObjectPool.new(func(): return DummyPoolableItem.new(), cfg)
	for i in range(10000):
		var obj = pool.acquire()
		pool.release(obj)
	var stats := pool.get_stats()
	return stats.active_in_use == 0 and stats.total_acquires == 10000 and stats.total_releases == 10000

## 6. SoA 缓冲区数组长度对齐断言
func _test_soa_buffer_alignment() -> bool:
	var buf := SoAEntityBuffer.new()
	buf.active_count = 100
	buf.entity_ids.resize(100)
	buf.pos_x.resize(100)
	buf.pos_y.resize(100)
	buf.vel_x.resize(100)
	buf.vel_y.resize(100)
	buf.flags.resize(100)
	return buf.is_aligned()

## 7. 无头批处理管线零堆分配与正确性测试
func _test_headless_batch_pipeline_zero_allocation() -> bool:
	var buf := SoAEntityBuffer.new()
	var count := 1000
	buf.active_count = count
	buf.entity_ids.resize(count)
	buf.pos_x.resize(count)
	buf.pos_y.resize(count)
	buf.vel_x.resize(count)
	buf.vel_y.resize(count)
	buf.flags.resize(count)
	for i in range(count):
		buf.pos_x[i] = 0.0
		buf.vel_x[i] = 10.0
		buf.flags[i] = 0
	HeadlessBatchPipeline.step_kinematics(buf, 0.1, 1.0)
	# 步进后 pos_x 应精确累加至 1.0 (0 + 10.0 * 0.1)
	return absf(buf.pos_x[0] - 1.0) < 0.0001 and absf(buf.pos_x[count - 1] - 1.0) < 0.0001

## 8. 全局池管理器生命周期断言
func _test_pool_manager_assembly_lifecycle() -> bool:
	ObjectPoolManager.initialize()
	var pool := ObjectPoolManager.register_pool("dummy_pool", func(): return DummyPoolableItem.new())
	var acquired := ObjectPoolManager.get_pool("dummy_pool") != null
	ObjectPoolManager.teardown()
	var after_teardown := ObjectPoolManager.get_pool("dummy_pool") == null
	return acquired and after_teardown
```
